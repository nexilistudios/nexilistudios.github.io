23CommandResult ServerImpl::set_general_clientId(const DefaultArgs& args)
24{
25 Log::debug("setting::general::client_id");
26 auto& user = args.getUser();
27 auto data = args.getData();
28 if (!user.hasRootAccess())
29 {
30 Log::error("Client needs root access for changing id");
31 return CommandResult::unauthorized;
32 }
33 // We are parsing Command, so remove two bytes from this switch statement.
34 auto payload = Util::removeAmountOfBytesFromVector(data, 3);
35 uint64_t id = Util::convertToType<uint64_t>(payload);
36
37 auto& clients = ClientStorage::getAllClients();
38
39 for (auto c = clients.begin(); c != clients.end(); c++)
40 {
41 auto client = c->get();
42 if (*client == user)
43 {
44 assert(client->hasRootAccess());
45 assert(user.hasRootAccess());
46 client->setId(id);
47 return CommandResult::success;
48 }
49 }
50
51 Log::error("Error in CommandType::set::clientID");
52 return CommandResult::error;
53}

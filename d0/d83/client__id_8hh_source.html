29class GetClientIdCommand : public BaseAPICommand
30{
31public:
32 explicit GetClientIdCommand(uint64_t client_id, std::string udp_token = {})
33 : m_client_id(client_id),
34 m_udp_token(std::move(udp_token))
35 {
36 }
37
38 ReadResult execute(ClientAPI& api, ClientAPI::ClientAPIData& data) override
39 {
40 if (!m_udp_token.empty())
41 {
42 std::array<uint8_t, udp_auth::token_size> parsed{};
43 if (!udp_auth::fromHex(m_udp_token, parsed))
44 return ReadResult::invalid_input;
45 }
46 auto& mtx = data.getRoomsMutex();
47 if (!mtx)
48 return ReadResult::command_execution;
49 std::lock_guard<std::mutex> lock(*mtx);
50
51 api.setClientId(m_client_id);
52 if (!m_udp_token.empty())
53 api.setUdpToken(m_udp_token);
54 data.initialize();
55
56 assert(api.getClientId() == m_client_id);
57 assert(data.isInitialized());
58
59 auto p_data = _Packet::clientIdentification(api);
60 assert(p_data.size() >= 8);
61 assert(Util::convertToType<uint64_t>({p_data.begin(), p_data.begin() + 8}) == api.getClientId());
62
63 return ReadResult::success;
64 }
65
66private:
67 uint64_t m_client_id;
68 std::string m_udp_token;
69};

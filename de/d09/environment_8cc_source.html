28EnvironmentType detectRuntimeType()
29{
30 const char* ci_vars[]{
31 "GITHUB_ACTIONS",
32 "GITLAB_CI",
33 "CI",
34 "TRAVIS",
35 "CIRCLECI",
36 nullptr};
37
38 // Runtime detection first
39 bool is_ci = false;
40 bool is_nix = false;
41
42 for (const char** var = ci_vars; *var; var++)
43 {
44 if (isEnvDef(*var))
45 {
46 is_ci = true;
47 break;
48 }
49 }
50
51 if (isEnvDef("NIX_BUILD_CORES"))
52 {
53 is_nix = true;
54 }
55
56 if (is_ci)
57 return EnvironmentType::ci;
58 if (is_nix)
59 return EnvironmentType::nixos;
60 return EnvironmentType::local;
61}

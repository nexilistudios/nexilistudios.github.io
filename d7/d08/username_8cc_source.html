25CommandResult ServerImpl::set_general_username(const DefaultArgs& args)
26{
27 Log::debug("setting::general::username", args);
28 auto payload = Util::removeAmountOfBytesFromVector(args.getData(), 3);
29 std::string username = Util::convertToString(payload);
30
31 // Setting the username for internal client.
32 auto* client = ClientStorage::getClientById(args.getUser().getId());
33 if (client)
34 {
35 client->setUsername(username);
36 }
37 else
38 {
39 return CommandResult::error;
40 }
41
42 auto data = Command::clientMessageData(CommandType::setting, "username", args.getMessageId(),
43 {{"client_id", boost::json::value(args.getUser().getId())},
44 {"username", boost::json::value(username)}});
45
46 // Notify the room so peers can display this client's username.
47 if (client->getRoomId() != 0)
48 {
49 Command::sendRoomCommand(data, args.getUser(), args.getProtocol());
50 }
51 else
52 {
53 Command::sendMessageToClient(data, args.getUser(), args.getProtocol());
54 }
55
56 return CommandResult::success;
57}

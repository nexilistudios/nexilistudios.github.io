47const std::vector<unsigned char>* pskFromContext(SSL* ssl)
48{
49 SSL_CTX* ctx = SSL_get_SSL_CTX(ssl);
50 if (!ctx)
51 {
52 return nullptr;
53 }
54 return static_cast<const std::vector<unsigned char>*>(SSL_CTX_get_ex_data(ctx, pskExDataIndex()));
55}

57unsigned int pskClientCallback(SSL* ssl,
58 const char* /*hint*/,
59 char* identity,
60 unsigned int max_identity_len,
61 unsigned char* psk,
62 unsigned int max_psk_len)
63{
64 const auto* key = pskFromContext(ssl);
65 if (!key || key->size() > max_psk_len)
66 {
67 return 0;
68 }
69
70 constexpr std::size_t identity_len = sizeof(kPskIdentity); // includes the NUL terminator.
71 if (max_identity_len < identity_len)
72 {
73 return 0;
74 }
75 std::memcpy(identity, kPskIdentity, identity_len);
76 std::memcpy(psk, key->data(), key->size());
77 return static_cast<unsigned int>(key->size());
78}

98 // Domain-separated from the room-message keys: the same password produces

103std::shared_ptr<boost::asio::ssl::context> createPskContext(const std::string& password, bool server)
104{
105 const std::string psk = derivePsk(password);
106 if (psk.size() != kPskLength)
107 {
108 return nullptr;
109 }
110
111 auto ctx = std::make_shared<boost::asio::ssl::context>(
112 server ? boost::asio::ssl::context::tls_server
113 : boost::asio::ssl::context::tls_client);
114
115 ctx->set_options(
116 boost::asio::ssl::context::default_workarounds |
117 boost::asio::ssl::context::no_sslv2 |
118 boost::asio::ssl::context::no_sslv3 |
119 boost::asio::ssl::context::no_tlsv1 |
120 boost::asio::ssl::context::no_tlsv1_1);
121
122 SSL_CTX* native = ctx->native_handle();
123
124 // Restrict the ciphers to PSK key exchange so no certificates are needed,
125 // and to SHA-256 based suites so the 32-byte key always matches the PRF.
126 if (SSL_CTX_set_cipher_list(native, "PSK-AES128-GCM-SHA256:PSK-CHACHA20-POLY1305") != 1)
127 {
128 return nullptr;
129 }
130 if (SSL_CTX_set_ciphersuites(native, "TLS_AES_128_GCM_SHA256:TLS_CHACHA20_POLY1305_SHA256") != 1)
131 {
132 return nullptr;
133 }
134
135 // Store the key against the context so multiple servers/clients with
136 // different passwords can coexist; OpenSSL calls the free function when
137 // the context is destroyed.
138 auto* keyStorage = new std::vector<unsigned char>(psk.begin(), psk.end());
139 SSL_CTX_set_ex_data(native, pskExDataIndex(), keyStorage);
140
141 if (server)
142 {
143 SSL_CTX_set_psk_server_callback(native, pskServerCallback);
144 }
145 else
146 {
147 SSL_CTX_set_psk_client_callback(native, pskClientCallback);
148 }
149
150 return ctx;
151}

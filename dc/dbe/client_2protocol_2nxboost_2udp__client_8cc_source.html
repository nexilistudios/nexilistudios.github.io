99void UDPClient::start()
100{
101 if (m_stopped->load())
102 {
103 Log::debug("BoostUDPClient already stopped, cannot start");
104 return;
105 }
106
107 auto port = getClientAPI()->getBoostUDPServerPortNumber();
108 if (port == 0)
109 {
110 port = Ports::getBoostUDPPort();
111 }
112
113 try
114 {
115 boost::asio::ip::udp::resolver resolver(*m_ioContext);
116 auto endpoints = resolver.resolve(boost::asio::ip::udp::v4(),
117 getClientAPI()->getBoostUDPServerAddress(),
118 std::to_string(port));
119 m_remoteEndpoint = *endpoints.begin();
120
121 m_sendEndpoint = m_remoteEndpoint;
122 m_ioContextThread = std::thread([this]()
123 { m_ioContext->run(); });
124 m_socket.open(boost::asio::ip::udp::v4());
125 // Windows rejects receive_from on an unbound socket (WSAEINVAL).
126 m_socket.bind(boost::asio::ip::udp::endpoint(boost::asio::ip::udp::v4(), 0));
127 m_receiveMessageThread = std::thread(&UDPClient::receiveLoop, this);
128
129 ClientProtocol::start(getType());
130 }
131 catch (const std::exception& e)
132 {
133 Log::error("Failed to start BoostUDPClient: ", e.what());
134 stop();
135 }
136}

138void UDPClient::stop()
139{
140 if (!m_stopped || m_stopped->exchange(true))
141 {
142 Log::debug("BoostUDPClient stop already in progress or completed");
143 return;
144 }
145
146 boost::system::error_code ec;
147 std::unique_lock<std::mutex> lock(*m_mutex);
148 if (m_socket.is_open())
149 {
150 m_socket.shutdown(boost::asio::ip::udp::socket::shutdown_both, ec);
151
152 if (m_socket.cancel(ec))
153 {
154 Log::error("Error cancelling socket: ", ec.message());
155 }
156
157 if (m_socket.close(ec))
158 {
159 Log::error("Error closing socket: ", ec.message());
160 }
161 }
162 lock.unlock();
163
164 if (m_ioContext)
165 {
166 Log::debug("BoostUDPClient io context stopped");
167 m_ioContext->stop();
168 }
169
170 if (m_ioContextThread.joinable())
171 {
172 Log::debug("BoostUDPClient io context thread stopped");
173 m_ioContextThread.join();
174 }
175
176 if (m_receiveMessageThread.joinable())
177 {
178 Log::debug("BoostUDPClient receive message thread stopped");
179 m_receiveMessageThread.join();
180 }
181
182 if (m_pendingSendsMutex)
183 {
184 std::lock_guard<std::mutex> pending_send_lock(*m_pendingSendsMutex);
185 for (auto& pair : m_pendingSends)
186 {
187 try
188 {
189 pair.second->set_exception(
190 std::make_exception_ptr(std::runtime_error("Connection closed")));
191 }
192 catch (...)
193 {
194 }
195 }
196 m_pendingSends.clear();
197 }
198
199 Log::debug("BoostUDPClient stopped");
200}

202void UDPClient::receiveLoop()
203{
204 while (m_socket.is_open())
205 {
206 boost::system::error_code recvEc;
207 boost::asio::ip::udp::endpoint senderEndpoint;
208 std::size_t receivedBytes = m_socket.receive_from(boost::asio::buffer(m_receiveBuffer), senderEndpoint, 0, recvEc);
209
210 if (recvEc)
211 {
212 if (m_stopped->load() || recvEc == boost::asio::error::operation_aborted ||
213 recvEc == boost::asio::error::bad_descriptor)
214 {
215 break;
216 }
217 Log::error("BoostUDPClient receive failed: ", recvEc.message());
218 continue;
219 }
220
221 if (m_stopped->load())
222 {
223 break;
224 }
225
226 if (receivedBytes == 0 || senderEndpoint != m_sendEndpoint)
227 {
228 continue;
229 }
230
231 Log::info("Received bytes: ", receivedBytes);
232
233 auto byteVector = Util::convertToByteVector(m_receiveBuffer.data(), receivedBytes);
234 if (udp_auth::hasPrefix(byteVector, udp_auth::challenge))
235 {
236 if (byteVector.size() == 4 + udp_auth::cookie_size)
237 {
238 nx_data pending;
239 {
240 std::lock_guard<std::mutex> lock(*m_mutex);
241 pending = m_pendingAuth;
242 }
243 if (!pending.empty())
244 {
245 nx_data cookie(byteVector.begin() + 4, byteVector.end());
246 auto auth = udp_auth::makeAuthentication(cookie, pending);
247 m_socket.send_to(boost::asio::buffer(auth), m_sendEndpoint);
248 }
249 }
250 continue;
251 }
252 auto result = ClientProtocol::getClientAPI()->readMessage(byteVector);
253 if (result == ReadResult::success)
254 {
255 Log::debug("BoostUDPClient: Message read successfully");
256 }
257 else
258 {
259 Log::error("Received unexpected message: ");
260 Util::debugUint8Vector(byteVector);
261 }
262 }

265void UDPClient::sendMessage(const nx_data& payload)
266{
267 if (m_stopped->load())
268 {
269 Log::error("UDPClient::send(): client is stopped");
270 return;
271 }
272
273 if (!m_socket.is_open())
274 {
275 Log::error("UDPClient::send(): socket is not open");
276 return;
277 }
278
279 try
280 {
281 nx_data packet;
282 auto token = getClientAPI()->getUdpToken();
283 if (token.empty())
284 {
285 std::lock_guard<std::mutex> lock(*m_mutex);
286 m_pendingAuth = payload;
287 packet = udp_auth::makeHello();
288 }
289 else
290 {
291 packet = udp_auth::makeSession(payload, token);
292 }
293 if (packet.empty())
294 return;
295 size_t bytes_sent = m_socket.send_to(boost::asio::buffer(packet), m_sendEndpoint);
296 Log::debug("Sent ", bytes_sent, " bytes to server");
297 }
298 catch (const boost::system::system_error& e)
299 {
300 Log::error("Failed to send message: ", e.what());
301 }
302 catch (const std::exception& e)
303 {
304 Log::error("Exception in sendMessage: ", e.what());
305 }

313std::future<void> UDPClient::sendMessageAsync(const nx_data& message)
314{
315 if (m_stopped->load())
316 {
317 std::promise<void> promise;
318 promise.set_exception(
319 std::make_exception_ptr(std::runtime_error("Client is stopped")));
320 return promise.get_future();
321 }
322
323 if (!m_socket.is_open())
324 {
325 std::promise<void> promise;
326 promise.set_exception(
327 std::make_exception_ptr(std::runtime_error("Socket is not open")));
328 return promise.get_future();
329 }
330
331 // Extract message ID (second 8 bytes) for tracking.
332 uint64_t messageId = 0;
333 if (message.size() >= 16)
334 {
335 std::memcpy(&messageId, message.data() + 8, sizeof(uint64_t));
336 }
337
338 auto promise = std::make_shared<std::promise<void>>();
339 auto future = promise->get_future();
340
341 auto packet = udp_auth::makeSession(message, getClientAPI()->getUdpToken());
342 if (packet.empty())
343 {
344 promise->set_exception(std::make_exception_ptr(std::runtime_error("UDP session is not authenticated")));
345 return future;
346 }
347 if (messageId != 0)
348 {
349 std::lock_guard<std::mutex> lock(*m_pendingSendsMutex);
350 m_pendingSends[messageId] = promise;
351 }
352 auto messageCopy = std::make_shared<nx_data>(std::move(packet));
353
354 std::thread([this, messageCopy, messageId]()
355 {
356 try
357 {
358 size_t bytes_sent = m_socket.send_to(
359 boost::asio::buffer(*messageCopy), m_sendEndpoint);
360 Log::debug("AsyncSent ", bytes_sent, " bytes to server");
361
362 if (messageId != 0)
363 {
364 std::lock_guard<std::mutex> lock(*m_pendingSendsMutex);
365 auto it = m_pendingSends.find(messageId);
366 if (it != m_pendingSends.end())
367 {
368 it->second->set_value();
369 m_pendingSends.erase(it);
370 }
371 }
372 }
373 catch (const std::exception& e)
374 {
375 Log::error("Async send failed: ", e.what());
376 if (messageId != 0)
377 {
378 std::lock_guard<std::mutex> lock(*m_pendingSendsMutex);
379 auto it = m_pendingSends.find(messageId);
380 if (it != m_pendingSends.end())
381 {
382 it->second->set_exception(std::current_exception());
383 m_pendingSends.erase(it);
384 }
385 }
386 } })
387 .detach();
388
389 return future;
390}

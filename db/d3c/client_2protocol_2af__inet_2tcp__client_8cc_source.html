33TCPClient::TCPClient(ClientAPI& clientApi)
34 : NxClass("client::af_inet::TCPClient"),
35 NXProtocol(),
36 ClientProtocol(&clientApi),
37 m_clientSocket(-1)
38{
39 memset(&m_serverAddr, 0, sizeof(m_serverAddr));
40 m_serverAddr.sin_family = AF_INET;
41
42 uint16_t serverPort = clientApi.getInetTCPServerPortNumber();
43 if (serverPort == 0)
44 {
45 // No explicit port configured, fall back to the default inet TCP port.
46 serverPort = Ports::getInetTCPPort();
47 }
48 m_serverAddr.sin_port = htons(serverPort);
49 m_serverAddr.sin_addr.s_addr = inet_addr(clientApi.getInetTCPServerAddress().c_str());
50}

95void TCPClient::start()
96{
97 if (m_clientSocket == -1)
98 {
99 m_clientSocket = socket(AF_INET, SOCK_STREAM, 0);
100 if (m_clientSocket == -1)
101 {
102 Log::critical(header(), "TCPClient: Error creating socket");
103 return;
104 }
105 }
106
107 if (!connectToServer())
108 {
109 Log::error(header(), "TCPClient: Error connecting to server");
110 return;
111 }
112
113 m_receiveThread = std::thread([this]()
114 { receiveLoop(); });
115
116 try
117 {
118 ClientProtocol::start(getType());
119 }
120 catch (const std::exception& e)
121 {
122 Log::error("Failed to start TCPClient: ", e.what());
123 stop();
124 }
125}

127void TCPClient::stop()
128{

130 {
131 // Shutdown before close so that a receive thread blocked in read()
132 // is woken up with EOF instead of hanging forever.

161std::future<void> TCPClient::sendMessageAsync(const nx_data& message)
162{
163 if (m_clientSocket == -1)
164 {
165 std::promise<void> promise;
166 promise.set_exception(std::make_exception_ptr(std::runtime_error("Socket is not open")));
167 return promise.get_future();
168 }
169
170 auto promise = std::make_shared<std::promise<void>>();
171 auto future = promise->get_future();
172
173 auto request = authenticate(message);
174 if (request.empty())
175 {
176 promise->set_exception(std::make_exception_ptr(std::runtime_error("Cannot authenticate request")));
177 return future;
178 }
179 auto messageCopy = std::make_shared<nx_data>(std::move(request));
180
181 std::thread([this, messageCopy, promise]()
182 {
183 try
184 {
185 if (m_clientSocket == -1)
186 {
187 throw std::runtime_error("Socket is not open");
188 }
189
190 if (!sendFrame(reinterpret_cast<const char*>(messageCopy->data()), messageCopy->size()))
191 {
192 throw std::runtime_error(std::string("Failed to send message: ") + std::strerror(errno));
193 }
194 promise->set_value();
195 }
196 catch (...)
197 {
198 promise->set_exception(std::current_exception());
199 } })
200 .detach();
201
202 return future;
203}

205bool TCPClient::sendFrame(const char* data, size_t dataSize)
206{
207 if (m_clientSocket == -1)
208 {
209 Log::error(header(), "TCPClient: Error sending message, socket is not open");
210 return false;
211 }
212
213 // Write the frame size as a 4-byte big-endian value.
214 uint32_t frameSize = static_cast<uint32_t>(dataSize);
215 uint32_t networkSize = htonl(frameSize);
216
217 ssize_t frameBytes = send(m_clientSocket, &networkSize, sizeof(networkSize), MSG_NOSIGNAL);
218 if (frameBytes <= 0)
219 {
220 Log::error(header(), "TCPClient: Error sending frame size");
221 return false;
222 }
223
224 size_t bytesSent = 0;
225 while (bytesSent < dataSize)
226 {
227 ssize_t sent = send(m_clientSocket, data + bytesSent, dataSize - bytesSent, MSG_NOSIGNAL);
228 if (sent <= 0)
229 {
230 Log::error(header(), "TCPClient: Error sending message");
231 return false;
232 }
233 bytesSent += static_cast<size_t>(sent);
234 }
235
236 return true;
237}

239ssize_t TCPClient::receive(char* buffer, size_t bufferSize)
240{
241 uint32_t networkSize = 0;
242 ssize_t frameBytes = read(m_clientSocket, &networkSize, sizeof(networkSize));
243 if (frameBytes <= 0)
244 {
245 return -1;
246 }
247
248 if (frameBytes != static_cast<ssize_t>(sizeof(networkSize)))
249 {
250 Log::error(header(), "TCPClient: Error receiving frame size");
251 return -1;
252 }
253
254 uint32_t frameSize = ntohl(networkSize);
255
256 if (frameSize > bufferSize)
257 {
258 Log::error(header(), "TCPClient: Frame size exceeds buffer size");
259 return -1;
260 }
261
262 size_t bytesReceived = 0;
263 while (bytesReceived < frameSize)
264 {
265 ssize_t bytesRead = read(m_clientSocket, buffer + bytesReceived, frameSize - bytesReceived);
266 if (bytesRead <= 0)
267 {
268 return -1;
269 }
270 bytesReceived += static_cast<size_t>(bytesRead);
271 }
272
273 return static_cast<ssize_t>(bytesReceived);
274}

276void TCPClient::receiveLoop()
277{
278 while (true)
279 {
280 char buffer[NEXILIS_BUFFER];
281 ssize_t receivedBytes = receive(buffer, sizeof(buffer));
282 if (receivedBytes <= 0)
283 {
284 break;
285 }
286
287 nx_data receivedData(buffer, buffer + receivedBytes);
288 getClientAPI()->readMessage(receivedData);
289 }
290}

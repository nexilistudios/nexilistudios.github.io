481 struct ChatSnapshot
482 {
483 uint64_t sender_id = 0;
484 std::string message;
485 };
486
487 std::vector<ChatSnapshot> getChatSnapshot(uint64_t room_id)
488 {
489 std::lock_guard<std::mutex> lock(*m_clientAPIData.getRoomsMutex());
490 std::vector<ChatSnapshot> result;
491 for (const auto& room : m_clientAPIData.getCurrentlyActiveRooms())
492 {
493 if (room.getId() != room_id)
494 continue;
495
496 const auto& messages = room.getMessages();
497 std::transform(messages.begin(), messages.end(), std::back_inserter(result),
498 [](const Room::Communication& communication)
499 { return ChatSnapshot{communication.getSenderId(), communication.getPayload()}; });
500 break;
501 }
502 return result;
503 }

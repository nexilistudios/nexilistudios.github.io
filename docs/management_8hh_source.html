29 RoomManagementCommand(std::string action, uint64_t room_id, uint64_t client_id,
30 std::string room_name = "", uint64_t room_context = 0,

32 : m_action(action), m_room_id(room_id), m_client_id(client_id),

38 ReadResult execute(ClientAPI& api, ClientAPI::ClientAPIData& data) override
39 {
40 auto& mtx = data.getRoomsMutex();
41 std::lock_guard<std::mutex> lock(*mtx);
42
43 auto& rooms = data.getCurrentlyActiveRooms();
44
45 if (m_action == "join")
46 {
47 for (auto& room : rooms)
48 {
49 if (room.getId() == m_room_id)
50 {
51 for (auto& client : room.getClients())
52 {
53 if (client.getId() == m_client_id)
54 {
55 if (!m_username.empty())
56 {
57 client.setUsername(m_username);
58 }
59 return ReadResult::success;
60 }
61 }
62
63 ClientSession session(m_client_id, &api);
64 session.setUsername(m_username);
65 room.addClient(std::move(session));
66 return ReadResult::success;
67 }
68 }
69 return ReadResult::command_execution;
70 }
71 else if (m_action == "leave")
72 {
73 for (auto& room : rooms)
74 {
75 if (room.getId() == m_room_id)
76 {
77 room.removeClient(m_client_id);
78 return ReadResult::success;
79 }
80 }
81 return ReadResult::command_execution;
82 }
83 else if (m_action == "create")
84 {
85 auto roomData = RoomData(m_client_id, m_room_name, m_room_id,
86 static_cast<RoomData::Context>(m_room_context));
87 rooms.emplace_back(Room(roomData, std::vector<ClientSession>()));
88 return ReadResult::success;
89 }
90 else if (m_action == "set_overlap")
91 {
92 api.setOverLapStatus2D(m_overlap_allowed);
93 return ReadResult::success;
94 }
95 return ReadResult::not_found;
96 }
97
98private:
99 std::string m_action;
100 uint64_t m_room_id;
101 uint64_t m_client_id;
102 std::string m_room_name;
103 uint64_t m_room_context;
104 std::string m_username;
105 bool m_overlap_allowed;
106};

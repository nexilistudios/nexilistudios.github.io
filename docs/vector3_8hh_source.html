123 static Vector3 deserialize(const nx_data& data)
124 {
125 if (data.size() != sizeof(uint32_t) * 3)
126 {
127 // Handle invalid data size.
128 throw std::runtime_error("Invalid data size for deserialization");
129 }
130
131 uint32_t xBytes, yBytes, zBytes;
132 std::memcpy(&xBytes, data.data() + 0 * sizeof(uint32_t), sizeof(uint32_t));
133 std::memcpy(&yBytes, data.data() + 1 * sizeof(uint32_t), sizeof(uint32_t));
134 std::memcpy(&zBytes, data.data() + 2 * sizeof(uint32_t), sizeof(uint32_t));
135
136 if constexpr (std::is_floating_point_v<T>)
137 {
138 float _x, _y, _z;
139 std::memcpy(&_x, &xBytes, sizeof(float));
140 std::memcpy(&_y, &yBytes, sizeof(float));
141 std::memcpy(&_z, &zBytes, sizeof(float));
142
143 return Vector3(_x, _y, _z);
144 }
145 else
146 {
147 // Integer components are stored as their 32-bit representation.
148 return Vector3(static_cast<T>(xBytes), static_cast<T>(yBytes), static_cast<T>(zBytes));
149 }
150 }

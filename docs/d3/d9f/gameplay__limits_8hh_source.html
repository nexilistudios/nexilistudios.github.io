147 static float normalizeYaw(float yaw)
148 {
149 if (!std::isfinite(yaw))
150 {
151 return 0.0f;
152 }
153
154 constexpr float kPi = 3.14159265358979323846f;
155 constexpr float kTwoPi = 2.0f * kPi;
156
157 // Values already inside the range pass through unchanged so their
158 // exact bit pattern is preserved instead of picking up rounding from
159 // the wrap arithmetic below.
160 if (yaw >= -kPi && yaw <= kPi)
161 {
162 return yaw;
163 }
164
165 float wrapped = std::fmod(yaw + kPi, kTwoPi);
166 if (wrapped < 0.0f)
167 {
168 wrapped += kTwoPi;
169 }
170 return wrapped - kPi;
171 }

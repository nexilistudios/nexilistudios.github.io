33std::function<void()> create_wait_callback(std::shared_ptr<nexilis::client::ClientAPI> api,
34 std::shared_ptr<std::promise<void>> promise,
35 nexilis_WaitType wait_type)
36{
37 return [api = std::move(api), promise = std::move(promise), wait_type]
38 {
39 if (wait_type == NEXILIS_WAIT_UNTIL_ROOMS_CREATED)
40 api->waitUntilRoomsCreated (*promise)();
41 else
42 promise->set_exception(std::make_exception_ptr(std::runtime_error("Unknown wait type")));
43 };
44}

47nexilis_WaiterHandle* nexilis_waiter_create(nexilis_ClientAPI* client_api, nexilis_PromiseHandle* promise, nexilis_WaitType wait_type)
48{
49 if (!client_api || !client_api->owner || !promise || !promise->promise)
50 return nullptr;
51
52 try
53 {
54 auto waiter = std::make_unique<nexilis_WaiterHandle>();
55 waiter->callback = create_wait_callback(client_api->owner, promise->promise, wait_type);
56 return waiter.release();
57 }
58 catch (...)
59 {
60 return nullptr;
61 }
62}

76void nexilis_waiter_wait(nexilis_WaiterHandle* waiter)
77{
78 if (!waiter)
79 return;
80
81 std::lock_guard<std::mutex> lock(waiter->mutex);
82 if (waiter->waited)
83 return;
84
85 if (waiter->callback)
86 {
87 try
88 {
89 waiter->callback();
90 waiter->waited = true;
91 }
92 catch (...)
93 {
94 // Handle exceptions
95 }
96 }
97}

104 std::lock_guard<std::mutex> lock(waiter->mutex);

27std::thread startClient(client::ClientAPI& client_api, TCPClient& tcp_client, std::vector<RoomInfo>& rooms, std::atomic<bool>& ready, std::mutex& mtx)
28{
29 // clang-format off
30 return std::thread ([&client_api, &tcp_client, &rooms, &ready, &mtx]()
31 {
32 tcp_client.start();
33 if (tcp_client.getProtocolStatus() == client::ProtocolStatus::error)
34 return;
35
36 tcp_client.sendMessage(client::Packet::Get::General::clientId(client_api));
37 while (!client_api.isInitialized())
38 {
39 if (tcp_client.getProtocolStatus() == client::ProtocolStatus::error)
40 return;
41 std::this_thread::sleep_for(std::chrono::milliseconds(10));
42 }
43
44 tcp_client.sendMessage(client::Packet::Get::Info::rooms(client_api));
45
46 std::promise<void> roomsPromise;
47 auto roomsFuture = roomsPromise.get_future();
48 auto waitFn = client_api.waitUntilRoomsCreated(roomsPromise);
49 waitFn();
50 while (roomsFuture.wait_for(std::chrono::milliseconds(100)) != std::future_status::ready)
51 if (tcp_client.getProtocolStatus() == client::ProtocolStatus::error)
52 return;
53
54 {
55 std::lock_guard<std::mutex> lock(mtx);
56 const auto& active_rooms = client_api.getActiveRooms();
57 rooms.reserve(rooms.size() + active_rooms.size());
58 std::transform(active_rooms.begin(), active_rooms.end(), std::back_inserter(rooms),
59 [](const auto& room)
60 { return RoomInfo{room.getId(), room.getName()}; });
61 }
62 ready = true;
63 });
64 // clang-format on
65}

33inline bool isValidLabel(std::string_view value, size_t maxLength)
34{
35 return !value.empty() && value.size() <= maxLength &&

42inline bool isValidAssetPath(std::string_view path)
43{
44 if (path.empty() || path.size() > 256 || path.front() == '/')
45 return false;
46 size_t start = 0;
47 while (start < path.size())
48 {
49 const size_t end = path.find('/', start);
50 const auto segment = path.substr(start, end == std::string_view::npos ? end : end - start);
51 if (segment.empty() || segment == "." || segment == ".." ||
52 !std::all_of(segment.begin(), segment.end(), [](unsigned char ch)
53 { return (ch >= 'a' && ch <= 'z') || (ch >= 'A' && ch <= 'Z') ||
54 (ch >= '0' && ch <= '9') || ch == '_' || ch == '-' || ch == '.'; }))
55 return false;
56 if (end == std::string_view::npos)
57 return true;
58 start = end + 1;
59 }
60 return false;
61}

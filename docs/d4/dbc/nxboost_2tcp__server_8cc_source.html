107void TCPServer::start()
108{
109 // Build the shared TLS context once. When TLS is enabled but the context
110 // cannot be created (e.g. no passphrase configured), connections are
111 // refused rather than silently served in plaintext.
112 if (getSettings().isTlsEnabled())
113 {
114 m_tlsContext = tls::createPskContext(getSettings().getPassword(), true);
115 if (!m_tlsContext)
116 {
117 Log::critical(header(), "TLS-PSK context creation failed, TLS connections will be refused "
118 "(is the passphrase set?)");
119 }
120 }
121
122 // clang-format off
123 m_ioContextThread = std::thread([this]()
124 {
125 m_ioContext->run();
126 });
127
128 m_listenThread = std::thread([this]()
129 {
130 if (startListening())
131 {
132 acceptClients();
133 }
134 });
135 // clang-format on
136}

138void TCPServer::stop()
139{
140 if (!m_stopped || m_stopped->exchange(true))
141 {
142 Log::debug(header(), "stop already in progress or completed.");
143 return;
144 }
145
146 // Stop io context.
147 if (m_ioContext)
148 {
149 Log::debug(header(), "TCPServer stopping io_context");
150 m_ioContext->stop();
151 }
152
153 {
154 std::lock_guard<std::mutex> lock(*m_mutex);
155 for (auto& thread : m_clientThreads)
156 {
157 if (thread.joinable())
158 {
159 Log::debug(header(), "force closing client thread");
160 thread.detach();
161 }
162 }
163 m_clientThreads.clear();
164 }
165
166 auto tryJoinThread = [this](std::thread& thread, const std::string& name)
167 {
168 if (thread.joinable())
169 {
170 Log::debug(header(), "joining ", name);
171 thread.join();
172 }
173 };
174
175 // The accept loops poll m_stopped with a short select() timeout and exit
176 // on their own. Join them before touching the acceptors: closing an
177 // acceptor from this thread while the accept thread is blocked on it is a
178 // data race that can crash on shutdown.
179 tryJoinThread(m_switchedAcceptThread, "switchAcceptThread");
180 tryJoinThread(m_listenThread, "listenThread");
181 tryJoinThread(m_ioContextThread, "ioContextThread");
182
183 // Close sockets. Only the accept thread was using the acceptors, and it
184 // has been joined by now.
185 boost::system::error_code ec;
186 ec = m_acceptor.close(ec);
187 if (ec)
188 {
189 Log::error(header(), "Error closing m_acceptor: ", ec.message());
190 }
191 ec = m_switchedAcceptor.close(ec);
192 if (ec)
193 {
194 Log::error(header(), "Error closing m_switchedAcceptor: ", ec.message());
195 }
196
197 Log::debug(header(), "stop complete.");
198}

200void TCPServer::startSwitchedAccepting()
201{
202 // clang-format off
203 m_switchedAcceptThread = std::thread([this]()
204 {
205 while (!m_stopped->load())
206 {
207 try
208 {
209 boost::asio::ip::tcp::socket socket(*m_ioContext);
210 boost::system::error_code ec;
211
212 if (!m_switchedAcceptor.is_open())
213 {
214 if (!m_stopped->load())
215 {
216 Log::warning(header(), "Acceptor closed unexpectedly");
217 }
218 break;
219 }
220
221 // Same select-based approach for switched acceptor
222 fd_set read_fds;
223 FD_ZERO(&read_fds);
224 FD_SET(m_switchedAcceptor.native_handle(), &read_fds);
225
226 timeval timeout;
227 timeout.tv_sec = 0;
228 timeout.tv_usec = 100000; // 100ms timeout
229
230 int result = select(m_switchedAcceptor.native_handle() + 1, &read_fds, nullptr, nullptr, &timeout);
231
232 if (result > 0)
233 {
234 ec = m_switchedAcceptor.accept(socket, ec);
235
236 if (ec)
237 {
238 if (ec == boost::asio::error::bad_descriptor)
239 {
240 if (!m_stopped->load())
241 {
242 Log::warning(header(), "Accept on closed acceptor");
243 }
244 break;
245 }
246 else if (ec != boost::asio::error::operation_aborted)
247 {
248 Log::error(header(), "Accept error: ", ec.message());
249 }
250 continue;
251 }
252
253 if (socket.is_open())
254 {
255 Log::debug(header(), "Accepted switched connection");
256 handleClient(std::move(socket));
257 }
258 }
259 else if (result < 0 && errno != EINTR)
260 {
261 Log::error(header(), "Switched select error: ", strerror(errno));
262 }
263 }
264 catch (...)
265 {
266 if (!m_stopped->load())
267 {
268 Log::error(header(), "Error in switched accept");
269 }
270 }
271 }
272 });
273 // clang-format on
274}

283nx_data TCPServer::receiveMessage(boost::asio::ip::tcp::socket& socket)
284{
285 nx_data data;
286
287 boost::asio::streambuf receiveBuffer;
288 boost::system::error_code error_code;
289
290 boost::asio::read(socket, receiveBuffer, boost::asio::transfer_at_least(1), error_code);
291
292 if (error_code == boost::asio::error::eof)
293 {
294 return data;
295 }
296 else if (error_code == boost::asio::error::operation_aborted)
297 {
298 Log::debug(header(), "receiveMessage aborted due to shutdown");
299 return data;
300 }
301 else if (error_code)
302 {
303 // Handle other errors
304 Log::error(header(), "Error reading from client: ", error_code.message());
305 return data;
306 }
307
308 // Get the sequence of const buffers from the streambuf.
309 const boost::asio::const_buffer& receive_buffer = receiveBuffer.data();
310
311 // Check if the buffer is empty.
312 if (receive_buffer.size() == 0)
313 {
314 Log::info(header(), "Empty data");
315 return data;
316 }
317
318 // Extract the data.
319 const uint8_t* buffer_data = static_cast<const uint8_t*>(receive_buffer.data());
320 size_t buffer_size = receive_buffer.size();
321
322 // Insert the data from the buffer into the vector.
323 data.insert(data.end(), buffer_data, buffer_data + buffer_size);
324
325 return data;
326}

344void TCPServer::handleHandshake(boost::asio::ip::tcp::socket socket, std::function<void()> onCompleted)
345{
346 ServerProtocol::connectionEstablished();
347 try
348 {
349 auto socket_wrapper = std::make_shared<nexilis::boost_tcp::Socket>(*m_ioContext);
350 if (getSettings().isTlsEnabled())
351 {
352 if (!m_tlsContext)
353 {
354 Log::error(header(), "TLS requested but no TLS context is available, refusing connection");
355 return;
356 }
357 socket_wrapper->enableTls(m_tlsContext);
358 }
359 socket_wrapper->assign(std::move(socket));
360
361 // clang-format off
362 auto thread = std::thread([this, socket_wrapper, cb = std::move(onCompleted)]() mutable
363 {
364 try
365 {
366 while (!m_stopped->load() && socket_wrapper->isOpen())
367 {
368 nx_data data;
369 bool got_data = socket_wrapper->receive(data);
370 if (m_stopped->load())
371 {
372 socket_wrapper->forceClose();
373 break;
374 }
375
376 if (!got_data)
377 {
378 if(!socket_wrapper->isOpen())
379 {
380 break;
381 }
382 continue;
383 }
384
385 auto clientAddress = socket_wrapper->getRemoteAddress();
386
387 // The first message from an unidentified client is the
388 // password (or a client_id handshake in skip mode). The
389 // MessageHandler parses it and returns an AuthMessage
390 // whose payload carries the first command byte.
391 auto handled_message = getMessageHandler().readMessage(clientAddress, data, &getSettings());
392
393 if (!handled_message)
394 {
395 continue;
396 }
397
398 // Bind the socket's send function to the user so that
399 // subsequent room commands can reach the client over the
400 // same Boost TCP socket used for the handshake.
401 handled_message->getUser()->setBoostTCPSend([socket_wrapper](const nx_data& bytes)
402 {
403 socket_wrapper->send(bytes);
404 });
405
406 auto type = handled_message->getType();
407
408 if (type == BaseMessage::Type::auth_message)
409 {
410 auto msgPtr = static_cast<AuthMessage*>(handled_message.get());
411
412 // The first byte of the auth payload is the command
413 // type; the rest is the password or client_id data.
414 CommandResult passCommand = getCommand().read(msgPtr->getData()[0], *msgPtr->getUser(), *this, msgPtr->getMessageId());
415
416 if (passCommand == CommandResult::success)
417 {
418 Log::info(header(), "Auth part 1 success");
419
420 if (!m_firstClientConnected->load())
421 {
422 Log::info(header(), "First client! Opening second port");
423 if (switchToSessionPort() == 0)
424 {
425 Log::error(header(), "Failed to open the second port");
426 break;
427 }
428 m_firstClientConnected->store(true);
429 }
430
431 uint16_t port = getPort();
432 if (port == 0)
433 {
434 Log::error(header(), "No port avainable for the client");
435 break;
436 }
437
438 nx_data command_data = { 0, 1, 0, 1 };
439 auto port_data = Util::convertToByteVector(getPort());
440 command_data.insert(command_data.end(), port_data.begin(), port_data.end());
441
442 CommandResult portCommand = getCommand().read(command_data, *msgPtr->getUser(), *this, msgPtr->getMessageId());
443
444 if (portCommand == CommandResult::success)
445 {
446 Log::info(header(), "Auth fully complete");
447 socket_wrapper->close();
448 cb();
449 break;
450 }
451 else
452 {
453 Log::error("Auth failed to port data");
454 }
455 }
456 else
457 {
458 Log::info(header(), "Result: ", asString(passCommand));
459 }
460 }
461 else
462 {
463 Log::critical(header(), "Wrong message type");
464 }
465 }
466 }
467 catch (...)
468 {
469 Log::debug(header(), "Error with handshake");
470 }
471
472 socket_wrapper->close();
473 });
474
475 {
476 std::lock_guard<std::mutex> lock(*m_mutex);
477 m_clientThreads.emplace_back(std::move(thread));
478 }
479 }
480 catch (const std::exception& e)
481 {
482 ServerProtocol::connectionClosed();
483 Log::error(header(), "Failed to create handshake thread: ", e.what());
484 }
485 catch (...)
486 {
487 ServerProtocol::connectionClosed();
488 Log::error(header(), "Other failure in handshake thread");
489 }
490 // clang-format on
491}

493void TCPServer::handleClient(boost::asio::ip::tcp::socket socket)
494{
495 auto client_address = getClientAddress(socket);
496 auto socket_wrapper = std::make_shared<nexilis::boost_tcp::Socket>(*m_ioContext);
497 if (getSettings().isTlsEnabled())
498 {
499 if (!m_tlsContext)
500 {
501 Log::error(header(), "TLS requested but no TLS context is available, refusing connection");
502 return;
503 }
504 socket_wrapper->enableTls(m_tlsContext);
505 }
506 socket_wrapper->assign(std::move(socket));
507
508 // clang-format off
509 auto client_thread = std::thread([this, socket_wrapper, client_address]() mutable
510 {
511 if (!socket_wrapper->isOpen())
512 {
513 return;
514 }
515
516 uint64_t connectedUserId = 0;
517
518 try
519 {
520 while (!m_stopped->load() && socket_wrapper->isOpen())
521 {
522 nx_data data;
523 bool got_data = socket_wrapper->receive(data);
524
525 if (m_stopped->load())
526 {
527 socket_wrapper->forceClose();
528 break;
529 }
530
531 if (!got_data)
532 {
533 if (!socket_wrapper->isOpen())
534 {
535 break;
536 }
537 continue;
538 }
539
540 auto handledMessage = getMessageHandler().readMessage(client_address, data, &getSettings());
541
542 if (!handledMessage)
543 {
544 Log::error(header(), "Received invalid message");
545 continue;
546 }
547
548 // Track the user ID for cleanup on disconnect.
549 if (handledMessage->getUser() && connectedUserId == 0)
550 {
551 connectedUserId = handledMessage->getUser()->getId();
552 }
553
554 // Always update the sending function, implement some caching later.
555 handledMessage->getUser()->setBoostTCPSend([socket_wrapper](const nx_data& bytes)
556 {
557 socket_wrapper->send(bytes);
558 });
559
561 auto type = handledMessage->getType();
562
563 if (type == BaseMessage::Type::message)
564 {
565 auto msgPtr = static_cast<Message*>(handledMessage.get());
566
567 // Everything ok with initialization.
568 CommandResult passCommand = getCommand().read(msgPtr->getData(), *msgPtr->getUser(), *this, msgPtr->getMessageId());
569
570 if (passCommand == CommandResult::success)
571 {
572 Log::debug(header(), "command success");
573 }
574 else
575 {
576 Log::warning(header(), "Result: ", asString(passCommand));
577 }
578 }
579 else
580 {
581 Log::critical(header(), "Wrong message type");
582 }
583 }
584 }
585 catch (const boost::system::system_error& e)
586 {
587 if (e.code() != boost::asio::error::operation_aborted)
588 {
589 Log::error(header(), "Error in client thread: ", e.what());
590 }
591 }
592 catch (const std::exception& e)
593 {
594 Log::error(header(), "Exception in client thread: ", e.what());
595 }
596
597 // --- Cleanup: remove disconnected user from their room and client storage ---
598 if (connectedUserId != 0)
599 {
600 auto* user = ClientStorage::getClientById(connectedUserId);
601 if (user)
602 {
603 uint64_t roomId = user->getRoomId();
604 if (roomId != 0)
605 {
606 auto* room = RoomStorage::getRoomById(roomId);
607 if (room)
608 {
609 Log::info(header(), "Cleaning up disconnected user ", connectedUserId, " from room ", roomId);
610
611 // Broadcast the standard leave command to the remaining clients
612 // so they remove the disconnected user's object from their scene.
613 nx_data leaveData = {
614 static_cast<uint8_t>(CommandType::room),
615 static_cast<uint8_t>(RoomCommandType::Root::management),
616 static_cast<uint8_t>(RoomCommandType::Management::leave)};
617 DefaultArgs args(*user, *this, leaveData, 0);
618 ServerImpl::room_management_leave(args);
619 }
620 }
621 ClientStorage::remove(connectedUserId);
622 }
623 }
624
625 if (socket_wrapper->isOpen())
626 {
627 socket_wrapper->close();
628 }
629 ServerProtocol::connectionClosed();
630 });
631
632 try
633 {
634 std::lock_guard<std::mutex> lock(*m_mutex);
635 m_clientThreads.emplace_back(std::move(client_thread));
636 }
637 catch (...)
638 {
639 Log::critical(header(), "Failed to launch client thread");
640 if (socket_wrapper->isOpen())
641 {
642 socket_wrapper->close();
643 }
644 }
645 // clang-format on
646}

648bool TCPServer::acceptClients()
649{
650 try
651 {
652 while (!m_stopped->load())
653 {
654 boost::asio::ip::tcp::socket socket(*m_ioContext);
655 boost::system::error_code ec;
656
657 if (!m_acceptor.is_open())
658 {
659 if (!m_stopped->load())
660 {
661 Log::warning(header(), "Acceptor closed suddenly");
662 }
663 break;
664 }
665
666 // Use select() for interruptible accept
667 fd_set read_fds;
668 FD_ZERO(&read_fds);
669 FD_SET(m_acceptor.native_handle(), &read_fds);
670
671 timeval timeout;
672 timeout.tv_sec = 0;
673 timeout.tv_usec = 100000; // 100ms timeout
674
675 int result = select(m_acceptor.native_handle() + 1, &read_fds, nullptr, nullptr, &timeout);
676
677 if (result > 0)
678 {
679 ec = m_acceptor.accept(socket, ec);
680 if (ec)
681 {
682 if (ec == boost::asio::error::bad_descriptor)
683 {
684 if (!m_stopped->load())
685 {
686 Log::warning(header(), "Accept on closed acceptor");
687 }
688 break;
689 }
690 else if (ec != boost::asio::error::operation_aborted)
691 {
692 Log::error(header(), "Accept error: ", ec.message());
693 }
694 continue;
695 }
696
697 if (socket.is_open())
698 {
699 Log::debug(header(), "New handshake connection");
700 handleHandshake(std::move(socket), []()
701 { Log::info("server::nxboost::TCPServer: Handshake completed!"); });
702 }
703 }
704 else if (result < 0 && errno != EINTR)
705 {
706 Log::error(header(), "Select error: ", strerror(errno));
707 }
708
709 if (m_stopped->load())
710 {
711 break;
712 }
713 }
714 }
715 catch (...)
716 {
717 if (!m_stopped->load())
718 {
719 Log::error(header(), "Exception in accept loop");
720 }
721 }
722 return true;
723}

741uint16_t TCPServer::switchToSessionPort()
742{
743 boost::system::error_code ec;
744
745 // Close the previous switched acceptor if any.
746 ec = m_switchedAcceptor.close(ec);
747 if (ec)
748 {
749 Log::error(header(), "Failed to close previous m_switchedAcceptor");
750 }
751
752 // Configure new switched port.
753 ec = m_switchedAcceptor.open(boost::asio::ip::tcp::v4(), ec);
754 if (ec)
755 {
756 Log::error(header(), "Failed to open switched acceptor: ", ec.message());
757 return 0;
758 }
759
760 m_switchedAcceptor.set_option(boost::asio::ip::tcp::acceptor::reuse_address(true), ec);
761 if (ec)
762 {
763 Log::error(header(), "Failed to configure switched acceptor: ", ec.message());
764 return 0;
765 }
766
767 ec = m_switchedAcceptor.bind(boost::asio::ip::tcp::endpoint(boost::asio::ip::tcp::v4(), Ports::getBoostTCPSessionPort()), ec);
768 if (ec)
769 {
770 Log::error(header(), "Failed to bind switched acceptor: ", ec.message());
771 return 0;
772 }
773
774 ec = m_switchedAcceptor.listen(boost::asio::socket_base::max_listen_connections, ec);
775 if (ec)
776 {
777 Log::error(header(), "Failed to listen on switched acceptor: ", ec.message());
778 return 0;
779 }
780
781 m_switchedPort->store(m_switchedAcceptor.local_endpoint().port());
782
783 startSwitchedAccepting();
784
785 Log::debug(header(), "Now listening on default port: ", m_defaultServerPort, " and switched port: ", m_switchedPort->load());
786 return m_switchedPort->load();
787}

41 template <typename T>
42 static T convertToType(nx_data bytes)
43 {
44 if (bytes.size() < sizeof(T))
45 {
46 // Handle error: insufficient data
47 throw std::runtime_error("Conversion failed: Insufficient data");
48 }
49
50 T result = 0;
51
52 if (server::Config::getBigEndian())
53 {
54 for (uint64_t i = 0; i < sizeof(T); ++i)
55 {
56 result |= static_cast<T>(bytes[i]) << (8 * (sizeof(T) - 1 - i));
57 }
58 }
59 else
60 {
61 for (uint64_t i = 0; i < sizeof(T); ++i)
62 {
63 result |= static_cast<T>(bytes[i]) << (8 * i);
64 }
65 }
66
67 return result;
68 }

123void ServerProtocol::expirePendingHandshakes(NXProtocol& protocol)
124{
125 const auto now = std::chrono::steady_clock::now();
126 const auto users = ClientStorage::getAllClients();

162void ServerProtocol::expireIdleDatagramTransports(User::Transport transport,
163 NXProtocol& protocol)
164{
165 const auto timeout = getSettings().getDatagramIdleTimeout();
166 if (timeout <= std::chrono::milliseconds::zero())
167 return;
168
169 const auto now = std::chrono::steady_clock::now();
170 const auto interval = std::min(std::chrono::milliseconds(1000),
171 std::max(std::chrono::milliseconds(10), timeout / 2));
172 if (m_lastDatagramSweep != std::chrono::steady_clock::time_point{} &&
173 now - m_lastDatagramSweep < interval)
174 return;
175 m_lastDatagramSweep = now;
176
177 const auto users = ClientStorage::getAllClients();
178 for (const auto& user : users)
179 {
180 if (!user)
181 continue;
182
183 const auto status = user->expireDatagramTransport(transport, now, timeout);
184 if (!status)
185 continue;
186
187 if (!status->anyActive)
188 leaveAndRemoveUser(*user, protocol);
189 else
190 user->clearSendCallback(transport);
191 }
192}

206void ServerProtocol::leaveAndRemoveUser(User& user, NXProtocol& protocol)
207{
208 try
209 {
210 if (user.getRoomId() != 0 && RoomStorage::getRoomById(user.getRoomId()))
211 {
212 const auto leaveData = leaveRoomData();
213 DefaultArgs args(user, protocol, leaveData, 0);
214 ServerImpl::room_management_leave(args);
215 }
216 }
217 catch (...)
218 {
219 // A closed transport must not prevent the user from being removed.
220 }
221
222 // Remove memberships left by earlier inconsistent state as well as the
223 // room recorded on the user. Never leave a dead member occupying capacity.
224 for (const auto& room : RoomStorage::getAllRooms())
225 {
226 if (room->contains(user.getId()))
227 room->leaveRoom(user.getId());
228 }
229 user.setRoomId(0);
230
231 try
232 {
233 ClientStorage::remove(user.getId());
234 }
235 catch (...)
236 {
237 }
238}

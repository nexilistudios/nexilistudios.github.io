28class RoomCommunicationCommand : public BaseAPICommand
29{
30public:
31 RoomCommunicationCommand(std::string action, uint64_t room_id, uint64_t client_id, const std::string& message)
32 : m_action(action), m_room_id(room_id), m_client_id(client_id), m_message(message)
33 {
34 }
35
36 ReadResult execute(ClientAPI& api, ClientAPI::ClientAPIData& data) override
37 {
38 auto& mtx = data.getRoomsMutex();
39 std::lock_guard<std::mutex> lock(*mtx);
40
41 auto& rooms = data.getCurrentlyActiveRooms();
42
43 if (m_action == "broadcast" || m_action == "broadcast_encrypted")
44 {
45 // End-to-end encrypted messages are relayed by the server as an
46 // opaque base64 ciphertext blob. Only the receiving client can
47 // turn that back into the original message.
48 std::string payload = m_message;
49 if (m_action == "broadcast_encrypted")
50 {
51 const auto key = crypto::deriveMessageKey(api.getClientPassword(), m_room_id);
52 const auto sealed = crypto::fromBase64(m_message);
53 if (!key.empty() && !sealed.empty() && crypto::decrypt(key, sealed, payload))
54 {
55 // Successfully decrypted into payload.
56 }
57 else
58 {
59 Log::error("broadcast_encrypted: failed to decrypt message, keeping the raw ciphertext");
60 }
61 }
62
63 for (auto&& room = rooms.begin(); room != rooms.end(); room++)
64 {
65 ClientSession* sender = nullptr;
66 for (auto& client : room->getClients())
67 {
68 if (client.getId() == m_client_id)
69 {
70 sender = &client;
71 }
72 }
73
74 if (room->getId() == m_room_id)
75 {
76 Room::Communication newMessage(payload, sender);
77 [[maybe_unused]] auto message_id = newMessage.getId();
78 room->addMessage(std::move(newMessage));
79
80 assert(room->containsCommunication(message_id));
81 return ReadResult::success;
82 }
83 }
84 return ReadResult::failure;
85 }
86 else if (m_action == "multicast")
87 {
88 // Multicast implementation would go here
89 return ReadResult::not_implemented;
90 }
91 return ReadResult::failure;
92 }
93
94private:
95 std::string m_action;
96 uint64_t m_room_id;
97 uint64_t m_client_id;
98 std::string m_message;
99};

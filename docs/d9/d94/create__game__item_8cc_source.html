30CommandResult ServerImpl::room_gameitem_create(const DefaultArgs& args)
31{
32 auto& user = args.getUser();
33 auto room = RoomStorage::getRoomById(user.getRoomId());
34 if (!room || !room->isClientGameItemCreationAllowed())
35 return CommandResult::failure;
36 if (room->getContext() != RoomData::Context::_3D)
37 return CommandResult::invalid_input;
38 auto data = args.getData();
39 auto payload = Util::removeAmountOfBytesFromVector(data, 3);
40 if (payload.size() < 29 || payload.size() > 24 + 64 + 1 + 32 + 1 + 256)
41 return CommandResult::invalid_input;
42
43 auto position = Util::vector3fFromFront(payload);
44 payload = Util::removeAmountOfBytesFromVector(payload, 12);
45 auto dimensions = Util::vector3fFromFront(payload);
46 payload = Util::removeAmountOfBytesFromVector(payload, 12);
47
48 const auto& limits = GameplayLimits::get();
49 const float maxPosition = limits.getMaxHorizontalPosition();
50 if (!std::isfinite(position.x) || !std::isfinite(position.y) || !std::isfinite(position.z) ||
51 !std::isfinite(dimensions.x) || !std::isfinite(dimensions.y) ||
52 !std::isfinite(dimensions.z) || std::abs(position.x) > maxPosition ||
53 std::abs(position.z) > maxPosition || position.y < limits.getMinHeight() ||
54 position.y > limits.getMaxHeight() || dimensions.x <= 0.0f ||
55 dimensions.y <= 0.0f || dimensions.z <= 0.0f ||
56 dimensions.x > maxPosition || dimensions.y > maxPosition ||
57 dimensions.z > maxPosition)
58 return CommandResult::invalid_input;
59
60 const std::string combined = Util::convertToString(payload);
61 const size_t firstSeparator = combined.find('\0');
62 if (firstSeparator == std::string::npos)
63 return CommandResult::invalid_input;
64 const size_t secondSeparator = combined.find('\0', firstSeparator + 1);
65 if (secondSeparator == std::string::npos ||
66 combined.find('\0', secondSeparator + 1) != std::string::npos)
67 return CommandResult::invalid_input;
68 const std::string item_type = combined.substr(0, firstSeparator);
69 const std::string status = combined.substr(firstSeparator + 1, secondSeparator - firstSeparator - 1);
70 const std::string filepath = combined.substr(secondSeparator + 1);
71 if (!validation::isValidLabel(item_type, 64) ||
72 !validation::isValidLabel(status, 32) ||
73 !validation::isValidAssetPath(filepath))
74 return CommandResult::invalid_input;
75
76 if (!room->canCreateGameItem(user.getId(), position, dimensions, item_type, status,
77 filepath))
78 return CommandResult::unauthorized;
79
80 auto item = GameItem(Util::getRandomUint64(), item_type, position, dimensions, status, filepath);
81 uint64_t itemId = item.getId();
82 Log::info("Created game item with id: ", itemId, " type: ", item_type);
83
84 if (!room->tryAddGameItem(std::move(item), RoomLimits::get().getMaxGameItemsPerRoom()))
85 return CommandResult::failure;
86
87 std::map<std::string, boost::json::value> params{
88 {"id", boost::json::value(itemId)},
89 {"x", boost::json::value(position.x)},
90 {"y", boost::json::value(position.y)},
91 {"z", boost::json::value(position.z)},
92 {"w", boost::json::value(dimensions.x)},
93 {"h", boost::json::value(dimensions.y)},
94 {"d", boost::json::value(dimensions.z)},
95 {"item_type", boost::json::value(item_type)},
96 {"status", boost::json::value(status)},
97 {"filepath", boost::json::value(filepath)}};
98
99 auto roomCommand = Command::createRoomCommand(user.getRoomId(), user, data, params, args.getMessageId());
100 if (Command::sendRoomCommand(roomCommand, user, args.getProtocol()))
101 return CommandResult::success;
102 else
103 return CommandResult::failed_room_send;
104}

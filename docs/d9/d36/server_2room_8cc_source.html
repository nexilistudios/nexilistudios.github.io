136bool Room::tryJoinRoom(uint64_t userId)
137{
138 {
139 std::lock_guard lock(m_membershipMutex);
140 if (std::find(m_clientIds.begin(), m_clientIds.end(), userId) != m_clientIds.end() ||
141 m_clientIds.size() >= getMaxSize())
142 return false;
143 m_clientIds.emplace_back(userId);
144 }
145 Log::info("New user in room: ", getId(), " user: ", userId);
146 return true;
147}

155bool Room::allowsUser(const User& user) const
156{
157 AdmissionHandler handler;
158 {
159 std::lock_guard lock(m_admissionMutex);
160 handler = m_admissionHandler;
161 }
162 try
163 {
164 // Without an application policy the room stays private. The creator and
165 // users who already joined can still see and rejoin their own room;
166 // every other client must be admitted by an explicit handler.
167 if (!handler)
168 return user.getId() == getCreatorId() || contains(user.getId());
169 return handler(user);
170 }
171 catch (...)
172 {
173 return false;
174 }
175}

229bool Room::damagePlayer(uint64_t clientId, float damage)
230{
231 std::lock_guard<std::mutex> lock(m_stateMutex);
232
233 auto it = m_playerHealth.find(clientId);
234 float health;
235 if (it != m_playerHealth.end())
236 {
237 // Ignore damage to players who are already dead. This is what prevents
238 // a single kill from being recorded twice: near-simultaneous shots from
239 // multiple clients all see health <= 0 and only the first one reports
240 // a death.
241 if (it->second <= 0.0f)
242 return false;
243 health = std::max(0.0f, it->second - damage);
244 it->second = health;
245 }
246 else
247 {
248 health = std::max(0.0f, m_defaultHealth - damage);
249 m_playerHealth[clientId] = health;
250 }
251
252 return health <= 0.0f;
253}

588bool Room::broadcastToAll(const nx_data& data)
589{
590 bool all_success = true;
591 for (auto clientId : getClients())
592 {
593 auto client = ClientStorage::getClientById(clientId);
594 if (!client)
595 continue;
596
597 if (client->isBoostTCPSet())
598 {
599 if (!client->boostTCPSend(data))
600 all_success = false;
601 }
602 else if (client->isBoostUDPSet())
603 {
604 if (!client->boostUDPSend(data))
605 all_success = false;
606 }
607 else if (client->isUnixStreamSet())
608 {
609 if (!client->unixStreamSend(data))
610 all_success = false;
611 }
612 else if (client->isInetTCPSet())
613 {
614 if (!client->inetTCPSend(data))
615 all_success = false;
616 }
617 else if (client->isInetUDPSet())
618 {
619 if (!client->inetUDPSend(data))
620 all_success = false;
621 }
622 else
623 {
624 all_success = false;
625 }
626 }
627 return all_success;
628}

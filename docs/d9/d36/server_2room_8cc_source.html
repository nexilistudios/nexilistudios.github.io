87bool Room::contains(uint64_t userId)
88{
89 // Use std::any_of algorithm to check if the user is in the room.
90 return std::any_of(m_clientIds.begin(), m_clientIds.end(),
91 [&userId](uint64_t id)
92 { return userId == id; });

95void Room::joinRoom(uint64_t userId)
96{
97 if (contains(userId))

99 Log::warning("User already in this room!");

108void Room::leaveRoom(uint64_t userId)
109{
110 m_clientIds.erase(std::remove_if(m_clientIds.begin(), m_clientIds.end(),
111 [&userId](uint64_t id)
112 { return userId == id; }),
113 m_clientIds.end());
114 std::lock_guard<std::mutex> lock(m_stateMutex);
115 m_playerHealth.erase(userId);
116 m_playerKills.erase(userId);
117 m_playerDeaths.erase(userId);
118 m_playerTeams.erase(userId);
119}

144bool Room::damagePlayer(uint64_t clientId, float damage)
145{
146 std::lock_guard<std::mutex> lock(m_stateMutex);
147
148 auto it = m_playerHealth.find(clientId);
149 float health;
150 if (it != m_playerHealth.end())
151 {
152 // Ignore damage to players who are already dead. This is what prevents
153 // a single kill from being recorded twice: near-simultaneous shots from
154 // multiple clients all see health <= 0 and only the first one reports
155 // a death.
156 if (it->second <= 0.0f)
157 return false;
158 health = std::max(0.0f, it->second - damage);
159 it->second = health;
160 }
161 else
162 {
163 health = std::max(0.0f, m_defaultHealth - damage);
164 m_playerHealth[clientId] = health;
165 }
166
167 return health <= 0.0f;
168}

341bool Room::broadcastToAll(const nx_data& data)
342{
343 bool all_success = true;
344 for (auto clientId : m_clientIds)
345 {
346 auto client = ClientStorage::getClientById(clientId);
347 if (!client)
348 continue;
349
350 if (client->isBoostTCPSet())
351 {
352 if (!client->boostTCPSend(data))
353 all_success = false;
354 }
355 else if (client->isBoostUDPSet())
356 {
357 if (!client->boostUDPSend(data))
358 all_success = false;
359 }
360 else if (client->isUnixStreamSet())
361 {
362 if (!client->unixStreamSend(data))
363 all_success = false;
364 }
365 else if (client->isInetTCPSet())
366 {
367 if (!client->inetTCPSend(data))
368 all_success = false;
369 }
370 else if (client->isInetUDPSet())
371 {
372 if (!client->inetUDPSend(data))
373 all_success = false;
374 }
375 else
376 {
377 all_success = false;
378 }
379 }
380 return all_success;
381}

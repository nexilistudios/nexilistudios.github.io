55 public IReadOnlyList<ClientSession> GetClients()
56 {
57 ThrowIfDisposed();
58
59 ulong numClients;
60 IntPtr clientsArrayPtr = RoomNative.nexilis_room_get_clients(_nativePtr, out numClients);
61
62 if (clientsArrayPtr == IntPtr.Zero || numClients == 0)
63 {
64 return Array.Empty<ClientSession>();
65 }
66
67 var clientSessions = new List<ClientSession>();
68 IntPtr[] clientPointers = new IntPtr[numClients];
69
70 try
71 {
72 Marshal.Copy(clientsArrayPtr, clientPointers, 0, (int)numClients);
73
74 // Create managed wrappers.
75 foreach (var ptr in clientPointers)
76 {
77 if (ptr != IntPtr.Zero)
78 {
79 clientSessions.Add(new ClientSession(ptr, ownsNativeInstance: false));
80 }
81 }
82 return clientSessions.AsReadOnly();
83 }
84 finally
85 {
86 // Pass 0 to only free the outer array, not the individual wrappers.
87 // The wrappers are owned by the returned ClientSession objects.
88 // TODO: add nexilis_client_session_free_wrapper to properly free wrappers on ClientSession disposal.
89 RoomNative.nexilis_room_free_client_array(clientsArrayPtr, 0);
90 }

120 public string GetClientUsername(ulong clientId)
121 {
122 ThrowIfDisposed();
123
124 var namePtr = ClientAPINative.nexilis_client_api_get_client_username(_clientApiPtr, clientId);
125 if (namePtr == IntPtr.Zero)
126 {
127 return string.Empty;
128 }
129
130 try
131 {
132 return Marshal.PtrToStringAnsi(namePtr) ?? string.Empty;

188 public string DrainGameEventsJson(int chatSince = 0)
189 {
190 ThrowIfDisposed();
191 var ptr = ClientAPINative.nexilis_client_api_drain_game_events(_clientApiPtr, new UIntPtr((uint)Math.Max(0, chatSince)));
192 if (ptr == IntPtr.Zero) return "{}";

25class GetInfoRoomsCommand : public BaseAPICommand
26{
27public:
28 explicit GetInfoRoomsCommand(boost::json::value rooms)
29 : m_rooms(std::move(rooms))
30 {
31 }
32
33 ReadResult execute(ClientAPI& api, ClientAPI::ClientAPIData& data) override
34 {
35 if (!m_rooms.is_array())
36 {
37 return ReadResult::invalid_input;
38 }
39
40 auto rooms = m_rooms.as_array();
41 std::vector<Room> newRooms;
42 for (const auto& room : rooms)
43 {
44 std::string name;
45 if (room.at("name").is_string())
46 name = room.at("name").as_string().c_str();
47
48 uint64_t maxSize = Util::toUint64(room.at("max_size"));
49 uint64_t creatorId = Util::toUint64(room.at("creator_id"));
50 uint64_t id = Util::toUint64(room.at("room_id"));
51 auto ctx = static_cast<RoomData::Context>(Util::toUint64(room.at("context")));
52
53 std::vector<ClientSession> roomClients;
54
55 if (room.as_object().contains("clients"))
56 {
57 auto clients = room.at("clients").as_array();
58
59 for (const auto& client : clients)
60 {
61 uint64_t client_id = Util::toUint64(client.at("id"));
62
63 std::string username;
64 if (client.at("name").is_string())
65 username = client.at("name").as_string().c_str();
66
67 float object2DX = 0.0f;
68 if (client.at("x").is_double())
69 object2DX = static_cast<float>(client.at("x").as_double());
70
71 float object2DY = 0.0f;
72 if (client.at("y").is_double())
73 object2DY = static_cast<float>(client.at("y").as_double());
74
75 float dimension2DX = 0.0f;
76 if (client.at("width").is_double())
77 dimension2DX = static_cast<float>(client.at("width").as_double());
78
79 float dimension2DY = 0.0f;
80 if (client.at("height").is_double())
81 dimension2DY = static_cast<float>(client.at("height").as_double());
82
83 ClientSession newClient(client_id, &api);
84 newClient.getObject2D().setPosition({object2DX, object2DY});
85 newClient.getObject2D().setDimensions({dimension2DX, dimension2DY});
86 newClient.setUsername(username);
87 roomClients.emplace_back(std::move(newClient));
88 }
89 }
90
91 auto roomData = RoomData(creatorId, name, id, ctx, maxSize);
92 newRooms.emplace_back(Room(roomData, std::move(roomClients)));
93
94 // 2D context room cannot have 3D objects.
95 if (ctx != RoomData::Context::_2D)
96 {
97 // Parse 3D objects from the room info.
98 if (room.as_object().contains("objects_3d"))
99 {
100 auto& roomRef = newRooms.back();
101 auto objects = room.at("objects_3d").as_array();
102 for (const auto& obj : objects)
103 {
104 uint64_t obj_id = Util::toUint64(obj.at("id"));
105 float x = static_cast<float>(obj.at("x").as_double());
106 float y = static_cast<float>(obj.at("y").as_double());
107 float z = static_cast<float>(obj.at("z").as_double());
108 float w = static_cast<float>(obj.at("w").as_double());
109 float h = static_cast<float>(obj.at("h").as_double());
110 float d = static_cast<float>(obj.at("d").as_double());
111 std::string filepath;
112 if (obj.as_object().contains("filepath"))
113 filepath = obj.at("filepath").as_string().c_str();
114
115 auto object3D = Object3D(obj_id, {x, y, z}, {w, h, d});
116 object3D.setFilepath(filepath);
117 roomRef.addObject(std::move(object3D));
118 }
119 }
120 }
121
122 // Parse 2D objects, they are allowed to exist in all contexts.
123 if (room.as_object().contains("objects_2d"))
124 {
125 auto& roomRef = newRooms.back();
126 auto objects = room.at("objects_2d").as_array();
127 for (const auto& obj : objects)
128 {
129 uint64_t obj_id = Util::toUint64(obj.at("id"));
130 float x = static_cast<float>(obj.at("x").as_double());
131 float y = static_cast<float>(obj.at("y").as_double());
132 float w = static_cast<float>(obj.at("w").as_double());
133 float h = static_cast<float>(obj.at("h").as_double());
134 std::string filepath;
135 if (obj.as_object().contains("filepath"))
136 filepath = obj.at("filepath").as_string().c_str();
137
138 auto object2D = Object2D(obj_id, {x, y}, {w, h});
139 object2D.setFilepath(filepath);
140 roomRef.addObject(std::move(object2D));
141 }
142 }
143 }
144
145 auto& mtx = data.getRoomsMutex();
146 if (mtx)
147 {
148 std::lock_guard<std::mutex> lock(*mtx);
149 data.setCurrentlyActiveRooms(std::move(newRooms));
150 }
151
152 return ReadResult::success;
153 }
154
155private:
156 boost::json::value m_rooms;
157};

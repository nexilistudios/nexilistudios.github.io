55bool MessageHandler::acceptRegistration(const std::string& address)
56{
57 const auto now = std::chrono::steady_clock::now();
58 std::lock_guard lock(m_registrationState->mutex);
59 auto& attempts = m_registrationState->byAddress;
60 auto it = attempts.find(address);
61 if (it != attempts.end() && now - it->second.windowStart >= kRegistrationWindow)
62 {
63 attempts.erase(it);
64 it = attempts.end();
65 }
66 if (it != attempts.end())
67 return ++it->second.count <= kMaxRegistrationsPerAddress;
68 if (attempts.size() >= kMaxTrackedRegistrationAddresses)
69 {
70 std::erase_if(attempts, [now](const auto& entry)
71 { return now - entry.second.windowStart >= kRegistrationWindow; });
72 if (attempts.size() >= kMaxTrackedRegistrationAddresses)
73 return false;
74 }
75 attempts.emplace(address, RegistrationState::Attempt{now, 1});
76 return true;
77}

84std::unique_ptr<BaseMessage> MessageHandler::readMessage(std::string address, const nx_data& payload, ServerConfig* authentication)
85{
86 Log::debug(header(), "Payload size: ", payload.size());
87
88 // In password_protected mode the very first message from an unidentified
89 // client is the raw password, which may be shorter than 8 bytes. Only attempt
90 // to read a client id when the payload is large enough; a short payload would
91 // otherwise parse as a zero client id and get rejected before the password
92 // comparison runs, leaking the length of the correct password.
93 uint64_t clientId = 0;
94 if (payload.size() >= sizeof(uint64_t))
95 {
96 clientId = Util::uint64FromFront(payload);
97 }
98
99 // Reject a zero client id in every mode except password_protected and skip:
100 // in password_protected the first message from an unidentified client is the
101 // password itself, and in skip mode the client has not been assigned an id
102 // yet and sends a "getting client_id" handshake to learn it.
103 if (clientId == 0 && authentication->getMode() != AuthenticationMode::password_protected && authentication->getMode() != AuthenticationMode::skip)
104 {
105 Log::error(header(), "Client id is zero");
106 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::client_id_failure);
107 }
108
109 // Authentication modes that never admit a client must not allocate a User:
110 // reject before creating registry state.
111 if (authentication->getMode() == AuthenticationMode::empty)
112 {
113 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::empty_authentication_mode);
114 }
115 if (authentication->getMode() == AuthenticationMode::admin_access ||
116 authentication->getMode() == AuthenticationMode::root_access)
117 {
118 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::not_implemented);
119 }
120
121 // Unknown identifiers are not users until their handshake succeeds.
122 auto user = ClientStorage::getClientById(clientId);
123 const bool newUser = !user;
124 auto createUser = [&]() -> bool
125 {
126 if (!acceptRegistration(address))
127 return false;
128 uint64_t newId = Util::getRandomUint64();
129 user = std::make_shared<User>(newId, address);
130 Log::info(header(), "Created new user: ", newId);
131 auto username = Util::getRandomString(10);
132 user->setUsername(username);
133 return true;
134 };
135
136 auto authenticatedMessage = [&]() -> std::unique_ptr<BaseMessage>
137 {
138 nx_data verified;
139 if (!message_auth::verify(payload, user->getUdpToken(), verified))
140 {
141 Log::warning(header(), "Rejected request with an invalid session MAC");
142 return nullptr;
143 }
144 nx_data idBytes(verified.begin() + 8, verified.begin() + 16);
145 if (!user->acceptMessageId(Util::uint64FromFront(idBytes)))
146 {
147 Log::warning(header(), "Rejected duplicate or stale message ID");
148 return nullptr;
149 }
150
151 // Drop excess authenticated traffic before parsing or dispatching it.
152 // The token bucket allows a one-second burst at the configured rate;
153 // a non-positive rate disables the limit.
154 const float maxCommandRate = authentication->getMaxCommandRate();
155 if (!user->acceptCommand(std::chrono::steady_clock::now(), maxCommandRate, maxCommandRate))
156 {
157 Log::debug(header(), "Dropped command over the per-client rate limit");
158 return nullptr;
159 }
160
161 return std::make_unique<Message>(handlePayload(verified, user, address));
162 };
163
164 switch (authentication->getMode())
165 {
166 case AuthenticationMode::empty:
167 {
168 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::empty_authentication_mode);
169 }
170 case AuthenticationMode::skip:
171 {
172 if (!newUser)
173 {
174 // Registered client: process the payload as a normal command.
175 return authenticatedMessage();
176 }
177
178 // A new skip-mode session is created only for the canonical
179 // getting/client_id handshake, not arbitrary unknown payloads.
180 if (clientId != 0 || payload.size() != 19 ||
181 payload[16] != static_cast<uint8_t>(CommandType::getting) ||
182 payload[17] != 0 || payload[18] != 0)
183 return nullptr;
184
185 // Brand new client. Register it so its client id stays stable and
186 // hand its id back through the same AuthMessage handshake used by
187 // password_protected mode (no credentials are verified here).
188 if (!createUser() || !ClientStorage::add(user))
189 return nullptr;
190
191 // This message is equal to Packet::getId (without client id):
192 // a "getting" / "client_id" command that delivers the assigned id.
193 // The session port is negotiated separately by the transport
194 // (e.g. Boost TCP opens a switched session port after auth).
195 nx_data message{1, 0, 0};
196 auto idBytes = Util::convertToByteVector(user->getId());
197 for (auto&& byte : idBytes)
198 {
199 message.emplace_back(byte);
200 }
201 std::vector<nx_data> test{message};
202
203 auto new_message_id = Util::getRandomUint64();
204 BaseMessage::Data base_data(new_message_id, address, user);
205 return std::make_unique<AuthMessage>(std::move(base_data), test);
206 }
207 case AuthenticationMode::admin_access:
208 case AuthenticationMode::root_access:
209 {
210 Log::error(header(), "Not implemented!");
211 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::not_implemented);
212 }
213 case AuthenticationMode::password_protected:
214 {
215 if (!newUser)
216 {
217 if (user->hasCommonAccess())
218 {
219 Log::debug(header(), "Message from known user");
220 return authenticatedMessage();
221 }
222 // Message from verified client that has no access.
223 else
224 {
225 Log::error(header(), "Message from verified client that has no access");
226 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::no_access);
227 }
228 }
229 else
230 {
231 // Only accept the password as a message from unidentified clients.
232 // Failed attempts are shared by source address across reconnects to
233 // this server instance so the shared passphrase cannot be
234 // brute-forced at line speed.
235 bool blocked = false;
236 {
237 const auto now = std::chrono::steady_clock::now();
238 std::lock_guard lock(m_passphraseAuthState->mutex);
239 auto& attempts = m_passphraseAuthState->byAddress;
240 auto it = attempts.find(address);
241 if (it != attempts.end() && now - it->second.firstFailure >= kPassphraseAuthWindow)
242 {
243 attempts.erase(it);
244 it = attempts.end();
245 }
246 if (it != attempts.end() && it->second.failures >= kMaxPassphraseAuthFailures)
247 {
248 blocked = true;
249 }
250 else if (it == attempts.end() &&
251 attempts.size() >= kMaxTrackedPassphraseAuthAddresses)
252 {
253 std::erase_if(attempts,
254 [now](const auto& entry)
255 {
256 return now - entry.second.firstFailure >= kPassphraseAuthWindow;
257 });
258 if (attempts.size() >= kMaxTrackedPassphraseAuthAddresses)
259 {
260 blocked = true;
261 }
262 }
263 }
264
265 if (blocked)
266 {
267 Log::warning(header(), "Too many failed passphrase attempts from ", address);
268 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::authentication_error);
269 }
270
271 // The comparison runs over the stored password length only, so the
272 // response does not reveal the length of the correct password.
273 if (Util::constantTimeEquals(authentication->getPassphrase(), payload))
274 {
275 if (!createUser())
276 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::authentication_error);
277 Log::info(header(), "Correct password by user ", user->getId());
278
279 {
280 std::lock_guard lock(m_passphraseAuthState->mutex);
281 m_passphraseAuthState->byAddress.erase(address);
282 }
283
284 user->setCommonAccess(true);
285
286 // Add to storage only if this was a new user.
287 if (!ClientStorage::add(user))
288 return nullptr;
289
290 // This message is equal to Packet::getId (without client id).
291 // The session port is negotiated separately by the
292 // transport (e.g. Boost TCP opens a switched session
293 // port after auth completes).
294 nx_data message{1, 0, 0};
295 auto idBytes = Util::convertToByteVector(user->getId());
296 for (auto&& byte : idBytes)
297 {
298 message.emplace_back(byte);
299 }
300 std::vector<nx_data> test{message};
301
302 auto new_message_id = Util::getRandomUint64();
303 BaseMessage::Data base_data(new_message_id, address, user);
304 return std::make_unique<AuthMessage>(std::move(base_data), test);
305 }
306 else
307 {
308 const auto now = std::chrono::steady_clock::now();
309 {
310 std::lock_guard lock(m_passphraseAuthState->mutex);
311 auto& attempts = m_passphraseAuthState->byAddress;
312 auto it = attempts.find(address);
313 if (it != attempts.end())
314 {
315 ++it->second.failures;
316 }
317 else
318 {
319 attempts.emplace(address, PassphraseAuthState::Attempt{now, 1});
320 }
321 }
322 Log::error(header(), "Authentication error: incorrect password");
323 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::authentication_error);
324 }
325 }
326 }
327 default:
328 {
329 Log::error(header(), "Missing authentication mode");
330 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::missing_authentication_mode);
331 }
332 }
333}

335Message MessageHandler::handlePayload(const nx_data& payload, std::shared_ptr<User> user, const std::string& address)
336{
337 // Vector without client id (8 bytes).
338 auto vectorWithoutClientId = Util::removeAmountOfBytesFromVector(payload, 8);
339
340 // Next eight bytes is the message id.
341 uint64_t messageId = Util::uint64FromFront(vectorWithoutClientId);
342
343 // Vector without message id (8 bytes).
344 auto messageVector = Util::removeAmountOfBytesFromVector(vectorWithoutClientId, 8);
345
346 BaseMessage::Data base_data(messageId, address, user);
347 return Message(std::move(base_data), messageVector);
348}

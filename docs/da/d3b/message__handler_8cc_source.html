52// Client id 8 bytes

56std::unique_ptr<BaseMessage> MessageHandler::readMessage(std::string address, const nx_data& payload, ServerConfig* authentication)
57{
58 Log::debug(header(), "Payload size: ", payload.size());
59
60 // In password_protected mode the very first message from an unidentified
61 // client is the raw password, which may be shorter than 8 bytes. Only attempt
62 // to read a client id when the payload is large enough; a short payload would
63 // otherwise parse as a zero client id and get rejected before the password
64 // comparison runs, leaking the length of the correct password.
65 uint64_t clientId = 0;
66 if (payload.size() >= sizeof(uint64_t))
67 {
68 clientId = Util::uint64FromFront(payload);
69 }
70
71 // Reject a zero client id in every mode except password_protected and skip:
72 // in password_protected the first message from an unidentified client is the
73 // password itself, and in skip mode the client has not been assigned an id
74 // yet and sends a "getting client_id" handshake to learn it.
75 if (clientId == 0 && authentication->getMode() != AuthenticationMode::password_protected && authentication->getMode() != AuthenticationMode::skip)
76 {
77 Log::error(header(), "Client id is zero");
78 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::client_id_failure);
79 }
80
81 bool newUser = false;
82
83 // Does the user actually exist?
84 auto* user = ClientStorage::getClientById(clientId);
85 std::unique_ptr<User> newUserPtr;
86 if (!user)
87 {
88 // Create a new user.
89 uint64_t newId = Util::getRandomUint64();
90 newUserPtr = std::make_unique<User>(newId, address);
91 Log::info(header(), "Created new user: ", newId);
92 auto username = Util::getRandomString(10);
93 newUserPtr->setUsername(username);
94 user = newUserPtr.get();
95 newUser = true;
96 }
97 assert(user);
98
99 auto authenticatedMessage = [&]() -> std::unique_ptr<BaseMessage>
100 {
101 nx_data verified;
102 if (!message_auth::verify(payload, user->getUdpToken(), verified))
103 {
104 Log::warning(header(), "Rejected request with an invalid session MAC");
105 return nullptr;
106 }
107 nx_data idBytes(verified.begin() + 8, verified.begin() + 16);
108 if (!user->acceptMessageId(Util::uint64FromFront(idBytes)))
109 {
110 Log::warning(header(), "Rejected duplicate or stale message ID");
111 return nullptr;
112 }
113 return std::make_unique<Message>(handlePayload(verified, user, address));
114 };
115
116 switch (authentication->getMode())
117 {
118 case AuthenticationMode::empty:
119 {
120 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::empty_authentication_mode);
121 }
122 case AuthenticationMode::skip:
123 {
124 if (!newUser)
125 {
126 // Registered client: process the payload as a normal command.
127 return authenticatedMessage();
128 }
129
130 // Brand new client. Register it so its client id stays stable and
131 // hand its id back through the same AuthMessage handshake used by
132 // password_protected mode (no credentials are verified here).
133 if (newUserPtr)
134 {
135 ClientStorage::add(std::move(newUserPtr));
136 }
137
138 uint64_t newClientId = user->getId();
139 auto realNewClient = ClientStorage::getClientById(newClientId);
140
141 // Checking successfull client creation.
142 assert(realNewClient);
143 assert(user->getId() == realNewClient->getId());
144
145 // This message is equal to Packet::getId (without client id):
146 // a "getting" / "client_id" command that delivers the assigned id.
147 // The session port is negotiated separately by the transport
148 // (e.g. Boost TCP opens a switched session port after auth).
149 nx_data message{1, 0, 0};
150 auto idBytes = Util::convertToByteVector(user->getId());
151 for (auto&& byte : idBytes)
152 {
153 message.emplace_back(byte);
154 }
155 std::vector<nx_data> test{message};
156
157 auto new_message_id = Util::getRandomUint64();
158 BaseMessage::Data base_data(new_message_id, address, realNewClient);
159 return std::make_unique<AuthMessage>(std::move(base_data), test);
160 }
161 case AuthenticationMode::admin_access:
162 case AuthenticationMode::root_access:
163 {
164 Log::error(header(), "Not implemented!");
165 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::not_implemented);
166 }
167 case AuthenticationMode::password_protected:
168 {
169 if (!newUser)
170 {
171 if (user->hasCommonAccess())
172 {
173 Log::debug(header(), "Message from known user");
174 return authenticatedMessage();
175 }
176 // Message from verified client that has no access.
177 else
178 {
179 Log::error(header(), "Message from verified client that has no access");
180 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::no_access);
181 }
182 }
183 else
184 {
185 // Only accept the password as a message from unidentified clients.
186 // Failed attempts are shared by source address across reconnects to
187 // this server instance so the shared passphrase cannot be
188 // brute-forced at line speed.
189 bool blocked = false;
190 {
191 const auto now = std::chrono::steady_clock::now();
192 std::lock_guard lock(m_passphraseAuthState->mutex);
193 auto& attempts = m_passphraseAuthState->byAddress;
194 auto it = attempts.find(address);
195 if (it != attempts.end() && now - it->second.firstFailure >= kPassphraseAuthWindow)
196 {
197 attempts.erase(it);
198 it = attempts.end();
199 }
200 if (it != attempts.end() && it->second.failures >= kMaxPassphraseAuthFailures)
201 {
202 blocked = true;
203 }
204 else if (it == attempts.end() &&
205 attempts.size() >= kMaxTrackedPassphraseAuthAddresses)
206 {
207 std::erase_if(attempts,
208 [now](const auto& entry)
209 {
210 return now - entry.second.firstFailure >= kPassphraseAuthWindow;
211 });
212 if (attempts.size() >= kMaxTrackedPassphraseAuthAddresses)
213 {
214 blocked = true;
215 }
216 }
217 }
218
219 if (blocked)
220 {
221 Log::warning(header(), "Too many failed passphrase attempts from ", address);
222 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::authentication_error);
223 }
224
225 // The comparison runs over the stored password length only, so the
226 // response does not reveal the length of the correct password.
227 if (Util::constantTimeEquals(authentication->getPassphrase(), payload))
228 {
229 Log::info(header(), "Correct password by user ", user->getId());
230
231 {
232 std::lock_guard lock(m_passphraseAuthState->mutex);
233 m_passphraseAuthState->byAddress.erase(address);
234 }
235
236 user->setCommonAccess(true);
237
238 // Add to storage only if this was a new user.
239 if (newUserPtr)
240 {
241 ClientStorage::add(std::move(newUserPtr));
242 }
243
244 uint64_t newClientId = user->getId();
245 auto realNewClient = ClientStorage::getClientById(newClientId);
246
247 // Checking successfull client creation.
248 assert(realNewClient);
249 assert(user->getId() == realNewClient->getId());
250
251 // This message is equal to Packet::getId (without client id).
252 // The session port is negotiated separately by the
253 // transport (e.g. Boost TCP opens a switched session
254 // port after auth completes).
255 nx_data message{1, 0, 0};
256 auto idBytes = Util::convertToByteVector(user->getId());
257 for (auto&& byte : idBytes)
258 {
259 message.emplace_back(byte);
260 }
261 std::vector<nx_data> test{message};
262
263 auto new_message_id = Util::getRandomUint64();
264 BaseMessage::Data base_data(new_message_id, address, realNewClient);
265 return std::make_unique<AuthMessage>(std::move(base_data), test);
266 }
267 else
268 {
269 const auto now = std::chrono::steady_clock::now();
270 {
271 std::lock_guard lock(m_passphraseAuthState->mutex);
272 auto& attempts = m_passphraseAuthState->byAddress;
273 auto it = attempts.find(address);
274 if (it != attempts.end())
275 {
276 ++it->second.failures;
277 }
278 else
279 {
280 attempts.emplace(address, PassphraseAuthState::Attempt{now, 1});
281 }
282 }
283 Log::error(header(), "Authentication error: incorrect password");
284 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::authentication_error);
285 }
286 }
287 }
288 default:
289 {
290 Log::error(header(), "Missing authentication mode");
291 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::missing_authentication_mode);
292 }
293 }

296Message MessageHandler::handlePayload(const nx_data& payload, User* user, const std::string& address)
297{
298 // Vector without client id (8 bytes).
299 auto vectorWithoutClientId = Util::removeAmountOfBytesFromVector(payload, 8);
300
301 // Next eight bytes is the message id.
302 uint64_t messageId = Util::uint64FromFront(vectorWithoutClientId);
303
304 // Vector without message id (8 bytes).
305 auto messageVector = Util::removeAmountOfBytesFromVector(vectorWithoutClientId, 8);
306
307 BaseMessage::Data base_data(messageId, address, user);
308 return Message(std::move(base_data), messageVector);
309}

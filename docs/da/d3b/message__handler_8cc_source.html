41// Payload handled by this class:
42// Client id 8 bytes
43// Message id 8 bytes
44// Command bytes (at least 2 bytes), second parameter of MessageHandler::Message.
45
46std::unique_ptr<BaseMessage> MessageHandler::readMessage(std::string address, const nx_data& payload, ServerConfig* authentication)
47{
48 Log::debug(header(), "Payload size: ", payload.size());
49
50 // In password_protected mode the very first message from an unidentified
51 // client is the raw password, which may be shorter than 8 bytes. Only attempt
52 // to read a client id when the payload is large enough; a short payload would
53 // otherwise parse as a zero client id and get rejected before the password
54 // comparison runs, leaking the length of the correct password.
55 uint64_t clientId = 0;
56 if (payload.size() >= sizeof(uint64_t))
57 {
58 clientId = Util::uint64FromFront(payload);
59 }
60
61 // Reject a zero client id in every mode except password_protected and skip:
62 // in password_protected the first message from an unidentified client is the
63 // password itself, and in skip mode the client has not been assigned an id
64 // yet and sends a "getting client_id" handshake to learn it.
65 if (clientId == 0 && authentication->getMode() != AuthenticationMode::password_protected && authentication->getMode() != AuthenticationMode::skip)
66 {
67 Log::error(header(), "Client id is zero");
68 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::client_id_failure);
69 }
70
71 bool newUser = false;
72
73 // Does the user actually exist?
74 auto* user = ClientStorage::getClientById(clientId);
75 std::unique_ptr<User> newUserPtr;
76 if (!user)
77 {
78 // Create a new user.
79 uint64_t newId = Util::getRandomUint64();
80 newUserPtr = std::make_unique<User>(newId, address);
81 Log::info(header(), "Created new user: ", newId);
82 auto username = Util::getRandomString(10);
83 newUserPtr->setUsername(username);
84 user = newUserPtr.get();
85 newUser = true;
86 }
87 assert(user);
88
89 auto authenticatedMessage = [&]() -> std::unique_ptr<BaseMessage>
90 {
91 nx_data verified;
92 if (!message_auth::verify(payload, user->getUdpToken(), verified))
93 {
94 Log::warning(header(), "Rejected request with an invalid session MAC");
95 return nullptr;
96 }
97 nx_data idBytes(verified.begin() + 8, verified.begin() + 16);
98 if (!user->acceptMessageId(Util::uint64FromFront(idBytes)))
99 {
100 Log::warning(header(), "Rejected duplicate or stale message ID");
101 return nullptr;
102 }
103 return std::make_unique<Message>(handlePayload(verified, user, address));
104 };
105
106 switch (authentication->getMode())
107 {
108 case AuthenticationMode::empty:
109 {
110 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::empty_authentication_mode);
111 }
112 case AuthenticationMode::skip:
113 {
114 if (!newUser)
115 {
116 // Registered client: process the payload as a normal command.
117 return authenticatedMessage();
118 }
119
120 // Brand new client. Register it so its client id stays stable and
121 // hand its id back through the same AuthMessage handshake used by
122 // password_protected mode (no credentials are verified here).
123 if (newUserPtr)
124 {
125 ClientStorage::add(std::move(newUserPtr));
126 }
127
128 uint64_t newClientId = user->getId();
129 auto realNewClient = ClientStorage::getClientById(newClientId);
130
131 // Checking successfull client creation.
132 assert(realNewClient);
133 assert(user->getId() == realNewClient->getId());
134
135 // This message is equal to Packet::getId (without client id):
136 // a "getting" / "client_id" command that delivers the assigned id.
137 // The session port is negotiated separately by the transport
138 // (e.g. Boost TCP opens a switched session port after auth).
139 nx_data message{1, 0, 0};
140 auto idBytes = Util::convertToByteVector(user->getId());
141 for (auto&& byte : idBytes)
142 {
143 message.emplace_back(byte);
144 }
145 std::vector<nx_data> test{message};
146
147 auto new_message_id = Util::getRandomUint64();
148 BaseMessage::Data base_data(new_message_id, address, realNewClient);
149 return std::make_unique<AuthMessage>(std::move(base_data), test);
150 }
151 case AuthenticationMode::admin_access:
152 case AuthenticationMode::root_access:
153 {
154 Log::error(header(), "Not implemented!");
155 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::not_implemented);
156 }
157 case AuthenticationMode::password_protected:
158 {
159 if (!newUser)
160 {
161 if (user->hasCommonAccess())
162 {
163 Log::debug(header(), "Message from known user");
164 return authenticatedMessage();
165 }
166 // Message from verified client that has no access.
167 else
168 {
169 Log::error(header(), "Message from verified client that has no access");
170 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::no_access);
171 }
172 }
173 else
174 {
175 // Only accept the password as a message from unidentified clients.
176 // The comparison runs over the stored password length only, so the
177 // response does not reveal the length of the correct password.
178 if (Util::constantTimeEquals(authentication->getPassphrase(), payload))
179 {
180 Log::info(header(), "Correct password by user ", user->getId());
181 user->setCommonAccess(true);
182
183 // Add to storage only if this was a new user.
184 if (newUserPtr)
185 {
186 ClientStorage::add(std::move(newUserPtr));
187 }
188
189 uint64_t newClientId = user->getId();
190 auto realNewClient = ClientStorage::getClientById(newClientId);
191
192 // Checking successfull client creation.
193 assert(realNewClient);
194 assert(user->getId() == realNewClient->getId());
195
196 // This message is equal to Packet::getId (without client id).
197 // The session port is negotiated separately by the
198 // transport (e.g. Boost TCP opens a switched session
199 // port after auth completes).
200 nx_data message{1, 0, 0};
201 auto idBytes = Util::convertToByteVector(user->getId());
202 for (auto&& byte : idBytes)
203 {
204 message.emplace_back(byte);
205 }
206 std::vector<nx_data> test{message};
207
208 auto new_message_id = Util::getRandomUint64();
209 BaseMessage::Data base_data(new_message_id, address, realNewClient);
210 return std::make_unique<AuthMessage>(std::move(base_data), test);
211 }
212 else
213 {
214 Log::error(header(), "Authentication error: incorrect password");
215 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::authentication_error);
216 }
217 }
218 }
219 default:
220 {
221 Log::error(header(), "Missing authentication mode");
222 return std::make_unique<ErrorMessage>(address, ErrorMessage::Type::missing_authentication_mode);
223 }
224 }
225}

227Message MessageHandler::handlePayload(const nx_data& payload, User* user, const std::string& address)
228{
229 // Vector without client id (8 bytes).
230 auto vectorWithoutClientId = Util::removeAmountOfBytesFromVector(payload, 8);
231
232 // Next eight bytes is the message id.
233 uint64_t messageId = Util::uint64FromFront(vectorWithoutClientId);
234
235 // Vector without message id (8 bytes).
236 auto messageVector = Util::removeAmountOfBytesFromVector(vectorWithoutClientId, 8);
237
238 BaseMessage::Data base_data(messageId, address, user);
239 return Message(std::move(base_data), messageVector);
240}

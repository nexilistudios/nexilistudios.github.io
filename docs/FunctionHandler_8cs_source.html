35 // Constructor

38 if (function == null)

41 _function = function;

48 }

53 public IntPtr HandlerPtr => _handlerPtr;

55 // Callback function for native code
56 private void OnLogMessage(LogLevel logLevel, IntPtr messagePtr)
57 {
58 string? message = Marshal.PtrToStringAnsi(messagePtr);
59 if (message == null)
60 {
61 Console.WriteLine("Warning: Received null message from native code.");
62 return;
63 }
64 _function(logLevel, message);
65 }

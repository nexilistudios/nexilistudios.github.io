25template <typename VectorType>
26bool nexilis_vector3_is_valid(void* vector_ptr)
27{
28 if (!vector_ptr)
29 {
30 return false;
31 }
32
33 try
34 {
35 VectorType* vec = reinterpret_cast<VectorType*>(vector_ptr);
36
37 if (!vec->vec)
38 {
39 return false;
40 }
41
42 using ComponentType = std::remove_reference_t<decltype(vec->vec->x)>;
43
44 volatile auto x = vec->vec->x;
45 volatile auto y = vec->vec->y;
46 volatile auto z = vec->vec->z;
47 (void)x;
48 (void)y;
49 (void)z;
50
51 if constexpr (std::is_floating_point_v<ComponentType>)
52 {
53 return std::isfinite(x) &&
54 std::isfinite(y) &&
55 std::isfinite(z);
56 }
57 else
58 {
59 return true;
60 }
61 }
62 catch (...)
63 {
64 return false;
65 }
66}

88nx_data ClientProtocol::frame(const nx_data& payload)
89{
90 nx_data framed;
91 uint32_t net_len = static_cast<uint32_t>(payload.size());
92 framed.push_back(static_cast<uint8_t>((net_len >> 24) & 0xFF));
93 framed.push_back(static_cast<uint8_t>((net_len >> 16) & 0xFF));
94 framed.push_back(static_cast<uint8_t>((net_len >> 8) & 0xFF));
95 framed.push_back(static_cast<uint8_t>(net_len & 0xFF));
96 framed.insert(framed.end(), payload.begin(), payload.end());
97 return framed;
98}

113std::pair<uint64_t, std::function<void()>> ClientProtocol::createCallback(const nx_data& message, const std::function<void()>& callback)
114{
115 // Vector without client id (8 bytes).
116 auto messageWithoutClientId = Util::removeAmountOfBytesFromVector(message, 8);
117
118 // Next eight bytes is the message id.
119 uint64_t messageId = Util::uint64FromFront(messageWithoutClientId);
120
121 return std::make_pair(messageId, callback);
122}

124void ClientProtocol::start(NXProtocol::Type type)
125{
126 if (m_api->isInitialized())
127 {
128 Log::debug(header(), "Password already sent using another protocol, waiting for ready");
129 switch (type)
130 {
131 case NXProtocol::Type::BOOST_TCP_CLIENT:
132 getClientAPI()->waitUntilBoostTCPReady();
133 break;
134 case NXProtocol::Type::BOOST_UDP_CLIENT:
135 getClientAPI()->waitUntilBoostUDPReady();
136 break;
137 case NXProtocol::Type::AF_INET_TCP_CLIENT:
138 getClientAPI()->waitUntilInetTCPReady();
139 break;
140 case NXProtocol::Type::AF_INET_UDP_CLIENT:
141 getClientAPI()->waitUntilInetUDPReady();
142 break;
143 case NXProtocol::Type::AF_UNIX_SOCK_STREAM_CLIENT:
144 getClientAPI()->waitUntilUnixStreamReady();
145 break;
146 case NXProtocol::Type::AF_UNIX_SOCK_DGRAM_CLIENT:
147 getClientAPI()->waitUntilUnixDgramReady();
148 break;
149 default:
150 break;
151 }
152 return;
153 }
154
155 switch (m_api->getMode())
156 {
157 case server::AuthenticationMode::password_protected:
158 {
159 auto password = m_api->getClientPassword();
160 Log::debug(header(), "Authenticating with server password");
161 auto message = Util::convertToByteVector(password.c_str(), password.size());
162 sendMessage(message);
163 break;
164 }
165 case server::AuthenticationMode::skip:
166 {
167 // The client does not know its id yet, so perform a "getting
168 // client_id" handshake with a zero client id prefix. The server
169 // replies with the assigned client id which initializes the client.
170 Log::debug(header(), "Authenticating by skipping authentication");
171 nx_data message;
172 _Packet::emplaceAll(message, static_cast<uint64_t>(0)); // client id
173 _Packet::emplaceAll(message, getClientAPI()->getNewMessageId()); // message id
174 message.emplace_back(static_cast<uint8_t>(CommandType::getting));
175 message.emplace_back(0);
176 message.emplace_back(0);
177 sendMessage(message);
178 break;
179 }
180 default:
181 {
182 Log::error("Unhandled authentication mode: ", static_cast<int>(m_api->getMode()));
183 break;
184 }
185 }
186
187 Log::debug(header(), "Client protocol type: ", NXProtocol::typeToString(type));
188 switch (type)
189 {
190 case NXProtocol::Type::BOOST_TCP_CLIENT:
191 Log::debug(header(), "Waiting for Boost TCP to be ready");
192 getClientAPI()->waitUntilBoostTCPReady();
193 Log::debug(header(), "Boost TCP is ready");
194 break;
195 case NXProtocol::Type::BOOST_UDP_CLIENT:
196 Log::debug(header(), "Waiting for Boost UDP to be ready");
197 getClientAPI()->waitUntilBoostUDPReady();
198 Log::debug(header(), "Boost UDP is ready");
199 break;
200 case NXProtocol::Type::AF_INET_TCP_CLIENT:
201 getClientAPI()->waitUntilInetTCPReady();
202 break;
203 case NXProtocol::Type::AF_INET_UDP_CLIENT:
204 getClientAPI()->waitUntilInetUDPReady();
205 break;
206 case NXProtocol::Type::AF_UNIX_SOCK_STREAM_CLIENT:
207 getClientAPI()->waitUntilUnixStreamReady();
208 break;
209 case NXProtocol::Type::AF_UNIX_SOCK_DGRAM_CLIENT:
210 getClientAPI()->waitUntilUnixDgramReady();
211 break;
212
213 case NXProtocol::Type::AF_INET_TCP_SERVER:
214 case NXProtocol::Type::AF_INET_UDP_SERVER:
215 case NXProtocol::Type::AF_UNIX_SOCK_DGRAM_SERVER:
216 case NXProtocol::Type::AF_UNIX_SOCK_STREAM_SERVER:
217 case NXProtocol::Type::BOOST_TCP_SERVER:
218 case NXProtocol::Type::BOOST_UDP_SERVER:
219 Log::error(header(), "This function cannot be called via server protocol");
220 return;
221
222 case NXProtocol::Type::UNKNOWN:
223 Log::error(header(), "This function called via unknown protocol");
224 return;
225 }
226}

101void UDPClient::createSocket()
102{
103 m_clientSocket = socket(AF_INET, SOCK_DGRAM, 0);
104
105 if (m_clientSocket == -1)
106 {
107 Log::error("UDPClient: Error creating socket: ", strerror(errno));
108 return;
109 }
110
111 // Set a receive timeout so the receive loop can periodically check the
112 // stopped flag and exit cleanly on stop() instead of blocking forever.
113 timeval timeout{};
114 timeout.tv_sec = 0;
115 timeout.tv_usec = 100000; // 100 ms
116 if (setsockopt(m_clientSocket, SOL_SOCKET, SO_RCVTIMEO, &timeout, sizeof(timeout)) == -1)
117 {
118 Log::error("UDPClient: Error setting receive timeout: ", strerror(errno));
119 }
120}

122void UDPClient::start()
123{
124 if (m_stopped->load())
125 {
126 Log::debug("UDPClient already stopped, cannot start");
127 return;
128 }
129
130 uint16_t serverPort = getClientAPI()->getInetUDPServerPortNumber();
131 if (serverPort == 0)
132 {
133 serverPort = Ports::getInetUDPPort();
134 }
135
136 const auto& serverAddress = getClientAPI()->getInetUDPServerAddress();
137 if (inet_pton(AF_INET, serverAddress.c_str(), &m_serverAddr.sin_addr) <= 0)
138 {
139 Log::error("UDPClient: Invalid server address: ", serverAddress);
140 updateProtocolStatus(ProtocolStatus::error);
141 return;
142 }
143 m_serverAddr.sin_port = htons(serverPort);
144
145 createSocket();
146
147 if (m_clientSocket == -1)
148 {
149 Log::error("UDPClient: cannot start, socket is not open");
150 updateProtocolStatus(ProtocolStatus::error);
151 return;
152 }
153
154 m_receiveThread = std::thread(&UDPClient::receiveLoop, this);
155
156 try
157 {
158 ClientProtocol::start(getType());
159 }
160 catch (const std::exception& e)
161 {
162 Log::error("Failed to start UDPClient: ", e.what());
163 stop();
164 }

167void UDPClient::stop()
168{
169 if (!m_stopped || m_stopped->exchange(true))
170 {
171 Log::debug("UDPClient stop already in progress or completed");
172 return;
173 }
174
175 std::unique_lock<std::mutex> lock(*m_mutex);
176
177 if (m_clientSocket != -1)
178 {
179 if (close(m_clientSocket) == -1)
180 {
181 Log::error("UDPClient: Error closing socket: ", strerror(errno));
182 }
183 m_clientSocket = -1;
184 }
185 lock.unlock();
186
187 if (m_receiveThread.joinable())
188 {
189 Log::debug("UDPClient closing receiveThread");
190 m_receiveThread.join();
191 }
192
193 Log::debug("UDPClient stopped");
194}

196void UDPClient::receiveLoop()
197{
198 while (!m_stopped->load() && m_clientSocket != -1)
199 {
200 nx_data buffer(NEXILIS_BUFFER);
201
202 sockaddr_in srcAddr{};
203 socklen_t srcAddrLen = sizeof(srcAddr);
204 ssize_t bytesRead = recvfrom(m_clientSocket, buffer.data(), buffer.size(), 0,
205 reinterpret_cast<sockaddr*>(&srcAddr), &srcAddrLen);
206
207 if (bytesRead > 0)
208 {
209 if (srcAddr.sin_addr.s_addr != m_serverAddr.sin_addr.s_addr ||
210 srcAddr.sin_port != m_serverAddr.sin_port)
211 continue;
212 buffer.resize(bytesRead);
213 Log::info("UDPClient received ", bytesRead, " bytes");
214
215 if (udp_auth::hasPrefix(buffer, udp_auth::challenge))
216 {
217 if (buffer.size() == 4 + udp_auth::cookie_size)
218 {
219 nx_data pending;
220 {
221 std::lock_guard<std::mutex> lock(*m_mutex);
222 pending = m_pendingAuth;
223 }
224 if (!pending.empty())
225 {
226 nx_data cookie(buffer.begin() + 4, buffer.end());
227 auto auth = udp_auth::makeAuthentication(cookie, pending);
228 std::lock_guard<std::mutex> lock(*m_mutex);
229 sendData(reinterpret_cast<const char*>(auth.data()), auth.size());
230 }
231 }
232 continue;
233 }
234
235 auto result = ClientProtocol::getClientAPI()->readMessage(buffer);
236 if (result == ReadResult::success)
237 {
238 Log::debug("UDPClient: Message read successfully");
239 }
240 else
241 {
242 Log::error("UDPClient: Received unexpected message");
243 Util::debugUint8Vector(buffer);
244 }
245 }
246 else if (bytesRead == -1)
247 {
248 if (errno != EINTR && errno != EAGAIN && errno != EWOULDBLOCK)
249 {
250 Log::error("UDPClient: Error receiving message: ", strerror(errno));
251 }
252 }
253 }

256bool UDPClient::sendData(const char* data, size_t dataSize)
257{
258 if (m_clientSocket == -1)
259 {
260 Log::error("UDPClient::sendData(): socket is not open");
261 return false;
262 }
263
264 ssize_t sentBytes = sendto(m_clientSocket, data, dataSize, 0,
265 reinterpret_cast<const sockaddr*>(&m_serverAddr),
266 sizeof(m_serverAddr));
267
268 if (sentBytes == -1)
269 {
270 Log::error("UDPClient: Failed to send message to server: ", strerror(errno));
271 return false;
272 }
273
274 Log::debug("UDPClient sent ", sentBytes, " bytes to server");
275 return true;
276}

278void UDPClient::sendMessage(const nx_data& message)
279{
280 if (m_stopped->load())
281 {
282 Log::error("UDPClient::sendMessage(): client is stopped");
283 return;
284 }
285
286 if (m_clientSocket == -1)
287 {
288 Log::error("UDPClient::sendMessage(): socket is not open");
289 return;
290 }
291
292 nx_data packet;
293 auto token = getClientAPI()->getUdpToken();
294 if (token.empty())
295 {
296 packet = udp_auth::makeHello();
297 }
298 else
299 {
300 packet = udp_auth::makeSession(message, token);
301 }
302 if (packet.empty())
303 return;
304 std::lock_guard<std::mutex> lock(*m_mutex);
305 if (token.empty())
306 m_pendingAuth = message;
307 sendData(reinterpret_cast<const char*>(packet.data()), packet.size());
308}

315std::future<void> UDPClient::sendMessageAsync(const nx_data& message)
316{
317 if (m_stopped->load())
318 {
319 std::promise<void> promise;
320 promise.set_exception(std::make_exception_ptr(std::runtime_error("Client is stopped")));
321 return promise.get_future();
322 }
323
324 if (m_clientSocket == -1)
325 {
326 std::promise<void> promise;
327 promise.set_exception(std::make_exception_ptr(std::runtime_error("Socket is not open")));
328 return promise.get_future();
329 }
330
331 auto promise = std::make_shared<std::promise<void>>();
332 auto future = promise->get_future();
333
334 auto packet = udp_auth::makeSession(message, getClientAPI()->getUdpToken());
335 if (packet.empty())
336 {
337 promise->set_exception(std::make_exception_ptr(std::runtime_error("UDP session is not authenticated")));
338 return future;
339 }
340 auto messageCopy = std::make_shared<nx_data>(std::move(packet));
341
342 std::thread([this, messageCopy, promise]()
343 {
344 try
345 {
346 std::lock_guard<std::mutex> lock(*m_mutex);
347
348 if (m_clientSocket == -1)
349 {
350 throw std::runtime_error("Socket is not open");
351 }
352
353 if (!sendData(reinterpret_cast<const char*>(messageCopy->data()), messageCopy->size()))
354 {
355 throw std::runtime_error(std::string("Failed to send message: ") + strerror(errno));
356 }
357
358 promise->set_value();
359 }
360 catch (...)
361 {
362 promise->set_exception(std::current_exception());
363 } })
364 .detach();
365
366 return future;
367}

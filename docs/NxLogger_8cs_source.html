44 public void AddHandler(Action<Logger.LogLevel, string> logFunction)
45 {
46 _logCallback = logFunction ?? throw new ArgumentNullException(nameof(logFunction));
47
48 try
49 {
50 // Create and register the function handler.

166 private void SafeInvokeLogCallback(Logger.LogLevel level, string message)
167 {
168 if (_disposed) throw new ObjectDisposedException(nameof(NxLogger));
169
170 _lock.EnterReadLock();
171 try
172 {
173 _logCallback?.Invoke(level, message);

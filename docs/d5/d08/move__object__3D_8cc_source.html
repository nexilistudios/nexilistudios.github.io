24CommandResult ServerImpl::room_object3d_move(const DefaultArgs& args)
25{
26 auto& user = args.getUser();
27
28 // Get messagedata.
29 auto payload = Util::removeAmountOfBytesFromVector(args.getData(), 3);
30 auto objectId = Util::uint64FromFront(payload);
31 auto position = Util::vector3fFromFront(Util::removeAmountOfBytesFromVector(payload, 8));
32
33 // Get object from server storage.
34 auto room = RoomStorage::getRoomById(user.getRoomId());
35 if (!room)
36 {
37 Log::error("Client room not found!");
38 return CommandResult::failure;
39 }
40
41 auto object = room->getObject3DById(objectId);
42 if (!object)
43 {
44 Log::error("Object not found with id: ", objectId);
45 return CommandResult::failure;
46 }
47
48 auto oldPosition = object->getPosition();
49 auto newPosition = oldPosition + position;
50
51 std::map<std::string, boost::json::value> params{
52 {"id", boost::json::value(objectId)},
53 {"x", boost::json::value(newPosition.x)},
54 {"y", boost::json::value(newPosition.y)},
55 {"z", boost::json::value(newPosition.z)}};
56
57 // Move object in server storage.
58 object->setPosition(newPosition);
59
60 auto roomCommand = Command::createRoomCommand(user.getRoomId(), user, args.getData(), params, args.getMessageId());
61 if (Command::sendRoomCommand(roomCommand, user, args.getProtocol()))
62 {
63 return CommandResult::success;
64 }
65 else
66 {
67 return CommandResult::failed_room_send;
68 }
69}

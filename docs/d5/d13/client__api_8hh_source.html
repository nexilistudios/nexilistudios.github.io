110 uint64_t nextMessageId()
111 {
112 uint64_t id = m_nextMessageId.load();
113 do
114 {
115 if (id == 0 || id == NEXILIS_MAX)
116 throw std::overflow_error("Client message ID sequence exhausted");
117 } while (!m_nextMessageId.compare_exchange_weak(id, id + 2));
118 return id;
119 }

127 void setUdpToken(std::string token)
128 {
129 std::lock_guard<std::mutex> lock(m_udpTokenMutex);
130 m_udpToken = std::move(token);
131 if (m_responseWindow)

151 bool acceptUdpResponseId(uint64_t id)
152 {
153 if (id == 0 || !m_responseWindow)
154 return false;
155 std::lock_guard lock(m_responseWindow->mutex);
156 auto& window = *m_responseWindow;
157 if (id > window.highest)
158 {
159 const auto delta = id - window.highest;
160 if (delta >= window.seen.size())
161 window.seen.reset();
162 else
163 window.seen <<= delta;
164 window.highest = id;
165 window.seen.set(0);
166 return true;
167 }
168 const auto offset = window.highest - id;
169 if (offset >= window.seen.size() || window.seen.test(offset))
170 return false;
171 window.seen.set(offset);
172 return true;
173 }

547 void discardRoomEvents(uint64_t roomId, bool wasCurrentRoom)
548 {
549 std::lock_guard<std::mutex> lock(*m_damageMutex);
550 std::erase_if(m_pendingAnimationEvents,
551 [roomId](const AnimationEvent& event)
552 { return event.room_id == roomId; });
553 std::erase_if(m_pendingServerRoomEvents,
554 [roomId](const ServerRoomEvent& event)
555 { return event.room_id == roomId; });
556 std::erase_if(m_pendingMatchEvents,
557 [roomId](const std::unique_ptr<MatchEvent>& event)
558 { return event && event->room_id == roomId; });
559 if (wasCurrentRoom)
560 {
561 m_pendingDamageEvents.clear();
562 m_pendingRespawnEvents.clear();
563 m_pendingLeaderboardEvents.clear();
564 m_pendingAudioEvents.clear();
565 m_pendingPlayer2DShootEvents.clear();
566 }

806 std::vector<ChatSnapshot> getChatSnapshot(uint64_t room_id)
807 {
808 std::lock_guard<std::mutex> lock(*m_clientAPIData.getRoomsMutex());
809 std::vector<ChatSnapshot> result;
810 for (const auto& room : m_clientAPIData.getCurrentlyActiveRooms())
811 {
812 if (room.getId() != room_id)
813 continue;
814
815 const auto& messages = room.getMessages();
816 std::transform(messages.begin(), messages.end(), std::back_inserter(result),
817 [](const Room::Communication& communication)
818 { return ChatSnapshot{communication.getSenderId(), communication.getPayload()}; });
819 break;
820 }
821 return result;
822 }

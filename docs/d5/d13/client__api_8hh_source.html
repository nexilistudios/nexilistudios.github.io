103 uint64_t nextMessageId()
104 {
105 uint64_t id = m_nextMessageId.load();
106 do
107 {
108 if (id == 0 || id == NEXILIS_MAX)
109 throw std::overflow_error("Client message ID sequence exhausted");
110 } while (!m_nextMessageId.compare_exchange_weak(id, id + 2));
111 return id;
112 }

665 std::vector<ChatSnapshot> getChatSnapshot(uint64_t room_id)
666 {
667 std::lock_guard<std::mutex> lock(*m_clientAPIData.getRoomsMutex());
668 std::vector<ChatSnapshot> result;
669 for (const auto& room : m_clientAPIData.getCurrentlyActiveRooms())
670 {
671 if (room.getId() != room_id)
672 continue;
673
674 const auto& messages = room.getMessages();
675 std::transform(messages.begin(), messages.end(), std::back_inserter(result),
676 [](const Room::Communication& communication)
677 { return ChatSnapshot{communication.getSenderId(), communication.getPayload()}; });
678 break;
679 }
680 return result;
681 }

55static BOOL WINAPI consoleCtrlHandler(DWORD ctrlType)
56{
57 switch (ctrlType)
58 {
59 case CTRL_C_EVENT:
60 case CTRL_CLOSE_EVENT:
61 case CTRL_BREAK_EVENT:
62 case CTRL_LOGOFF_EVENT:
63 case CTRL_SHUTDOWN_EVENT:
64 SignalHandler::s_shutdownRequested.store(true, std::memory_order_release);
65 SetEvent(shutdownEvent);
66 return TRUE;
67 default:
68 return FALSE;
69 }

108static void signalHandler(int signum)
109{
110 const int savedErrno = errno;
111 (void)signum;
112 SignalHandler::s_shutdownRequested.store(true, std::memory_order_release);
113 const char wake = 1;
114 const auto result = write(shutdownPipe[1], &wake, 1);
115 (void)result;
116 errno = savedErrno;
117}

119void SignalHandler::install()
120{
121 static std::once_flag flag;
122 std::call_once(flag, []()
123 {
124 if (pipe(shutdownPipe) != 0)
125 throw std::system_error(errno, std::generic_category(), "pipe");
126 const int flags = fcntl(shutdownPipe[1], F_GETFL);
127 if (flags == -1 || fcntl(shutdownPipe[1], F_SETFL, flags | O_NONBLOCK) == -1)
128 {
129 const int error = errno;
130 close(shutdownPipe[0]);
131 close(shutdownPipe[1]);
132 shutdownPipe[0] = shutdownPipe[1] = -1;
133 throw std::system_error(error, std::generic_category(), "fcntl");
134 }
135 struct sigaction sa;
136 sa.sa_handler = signalHandler;
137 sigemptyset(&sa.sa_mask);
138 sa.sa_flags = SA_RESTART;
139
140 if (sigaction(SIGINT, &sa, nullptr) != 0 ||
141 sigaction(SIGTERM, &sa, nullptr) != 0)
142 throw std::system_error(errno, std::generic_category(), "sigaction");
143
144#ifdef SIGHUP
145 if (sigaction(SIGHUP, &sa, nullptr) != 0)
146 throw std::system_error(errno, std::generic_category(), "sigaction");
147#endif
148 });
149}

158void SignalHandler::waitForShutdown()
159{
160 install();
161#if defined(_WIN32)
162 if (!isShutdownRequested() && WaitForSingleObject(shutdownEvent, INFINITE) != WAIT_OBJECT_0)
163 throw std::system_error(static_cast<int>(GetLastError()), std::system_category(),
164 "WaitForSingleObject");
165#else
166 while (!isShutdownRequested())
167 {
168 char wake;
169 if (read(shutdownPipe[0], &wake, 1) == -1 && errno != EINTR)
170 throw std::system_error(errno, std::generic_category(), "read");
171 }
172#endif
173 if (!callbackDelivered.exchange(true, std::memory_order_acq_rel))
174 {
175 std::function<void()> callback;
176 {
177 std::lock_guard<std::mutex> lock(callbackMutex);
178 callback = shutdownCallback;
179 }
180 if (callback)
181 callback();
182 }
183}

185void SignalHandler::clearShutdownRequested() noexcept
186{
187#if defined(_WIN32)
188 s_shutdownRequested.store(false, std::memory_order_release);
189 if (shutdownEvent != nullptr)
190 {
191 ResetEvent(shutdownEvent);
192 if (s_shutdownRequested.load(std::memory_order_acquire))
193 SetEvent(shutdownEvent);
194 }
195#else
196 s_shutdownRequested.store(false, std::memory_order_release);
197#endif
198 callbackDelivered.store(false, std::memory_order_release);
199}

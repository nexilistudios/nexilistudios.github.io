34 ReadResult execute(ClientAPI&, ClientAPI::ClientAPIData& data) override
35 {
36 auto& mtx = data.getRoomsMutex();
37 std::lock_guard<std::mutex> lock(*mtx);
38
39 if (m_action == "create")
40 {
41 for (auto& room : data.getCurrentlyActiveRooms())
42 {
43 if (room.getId() == m_room_id)
44 {
45 auto object = Object2D(m_object_id, {m_x, m_y}, {m_w, m_h});
46 object.setFilepath(m_filepath);
47 room.addObject(std::move(object));
48 return ReadResult::success;
49 }
50 }
51 return ReadResult::failure;
52 }
53 else if (m_action == "destroy")
54 {
55 for (auto& room : data.getCurrentlyActiveRooms())
56 {
57 if (room.getId() == m_room_id)
58 {
59 room.deleteObject2D(m_object_id);
60 return ReadResult::success;
61 }
62 }
63 return ReadResult::failure;
64 }
65 else if (m_action == "move")
66 {
67 for (auto& room : data.getCurrentlyActiveRooms())
68 {
69 if (room.getId() == m_room_id)
70 {
71 auto object = room.getObject2DById(m_object_id);
72 if (object)
73 {
74 object->setPosition({m_x, m_y});
75 return ReadResult::success;
76 }
77 }
78 }
79 return ReadResult::failure;
80 }
81 else if (m_action == "create_moving")
82 {
83 // Both createMoving and createMovingTest use same logic
84 for (auto& room : data.getCurrentlyActiveRooms())
85 {
86 if (room.getId() == m_room_id)
87 {
88 // Movement updates only carry the object id and its new
89 // position; the server reuses the "id"/x/y keys of the
90 // create broadcast.
91 if (m_create_moving_type == "update")
92 {
93 auto object = room.getObject2DById(m_object_id);
94 if (object)
95 {
96 object->setPosition({m_x, m_y});
97 return ReadResult::success;
98 }
99 return ReadResult::failure;
100 }
101
102 auto object = Object2D(m_object_id, {m_x, m_y}, {m_w, m_h});
103 object.setFilepath(m_filepath);
104 room.addObject(std::move(object));
105 return ReadResult::success;
106 }
107 }
108 return ReadResult::failure;
109 }
110 return ReadResult::not_found;
111 }
112

120 float m_h;
121 std::string m_filepath;

28static nexilis::Vector3f convert_vector3f(nexilis_Vector3f* vector)
29{
30 float x = vector->vec->x;
31 float y = vector->vec->y;
32 float z = vector->vec->z;
33 nexilis::FileLog::debug("Converted position x:", x, " y:", y, " z:", z);
34 return nexilis::Vector3f(x, y, z);
35}

54{

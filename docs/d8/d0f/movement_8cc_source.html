49std::thread Movement::object2D(std::unique_ptr<Movement2D> movement, User& user, NXProtocol& protocol)
50{
51 // clang-format off
52 return std::thread([movement = std::move(movement), &user, &protocol]()
53 {
54 try
55 {
56 auto* room = RoomStorage::getRoomById(user.getRoomId());
57 auto* serverObject = room->getObject2DById(movement->getObjectId());
58 // Capture start position so each tick computes startPos + f(progress)
59 // instead of accumulating offsets onto an already-moved position.
60 Vector2f startPosition = serverObject->getPosition();
61
62 Command::runWithTickrate(m_settings->getTickrate(), movement->getDeltatime(), [&movement, &user, &protocol, startPosition](double progress)
63 {
64 auto func = movement->getMovementFunc();
65 auto amount = movement->getAmount();
66
67 auto* room = RoomStorage::getRoomById(user.getRoomId());
68 auto* serverObject = room->getObject2DById(movement->getObjectId());
69
70 Vector2f newPosition = Vector2f(startPosition.x + func(progress, amount.x),
71 startPosition.y + func(progress, amount.y));
72
73 serverObject->setPosition(newPosition);
74 std::map<std::string, boost::json::value> messageParams
75 {
76 {"createMovingType", boost::json::value("update")},
77 {"id", boost::json::value(serverObject->getId())},
78 {"x", boost::json::value(newPosition.x)},
79 {"y", boost::json::value(newPosition.y)}
80 };
81
82 auto roomCommand = Command::createRoomCommand(user.getRoomId(), user, movement->getMessageData(), messageParams, movement->getMessageId());
83 Command::sendRoomCommand(roomCommand, user, protocol);
84 });
85 }
86 catch (const std::exception& e)
87 {
88 Log::error(e.what());
89 }
90 });
91 // clang-format on
92}

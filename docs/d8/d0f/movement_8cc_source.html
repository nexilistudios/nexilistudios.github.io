51void Movement::object2D(std::unique_ptr<Movement2D> movement, User& user, NXProtocol& protocol)
52{
53 if (!movement)
54 return;
55 auto room = RoomStorage::getRoomById(user.getRoomId());
56 if (!room)
57 return;
58 const auto amount = movement->getAmount();
59 const auto func = movement->getMovementFunc();
60 const auto& limits = GameplayLimits::get();
61 // Complete the displacement during dispatch; no task retains the user,
62 // protocol, room, or object after the command returns.
63 Vector2f position;
64 const bool found = room->withObject2D(movement->getObjectId(), [&](Object2D& object)
65 {
66 const auto start = object.getPosition();
67 position = Vector2f(limits.clampHorizontal(start.x + func(1.0, amount.x)),
68 limits.clampHorizontal(start.y + func(1.0, amount.y)));
69 if (validation::isFinite(position))
70 object.setPosition(position); });
71 if (!found || !validation::isFinite(position))
72 return;
73 std::map<std::string, boost::json::value> params{
74 {"createMovingType", boost::json::value("update")},
75 {"id", boost::json::value(movement->getObjectId())},
76 {"x", boost::json::value(position.x)},
77 {"y", boost::json::value(position.y)}};
78 auto command = Command::createRoomCommand(user.getRoomId(), user, movement->getMessageData(), params, movement->getMessageId());
79 Command::sendRoomCommand(command, user, protocol);
80}

82void Movement::object3D(std::unique_ptr<Movement3D> movement, User& user, NXProtocol& protocol)
83{
84 if (!movement)
85 return;
86 auto room = RoomStorage::getRoomById(user.getRoomId());
87 if (!room)
88 return;
89 const auto amount = movement->getAmount();
90 const auto func = movement->getMovementFunc();
91 const auto& limits = GameplayLimits::get();
92 Vector3f position;
93 const bool found = room->withObject3D(movement->getObjectId(), [&](Object3D& object)
94 {
95 const auto start = object.getPosition();
96 position = Vector3f(limits.clampHorizontal(start.x + func(1.0, amount.x)),
97 limits.clampHeight(start.y + func(1.0, amount.y)),
98 limits.clampHorizontal(start.z + func(1.0, amount.z)));
99 if (validation::isFinite(position))
100 object.setPosition(position); });
101 if (!found || !validation::isFinite(position))
102 return;
103 std::map<std::string, boost::json::value> params{
104 {"createMovingType", boost::json::value("update")},
105 {"id", boost::json::value(movement->getObjectId())},
106 {"x", boost::json::value(position.x)},
107 {"y", boost::json::value(position.y)},
108 {"z", boost::json::value(position.z)}};
109 auto command = Command::createRoomCommand(user.getRoomId(), user, movement->getMessageData(), params, movement->getMessageId());
110 Command::sendRoomCommand(command, user, protocol);
111}

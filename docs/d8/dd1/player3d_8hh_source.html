26class RoomPlayer3DCommand : public BaseAPICommand
27{
28public:
29 RoomPlayer3DCommand(std::string action, uint64_t client_id, float x, float y, float z,
30 uint64_t target_id = 0, float damage = 0.0f, float new_health = 0.0f,
31 float yaw = 0.0f)
32 : m_action(action), m_client_id(client_id), m_x(x), m_y(y), m_z(z),
33 m_target_id(target_id), m_damage(damage), m_new_health(new_health), m_yaw(yaw)
34 {
35 }
36
37 ReadResult execute(ClientAPI&, ClientAPI::ClientAPIData& data) override
38 {
39 auto& mtx = data.getRoomsMutex();
40 std::lock_guard<std::mutex> lock(*mtx);
41
42 auto& rooms = data.getCurrentlyActiveRooms();
43
44 if (m_action == "position")
45 {
46 for (auto&& room = rooms.begin(); room != rooms.end(); room++)
47 {
48 for (auto& client : room->getClients())
49 {
50 if (client.getId() == m_client_id)
51 {
52 client.getObject3D().setPosition(Vector3(m_x, m_y, m_z));
53 client.getObject3D().setYaw(m_yaw);
54 return ReadResult::success;
55 }
56 }
57 }
58 return ReadResult::command_execution;
59 }
60 else if (m_action == "dimensions")
61 {
62 for (auto&& room = rooms.begin(); room != rooms.end(); room++)
63 {
64 for (auto& client : room->getClients())
65 {
66 if (client.getId() == m_client_id)
67 {
68 client.getObject3D().setDimensions(Vector3(m_x, m_y, m_z));
69 return ReadResult::success;
70 }
71 }
72 }
73 return ReadResult::command_execution;
74 }
75 else if (m_action == "movement")
76 {
77 for (auto&& room = rooms.begin(); room != rooms.end(); room++)
78 {
79 for (auto& client : room->getClients())
80 {
81 if (client.getId() == m_client_id)
82 {
83 client.getObject3D().setPosition(Vector3(m_x, m_y, m_z));
84 client.getObject3D().setYaw(m_yaw);
85 return ReadResult::success;
86 }
87 }
88 }
89 return ReadResult::command_execution;
90 }
91 else if (m_action == "shoot")
92 {
93 ClientAPI::DamageEvent event;
94 event.shooter_id = m_client_id;
95 event.target_id = m_target_id;
96 event.damage = m_damage;
97 event.new_health = m_new_health;
98 data.pushDamageEvent(std::move(event));
99 return ReadResult::success;
100 }
101 else if (m_action == "respawn")
102 {
103 ClientAPI::RespawnEvent event;
104 event.target_id = m_target_id;
105 data.pushRespawnEvent(std::move(event));
106 return ReadResult::success;
107 }
108 return ReadResult::not_found;
109 }
110
111private:
112 std::string m_action;
113 uint64_t m_client_id;
114 float m_x;
115 float m_y;
116 float m_z;
117 uint64_t m_target_id;
118 float m_damage;
119 float m_new_health;
120 float m_yaw;
121};

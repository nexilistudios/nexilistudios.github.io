94void nexilis_boost_tcp_client_send_message_with_callback(nexilis_BoostTCPClient* client, const uint8_t message[], size_t message_size, void (*callback)(const uint8_t*, size_t))
95{
96 if (!client || !client->client)
97 {
98 return;
99 }
100
101 nexilis::nx_data data(message, message + message_size);
102
103 std::function<void()> cb;
104 if (callback)
105 {
106 // Capture the data by value.
107 cb = [callback, data]()
108 {
109 callback(data.data(), data.size());
110 };
111 }
112 else
113 {
114 cb = []() {};
115 }
116
117 client->client->sendMessage(data, cb);
118}

129bool nexilis_start_client(nexilis_ClientAPI* client_api, nexilis_BoostTCPClient* tcp_client)
130{
131 if (!client_api || !client_api->api || !tcp_client || !tcp_client->client)
132 {
133 return false;
134 }
135
136 // 1. Start TCP client (connect + authenticate). start() is void and only
137 // logs on failure, so check the connection status explicitly: without
138 // this, a failed connection would fall through to the init wait below
139 // and block forever.
140 tcp_client->client->start();
141
142 if (tcp_client->client->getProtocolStatus() != nexilis::client::ProtocolStatus::connected)
143 {
144 return false;
145 }
146
147 // 2. Send clientId packet
148 auto clientIdPacket = nexilis::client::Packet::Get::General::clientId(*client_api->api);
149 tcp_client->client->sendMessage(clientIdPacket);
150
151 // 3. Wait for initialization, bounded so a dead connection cannot hang
152 // the caller indefinitely. Also bail out early if the connection drops
153 // while we wait.
154 constexpr auto initTimeout = std::chrono::seconds(5);
155 const auto deadline = std::chrono::steady_clock::now() + initTimeout;
156 while (!client_api->api->isInitialized())
157 {
158 if (tcp_client->client->getProtocolStatus() == nexilis::client::ProtocolStatus::error ||
159 std::chrono::steady_clock::now() >= deadline)
160 {
161 return false;
162 }
163 std::this_thread::sleep_for(std::chrono::milliseconds(10));
164 }
165
166 // 4. Send rooms info request
167 auto roomsPacket = nexilis::client::Packet::Get::Info::rooms(*client_api->api);
168 tcp_client->client->sendMessage(roomsPacket);
169
170 return true;
171}

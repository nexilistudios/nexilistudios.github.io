29boost::json::object ServerJson::getRoomData()
30{
31 boost::json::object roomDataObj;
32
33 auto& rooms = RoomStorage::getAllRooms();
34
35 if (rooms.empty())
36 {
37 Log::info("Empty room data");
38 roomDataObj["room_amount"] = 0;
39 }
40 else
41 {
42 roomDataObj["room_amount"] = rooms.size();
43 auto roomData = roomsToJSON(rooms);
44 roomDataObj["rooms"] = std::move(roomData);
45 }
46 return roomDataObj;
47}

79boost::json::array ServerJson::clientsToJSON(const std::vector<std::unique_ptr<User>>& clients)
80{
81 boost::json::array resultingArray;
82
83 for (const auto& client : clients)
84 {
85 boost::json::object clientObj;
86 if (client->getUsername().empty())
87 {
88 clientObj["username"] = "UNIDENTIFIED_CLIENT";
89 }
90 else
91 {
92 clientObj["username"] = client->getUsername();
93 }
94 clientObj["id"] = client->getId();
95 clientObj["room_id"] = client->getRoomId();
96 resultingArray.emplace_back(std::move(clientObj));
97 }
98 return resultingArray;
99}

101boost::json::array ServerJson::roomsToJSON(const std::vector<Room>& rooms)
102{
103 boost::json::array resultingArray;
104 resultingArray.reserve(rooms.size());
105
106 for (const auto& room : rooms)
107 {
108 boost::json::object roomObj;
109
110 if (room.getName().empty())
111 {
112 roomObj["name"] = "UNIDENTIFIED_ROOM";
113 }
114 else
115 {
116 roomObj["name"] = room.getName();
117 }
118
119 roomObj["max_size"] = room.getMaxSize();
120 roomObj["room_id"] = room.getId();
121 roomObj["creator_id"] = room.getCreatorId();
122 auto ctx = room.getContext();
123 roomObj["context"] = static_cast<uint32_t>(ctx);
124
125 // Get data from clients in a room.
126 boost::json::array clientArray;
127 clientArray.reserve(room.getClients().size());
128
129 auto& roomClients = room.getClients();
130
131 for (auto& client : roomClients)
132 {
133 auto* clientPointer = ClientStorage::getClientById(client);
134
135 if (clientPointer)
136 {
137 boost::json::object clientObj;
138 clientObj["id"] = clientPointer->getId();
139
140 if (clientPointer->getUsername().empty())
141 {
142 Log::warning("Client has empty username!");
143 clientObj["name"] = "";
144 }
145 else
146 {
147 clientObj["name"] = clientPointer->getUsername();
148 }
149
150 auto position2D = clientPointer->getObject2D().getPosition();
151 clientObj["x"] = std::move(position2D.x);
152 clientObj["y"] = std::move(position2D.y);
153
154 auto dimension2D = clientPointer->getObject2D().getDimensions();
155 clientObj["width"] = std::move(dimension2D.x);
156 clientObj["height"] = std::move(dimension2D.y);
157
158 clientArray.emplace_back(std::move(clientObj));
159 }
160 }
161
162 if (roomClients.size() > 0)
163 {
164 roomObj["clients"] = std::move(clientArray);
165 }
166
167 // 2D context cannot have 3D objects.
168 if (ctx != RoomData::Context::_2D)
169 {
170 if (!room.getObjects3D().empty())
171 {
172 boost::json::array obj3DArray;
173 for (const auto& obj : room.getObjects3D())
174 {
175 auto pos = obj.getPosition();
176 auto dim = obj.getDimensions();
177 boost::json::object objData;
178 objData["id"] = obj.getId();
179 objData["x"] = static_cast<double>(pos.x);
180 objData["y"] = static_cast<double>(pos.y);
181 objData["z"] = static_cast<double>(pos.z);
182 objData["w"] = static_cast<double>(dim.x);
183 objData["h"] = static_cast<double>(dim.y);
184 objData["d"] = static_cast<double>(dim.z);
185 objData["filepath"] = obj.getFilepath();
186 obj3DArray.emplace_back(std::move(objData));
187 }
188 roomObj["objects_3d"] = std::move(obj3DArray);
189 }
190 }
191
192 // 3D context can have 2D objects, they just don't have depth.
193 if (!room.getObjects2D().empty())
194 {
195 boost::json::array obj2DArray;
196 for (const auto& obj : room.getObjects2D())
197 {
198 auto pos = obj.getPosition();
199 auto dim = obj.getDimensions();
200 boost::json::object objData;
201 objData["id"] = obj.getId();
202 objData["x"] = static_cast<double>(pos.x);
203 objData["y"] = static_cast<double>(pos.y);
204 objData["w"] = static_cast<double>(dim.x);
205 objData["h"] = static_cast<double>(dim.y);
206 objData["filepath"] = obj.getFilepath();
207 obj2DArray.emplace_back(std::move(objData));
208 }
209 roomObj["objects_2d"] = std::move(obj2DArray);
210 }
211
212 resultingArray.emplace_back(std::move(roomObj));
213 }
214 return resultingArray;
215}

40 ReadResult execute(ClientAPI& api, ClientAPI::ClientAPIData& data) override
41 {
42 auto& mtx = data.getRoomsMutex();
43 std::lock_guard<std::mutex> lock(*mtx);
44
45 auto& rooms = data.getCurrentlyActiveRooms();
46
47 if (m_action == "join")
48 {
49 for (auto& room : rooms)
50 {
51 if (room.getId() == m_room_id)
52 {
53 for (auto& client : room.getClients())
54 {
55 if (client.getId() == m_client_id)
56 {
57 if (!m_username.empty())
58 {
59 client.setUsername(m_username);
60 }
61 return ReadResult::success;
62 }
63 }
64
65 ClientSession session(m_client_id, &api);
66 session.setUsername(m_username);
67 room.addClient(std::move(session));
68 return ReadResult::success;
69 }
70 }
71 return ReadResult::command_execution;
72 }
73 else if (m_action == "leave")
74 {
75 for (auto& room : rooms)
76 {
77 if (room.getId() == m_room_id)
78 {
79 room.removeClient(m_client_id);
80 return ReadResult::success;
81 }
82 }
83 return ReadResult::command_execution;
84 }
85 else if (m_action == "create")
86 {
87 auto roomData = RoomData(m_client_id, m_room_name, m_room_id,
88 static_cast<RoomData::Context>(m_room_context), m_max_size);
89 rooms.emplace_back(Room(roomData, std::vector<ClientSession>()));
90 return ReadResult::success;
91 }
92 else if (m_action == "remove")
93 {
94 const bool wasCurrentRoom = api.getCurrentRoomId() == m_room_id;
95 std::erase_if(rooms, [this](const Room& room)
96 { return room.getId() == m_room_id; });
97 if (wasCurrentRoom)
98 {
99 api.setRoomId(0);
100 api.setOverLapStatus2D(false);
101 }
102 data.discardRoomEvents(m_room_id, wasCurrentRoom);
103 return ReadResult::success;
104 }
105 else if (m_action == "set_overlap")
106 {
107 api.setOverLapStatus2D(m_overlap_allowed);
108 return ReadResult::success;
109 }
110 return ReadResult::not_found;
111 }
112
113private:
114 std::string m_action;
115 uint64_t m_room_id;
116 uint64_t m_client_id;
117 std::string m_room_name;
118 uint64_t m_room_context;
119 std::string m_username;
120 bool m_overlap_allowed;
121 uint32_t m_max_size;
122};

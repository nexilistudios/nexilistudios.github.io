118 void addClient(ClientSession&& client)
119 {
120 const auto& clientId = client.getId();
121 const auto duplicate = std::find_if(m_clients.begin(), m_clients.end(),
122 [&clientId](const ClientSession& session)
123 { return session.getId() == clientId; });
124 if (duplicate != m_clients.end())
125 return;
126
127 m_clients.emplace_back(std::move(client));
128 }

133 [&clientId](const ClientSession& session)

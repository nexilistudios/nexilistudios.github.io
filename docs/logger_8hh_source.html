145 template <typename T, typename... Args>
146 void concatAndEmit(const LogLevel& logLevel, const T& data, const Args&... args)
147 {
148 // Return if loglevel is not on.
149 if (!getLevel(logLevel))
150 {
151 return;
152 }
153
154 // Concat arguments.
155 std::stringstream ss;
156
157 auto processArg = [&ss](const auto& arg)
158 {
159 using ArgType = std::decay_t<decltype(arg)>;
160
161 if constexpr (std::is_same_v<ArgType, nx_data>)
162 {
163 ss << std::string_view(Util::convertToString(arg));
164 }
165 else if constexpr (std::is_convertible_v<ArgType, std::string_view>)
166 {
167 // For string-like types that can be converted to string_view.
168 ss << std::string_view(arg);
169 }
170 else
171 {
172 // For all other types.
173 ss << arg;
174 }
175 };
176
177 // Process the first argument.
178 processArg(data);
179
180 // Process the remaining arguments.
181 (processArg(args), ...);
182
183 emitLog(logLevel, ss.str());
184 }
185
186 template <typename T>
187 void emitLog(const LogLevel& logLevel, const T& data)
188 {
189 // logLevel to string
190 std::string logLevelStr;
191 switch (logLevel)
192 {
193 case LogLevel::Debug:
194 logLevelStr = "DEBUG: ";
195 break;
196 case LogLevel::Info:
197 logLevelStr = "INFO: ";
198 break;
199 case LogLevel::Warning:
200 logLevelStr = "WARNING: ";
201 break;
202 case LogLevel::Error:
203 logLevelStr = "ERROR: ";
204 break;
205 case LogLevel::Critical:
206 logLevelStr = "CRITICAL: ";
207 break;
208 }
209
210 // Create final string to be logged.
211 std::ostringstream ss;
212 ss << logLevelStr << data;
213
214 std::lock_guard<std::mutex> lock(m_mtx);
215
216 // emit message to all handlers
217 for (auto& handler : m_handlers)
218 {
219 if (handler)
220 {
221 handler->emit(logLevel, ss.str());
222 }
223 }
224 }
225
226private:
228 std::vector<std::unique_ptr<BaseHandler>> m_handlers;
229
231 uint8_t m_logLevel = 0;
232
233 std::mutex m_mtx;
234};

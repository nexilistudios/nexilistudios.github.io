53 public Room GetRoom(ulong index)

95 public IReadOnlyList<RoomInfo> GetRoomInfos()
96 {
97 ThrowIfDisposed();
98
99 var infos = new List<RoomInfo>((int)Count);
100 for (ulong i = 0; i < Count; i++)
101 {
102 var ptr = RoomsCollectionNative.GetRoom(_nativeCollection, i);
103 if (ptr == IntPtr.Zero)
104 {
105 continue;
106 }
107
108 // Read everything out of the shared wrapper right away, the
109 // next GetRoom call overwrites it.
110 using (var room = new Room(ptr, ownsNativeInstance: false))
111 {
112 infos.Add(RoomInfo.FromRoom(room));
113 }
114 }
115 return infos.AsReadOnly();

17public static class FileLog
18{
19 static readonly object _lock = new object();
20 static readonly string _logFilePath = "/tmp/nexilis/csharp-bindings-log.txt";
21
22 static FileLog()
23 {
24 // Ensure the directory exists
25 Directory.CreateDirectory(Path.GetDirectoryName(_logFilePath));
26
27 // Initialize log file (or clear existing)
28 lock (_lock)
29 {
30 File.WriteAllText(_logFilePath, $"[{DateTime.UtcNow:yyyy-MM-dd HH:mm:ss.fff}] LOG INITIALIZED\n");
31 }
32 }
33
34 public static void Log(string message)
35 {
36 lock (_lock)
37 {
38 try
39 {
40 string logEntry = $"[{DateTime.UtcNow:yyyy-MM-dd HH:mm:ss.fff}] {message}\n";
41 File.AppendAllText(_logFilePath, logEntry);
42 }
43 catch (Exception ex)
44 {
45 Console.Error.WriteLine($"LOGGER FAILED: {ex.Message}");
46 }
47 }
48 }

36struct UDPServer::SocketState
37{
38 SocketState()
39 : socket(context, boost::asio::ip::udp::endpoint(boost::asio::ip::udp::v4(), Ports::getBoostUDPPort()))
40 {
41 socket.set_option(boost::asio::ip::udp::socket::reuse_address(true));
42 socket.non_blocking(true);
43 }
44
45 void disableSends()
46 {
47 std::lock_guard lock(mutex);
48 acceptingSends = false;
49 }
50
51 void close()
52 {
53 std::lock_guard lock(mutex);
54 acceptingSends = false;
55 boost::system::error_code ec;
56 socket.close(ec);
57 }
58
59 void send(const boost::asio::ip::udp::endpoint& endpoint, const nx_data& bytes)
60 {
61 std::lock_guard lock(mutex);
62 if (!acceptingSends || !socket.is_open())
63 throw std::runtime_error("Boost UDP socket is closed");
64 socket.send_to(boost::asio::buffer(bytes), endpoint);
65 }
66
67 size_t receive(nx_data& bytes, boost::asio::ip::udp::endpoint& endpoint,
68 boost::system::error_code& ec)
69 {
70 std::lock_guard lock(mutex);
71 return socket.receive_from(boost::asio::buffer(bytes), endpoint, 0, ec);
72 }
73
74 // All operations on this Asio socket share the mutex. Nonblocking reads
75 // release it immediately so sends and shutdown cannot wait on a datagram.
76 boost::asio::io_context context;
77 boost::asio::ip::udp::socket socket;
78 std::mutex mutex;
79 bool acceptingSends = true;
80};

138void UDPServer::start()
139{
140 // Password-protected sessions are established over pinned TLS Boost TCP;
141 // gameplay UDP is authenticated with the per-session token MAC.
142 if (!m_stopped || m_stopped->load() || m_receiveThread.joinable())
143 return;
144
145 Log::debug("Boost UDP server started on port: ", m_serverPort);
146 if (!Util::writePortToFile(m_serverPort, getType()))
147 Log::error("Failed to write Boost UDP server port to a file");
148
149 m_receiveThread = std::thread(&UDPServer::receiveFromClients, this);
150}

152void UDPServer::stop()
153{
154 if (!m_stopped)
155 return;
156 m_stopped->store(true);
157 m_socketState->disableSends();
158 // An application callback may stop dispatch on the receive worker. An
159 // external stop/destructor must still join it before destroying the owner.
160 if (m_receiveThread.joinable() && m_receiveThread.get_id() == std::this_thread::get_id())
161 return;
162 if (m_receiveThread.joinable())
163 m_receiveThread.join();
164 cleanupDatagramTransports(User::Transport::boostUdp, *this);
165 m_socketState->close();
166}

168void UDPServer::receiveFromClients()
169{
170 nx_data receiveBuffer(NEXILIS_BUFFER);
171 boost::asio::ip::udp::endpoint remoteEndpoint;
172 while (!m_stopped->load())
173 {
174 expireIdleDatagramTransports(User::Transport::boostUdp, *this);
175 const auto now = std::chrono::steady_clock::now();
176 if (now - m_lastEndpointSweep >= std::chrono::seconds(1))
177 {
178 m_lastEndpointSweep = now;
179 std::erase_if(m_clientEndpoints, [](const auto& entry)
180 {
181 auto user = ClientStorage::getClientById(entry.first);
182 return !user || !user->hasTransport(User::Transport::boostUdp); });
183 }
184 try
185 {
186 receiveBuffer.resize(NEXILIS_BUFFER);
187
188 boost::system::error_code ec;
189 size_t bytes_received = m_socketState->receive(receiveBuffer, remoteEndpoint, ec);
190 if (ec == boost::asio::error::would_block || ec == boost::asio::error::try_again)
191 {
192 std::this_thread::sleep_for(std::chrono::milliseconds(10));
193 continue;
194 }
195 if (ec)
196 throw boost::system::system_error(ec);
197
198 if (m_stopped->load())
199 {
200 break;
201 }
202
203 if (bytes_received == 0)
204 {
205 continue;
206 }
207
208 std::string address = remoteEndpoint.address().to_string();
209 // Create a new buffer containing only the received bytes.
210 nx_data received_message(receiveBuffer.begin(), receiveBuffer.begin() + bytes_received);
211
212 const auto senderEndpoint = remoteEndpoint;
213 const auto endpointKey = address + ":" + std::to_string(senderEndpoint.port());
214 nx_data payload;
215 if (udp_auth::hasPrefix(received_message, udp_auth::hello))
216 {
217 if (received_message.size() == udp_auth::makeHello().size())
218 {
219 auto challenge = udp_auth::makeChallenge(m_cookieKey, endpointKey);
220 if (!challenge.empty())
221 m_socketState->send(senderEndpoint, challenge);
222 }
223 continue;
224 }
225 if (udp_auth::hasPrefix(received_message, udp_auth::authentication))
226 {
227 if (getSettings().getMode() == AuthenticationMode::password_protected)
228 continue;
229 if (!udp_auth::checkAuthentication(received_message, m_cookieKey, endpointKey, payload))
230 continue;
231 if (payload.size() >= 8 && ClientStorage::getClientById(Util::uint64FromFront(payload)))
232 continue;
233 }
234 else if (udp_auth::hasPrefix(received_message, udp_auth::session))
235 {
236 if (received_message.size() < 4 + 8)
237 continue;
238 nx_data idBytes(received_message.begin() + 4, received_message.begin() + 12);
239 auto user = ClientStorage::getClientById(Util::uint64FromFront(idBytes));
240 if (!user || !udp_auth::readSession(received_message, user->getUdpToken(), payload))
241 continue;
242 auto bound = m_clientEndpoints.find(user->getId());
243 if (bound != m_clientEndpoints.end() && bound->second != senderEndpoint)
244 continue;
245 // MessageHandler independently verifies MAC and replay IDs. Preserve
246 // the signed request; readSession returns the untagged payload.
247 payload.assign(received_message.begin() + 4, received_message.end());
248 }
249 else
250 {
251 continue;
252 }
253
254 auto handledMessage = getMessageHandler().readMessage(address, payload, &getSettings());
255
256 if (!handledMessage || !handledMessage->getUser())
257 {
258 Log::error(header(), "Received invalid message or null user");
259 continue;
260 }
261
262 auto* user = handledMessage->getUser();
263 touchDatagramTransport(*user, User::Transport::boostUdp);
264 if (!user->isBoostUDPSet())
265 {
266 m_clientEndpoints[user->getId()] = senderEndpoint;
267 std::weak_ptr<User> recipient = handledMessage->getUserHandle();
268 auto initialReply = std::make_shared<std::atomic<bool>>(
269 handledMessage->getType() == BaseMessage::Type::auth_message);
270 user->setBoostUDPSend([socket = m_socketState, senderEndpoint, recipient, initialReply](const nx_data& bytes)
271 {
272 auto client = recipient.lock();
273 if (!client)
274 return;
275 auto packet = initialReply->exchange(false)
276 ? bytes
277 : udp_auth::makeResponse(bytes, client->getUdpToken(),
278 client->nextUdpResponseId());
279 if (!packet.empty())
280 socket->send(senderEndpoint, packet); });
281 }
282
283 auto msg_type = handledMessage->getType();
284
285 if (msg_type == BaseMessage::Type::auth_message)
286 {
287 auto auth_ptr = static_cast<AuthMessage*>(handledMessage.get());
288 CommandResult passCommand = getCommand().read(
289 auth_ptr->getData()[0], *auth_ptr->getUser(), *this, auth_ptr->getMessageId());
290 if (passCommand != CommandResult::success)
291 {
292 Log::error(header(), "UDP auth failed");
293 }
294 }
295 else if (msg_type == BaseMessage::Type::message)
296 {
297 auto msg_ptr = static_cast<Message*>(handledMessage.get());
298
299 CommandResult passCommand = getCommand().read(msg_ptr->getData(), *msg_ptr->getUser(), *this, msg_ptr->getMessageId());
300 switch (passCommand)
301 {
302 case CommandResult::success:
303 Log::debug("UDPServer: Passed");
304 break;
305
306 default:
307 Log::error("UDPServer: Command failed with result: ", asString(passCommand));
308 }
309 }
310 else
311 {
312 Log::critical(header(), "Unrecognized message type");
313 }
314 }
315 catch (...)
316 {
317 // Keep serving later datagrams. Exception text may contain secrets;
318 // a user-installed logger must not throw out of this boundary either.
319 try
320 {
321 Log::error("UDP request processing failed");
322 }
323 catch (...)
324 {
325 }
326 }
327 }
328}

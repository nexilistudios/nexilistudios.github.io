34 : ServerProtocol(settings),

37 m_ioContext(std::make_unique<boost::asio::io_context>()),
38 m_mutex(std::make_unique<std::mutex>()),

99void UDPServer::start()
100{
101 auto local_endpoint = m_socket.local_endpoint();
102 m_serverPort = local_endpoint.port();
103 Log::debug("Boost UDP server started on port: ", m_serverPort);
104 if (!Util::writePortToFile(m_serverPort, getType()))
105 {
106 Log::error("Failed to write Boost UDP server port to a file");
107 }
108
109 m_ioContextThread = std::thread([this]()
110 { m_ioContext->run(); });
111
112 m_receiveThread = std::thread(&UDPServer::receiveFromClients, this);
113}

115void UDPServer::stop()
116{
117 if (!m_stopped || m_stopped->exchange(true))
118 {
119 Log::debug("BoostUDPServer stop already in progress or completed.");
120 return;
121 }
122
123 boost::system::error_code ec;
124
125 if (m_socket.cancel(ec))
126 {
127 Log::error("Error cancelling socket: ", ec.message());
128 }
129 if (ec)
130 {
131 Log::error("Cancelling error: ", ec.message());
132 }
133
134 if (m_ioContext)
135 {
136 Log::debug("BoostUDPServer stopping io_context");
137 m_ioContext->stop();
138 }
139
140 if (m_socket.is_open())
141 {
142 m_socket.shutdown(boost::asio::ip::udp::socket::shutdown_both, ec);
143 if (m_socket.close(ec))
144 {
145 Log::error("Error closing socket: ", ec.message());
146 }
147 if (ec)
148 {
149 Log::error("Socket closing error: ", ec.message());
150 }
151 }
152
153 if (m_ioContextThread.joinable())
154 {
155 Log::debug("BoostUDPServer closing ioContextThread");
156 m_ioContextThread.join();
157 }
158
159 if (m_receiveThread.joinable())
160 {
161 Log::debug("BoostUDPServer closing receiveThread");
162 m_receiveThread.join();
163 }
164 Log::debug("BoostUDPServer stopped");
165}

167void UDPServer::receiveFromClients()
168{
169 while (m_socket.is_open())
170 {
171 try
172 {
173 m_receiveBuffer.resize(NEXILIS_BUFFER);
174
175 size_t bytes_received = m_socket.receive_from(boost::asio::buffer(m_receiveBuffer), m_remoteEndpoint);
176
177 if (m_stopped->load())
178 {
179 break;
180 }
181
182 if (bytes_received == 0)
183 {
184 continue;
185 }
186
187 std::string address = m_remoteEndpoint.address().to_string();
188 // Create a new buffer containing only the received bytes.
189 nx_data received_message(m_receiveBuffer.begin(), m_receiveBuffer.begin() + bytes_received);
190
191 const auto senderEndpoint = m_remoteEndpoint;
192 const auto endpointKey = address + ":" + std::to_string(senderEndpoint.port());
193 nx_data payload;
194 if (udp_auth::hasPrefix(received_message, udp_auth::hello))
195 {
196 if (received_message.size() == udp_auth::makeHello().size())
197 {
198 auto challenge = udp_auth::makeChallenge(m_cookieKey, endpointKey);
199 if (!challenge.empty())
200 m_socket.send_to(boost::asio::buffer(challenge), senderEndpoint);
201 }
202 continue;
203 }
204 if (udp_auth::hasPrefix(received_message, udp_auth::authentication))
205 {
206 if (!udp_auth::checkAuthentication(received_message, m_cookieKey, endpointKey, payload))
207 continue;
208 if (payload.size() >= 8 && ClientStorage::getClientById(Util::uint64FromFront(payload)))
209 continue;
210 }
211 else if (udp_auth::hasPrefix(received_message, udp_auth::session))
212 {
213 if (received_message.size() < 4 + 8)
214 continue;
215 nx_data idBytes(received_message.begin() + 4, received_message.begin() + 12);
216 auto* user = ClientStorage::getClientById(Util::uint64FromFront(idBytes));
217 if (!user || !udp_auth::readSession(received_message, user->getUdpToken(), payload))
218 continue;
219 auto bound = m_clientEndpoints.find(user->getId());
220 if (bound != m_clientEndpoints.end() && bound->second != senderEndpoint)
221 continue;
222 }
223 else
224 {
225 continue;
226 }
227
228 auto handledMessage = getMessageHandler().readMessage(address, payload, &getSettings());
229
230 if (!handledMessage || !handledMessage->getUser())
231 {
232 Log::error(header(), "Received invalid message or null user");
233 continue;
234 }
235
236 if (!handledMessage->getUser()->isBoostUDPSet())
237 {
238 m_clientEndpoints[handledMessage->getUser()->getId()] = senderEndpoint;
239 // clang-format off
240 handledMessage->getUser()->setBoostUDPSend([this, senderEndpoint](const nx_data& bytes)
241 {
242 if (m_socket.send_to(boost::asio::buffer(bytes), senderEndpoint) == 0)
243 {
244 Log::error("Failed to send message to client");
245 }
246 });
247 // clang-format on
248 }
249
250 auto msg_type = handledMessage->getType();
251
252 if (msg_type == BaseMessage::Type::auth_message)
253 {
254 auto auth_ptr = static_cast<AuthMessage*>(handledMessage.get());
255 CommandResult passCommand = getCommand().read(
256 auth_ptr->getData()[0], *auth_ptr->getUser(), *this, auth_ptr->getMessageId());
257 if (passCommand != CommandResult::success)
258 {
259 Log::error(header(), "UDP auth failed");
260 }
261 }
262 else if (msg_type == BaseMessage::Type::message)
263 {
264 auto msg_ptr = static_cast<Message*>(handledMessage.get());
265
266 CommandResult passCommand = getCommand().read(msg_ptr->getData(), *msg_ptr->getUser(), *this, msg_ptr->getMessageId());
267 switch (passCommand)
268 {
269 case CommandResult::success:
270 Log::debug("UDPServer: Passed");
271 break;
272
273 default:
274 Log::error("UDPServer: Command failed with result: ", asString(passCommand));
275 }
276 }
277 else
278 {
279 Log::critical(header(), "Unrecognized message type");
280 }
281 }
282 catch (const boost::system::system_error& e)
283 {
284 Log::error("Error receiving message: ", e.what());
285 }
286 }
287
288 Log::error("Socket is not open!");
289}

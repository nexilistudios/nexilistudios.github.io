28bool ClientStorage::add(std::shared_ptr<User> client)
29{
30 if (!client)
31 return false;
32 std::lock_guard lock(m_mutex);
33 if (std::any_of(m_clients.begin(), m_clients.end(), [&](const auto& existing)
34 { return existing->getId() == client->getId(); }))
35 return false;
36 m_clients.emplace_back(std::move(client));
37 return true;
38}

70void ClientStorage::clear()
71{
72 std::vector<std::shared_ptr<User>> removed;

82bool ClientStorage::updateId(User& user, uint64_t id)
83{
84 if (!user.hasRootAccess())
85 return false;
86 std::lock_guard lock(m_mutex);
87 bool found = false;
88 for (const auto& client : m_clients)
89 {
90 if (client.get() == &user)
91 found = true;
92 else if (client->getId() == id)
93 return false;
94 }
95 if (!found)
96 return false;
97 user.setId(id);
98 return true;
99}

101bool ClientStorage::remove(uint64_t id)
102{
103 std::shared_ptr<User> removed;
104 {
105 std::lock_guard lock(m_mutex);
106 auto it = std::find_if(m_clients.begin(), m_clients.end(),
107 [id](const auto& client)
108 { return client->getId() == id; });
109 if (it == m_clients.end())
110 return false;
111 removed = std::move(*it);
112 m_clients.erase(it);
113 }
114 // User send callbacks may own resources whose destructors reenter storage.
115 // Release the last registry reference after unlocking.
116 return true;
117}

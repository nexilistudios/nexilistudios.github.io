28bool ClientStorage::add(std::shared_ptr<User> client, std::size_t maximum,
29 std::size_t maximumPerAddress)
30{
31 if (!client || maximum == 0 || maximumPerAddress == 0)
32 return false;
33 std::lock_guard lock(m_mutex);
34 if (m_clients.size() >= std::min(maximum, maximumClients))
35 return false;
36 std::size_t sessionsFromAddress = 0;
37 for (const auto& existing : m_clients)
38 {
39 if (existing->getId() == client->getId())
40 return false;
41 if (existing->getIPAddress() == client->getIPAddress())
42 ++sessionsFromAddress;
43 }
44 if (sessionsFromAddress >= maximumPerAddress)
45 return false;
46 m_clients.emplace_back(std::move(client));
47 return true;
48}

50bool ClientStorage::addWithAccountLimit(std::shared_ptr<User> client,
51 std::size_t maximumPerAccount, std::size_t maximum,
52 std::size_t maximumPerAddress)
53{
54 if (!client || client->getAccountId().empty() || maximumPerAccount == 0 ||
55 maximum == 0 || maximumPerAddress == 0)
56 return false;
57 std::lock_guard lock(m_mutex);
58 if (m_clients.size() >= std::min(maximum, maximumClients))
59 return false;
60 std::size_t sessions = 0;
61 std::size_t sessionsFromAddress = 0;
62 for (const auto& existing : m_clients)
63 {
64 if (existing->getId() == client->getId())
65 return false;
66 if (existing->getAccountId() == client->getAccountId())
67 ++sessions;
68 if (existing->getIPAddress() == client->getIPAddress())
69 ++sessionsFromAddress;
70 }
71 if (sessions >= maximumPerAccount || sessionsFromAddress >= maximumPerAddress)
72 return false;
73 m_clients.emplace_back(std::move(client));
74 return true;
75}

107void ClientStorage::clear()
108{

112 removed.swap(m_clients);

119bool ClientStorage::updateId(User& user, uint64_t id)
120{
121 if (!user.hasRootAccess())
122 return false;
123 std::lock_guard lock(m_mutex);
124 bool found = false;
125 for (const auto& client : m_clients)
126 {
127 if (client.get() == &user)
128 found = true;
129 else if (client->getId() == id)
130 return false;
131 }
132 if (!found)
133 return false;
134 user.setId(id);
135 return true;
136}

138bool ClientStorage::remove(uint64_t id)
139{
140 std::shared_ptr<User> removed;
141 {
142 std::lock_guard lock(m_mutex);
143 auto it = std::find_if(m_clients.begin(), m_clients.end(),
144 [id](const auto& client)
145 { return client->getId() == id; });
146 if (it == m_clients.end())
147 return false;
148 removed = std::move(*it);
149 m_clients.erase(it);
150 }
151 // User send callbacks may own resources whose destructors reenter storage.
152 // Release the last registry reference after unlocking.
153 return true;
154}

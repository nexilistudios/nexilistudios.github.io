36 ReadResult execute(ClientAPI& api, ClientAPI::ClientAPIData& data) override
37 {
38 auto& mtx = data.getRoomsMutex();
39 if (!mtx)
40 return ReadResult::command_execution;
41 std::lock_guard<std::mutex> lock(*mtx);
42
43 api.setClientId(m_client_id);
44 data.initialize();
45
46 assert(api.getClientId() == m_client_id);
47 assert(data.isInitialized());
48
49 auto p_data = _Packet::clientIdentification(api);
50 assert(p_data.size() >= 8);
51 assert(Util::convertToType<uint64_t>({p_data.begin(), p_data.begin() + 8}) == api.getClientId());
52
53 return ReadResult::success;
54 }
55
56private:
57 uint64_t m_client_id;

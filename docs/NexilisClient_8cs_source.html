166 public async Task<bool> ConnectAsync()
167 {
168 ThrowIfDisposed();
169
170 NativeInterop.InitializeMainThread();
171
172 Logger = new NxLogger("NexilisClient");
173 Logger.Setup(LogHandler ?? DefaultLogHandler);
174
175 ProtocolManager = new ProtocolManager();
176
177 var clientConfig = new ClientConfig();
178 clientConfig.SetBoostTCPAddress(ServerAddress);
179 clientConfig.SetBoostUDP(ServerAddress);
180 clientConfig.SetPassword(Password);
181 clientConfig.SetMode(AuthenticationMode);
182
183 ClientApi = new ClientAPI(clientConfig);
184 TcpClient = new BoostTCPClient(ProtocolManager, ClientApi);
185 UdpClient = new BoostUDPClient(ProtocolManager, ClientApi);
186
187 Logger.Info("Starting BoostTCP client...");
188 bool tcpStarted = await Task.Run(() => TcpClient!.StartClient(ClientApi!));
189 if (!tcpStarted)
190 {
191 throw new InvalidOperationException("BoostTCP client failed to start.");
192 }
193 Logger.Info("BoostTCP client started.");
194
195 Logger.Info("Starting BoostUDP client...");
196 await Task.Run(() => UdpClient!.Start());
197 if (!UdpClient!.IsConnected())
198 {
199 throw new InvalidOperationException("BoostUDP client failed to start.");
200 }
201 Logger.Info("BoostUDP client started.");
202
203 await WaitForRoomsCreatedAsync();
204
205 if (AutoJoinOnConnect)
206 {
207 await JoinAnyRoomAsync();
208 }
209
210 return IsReady;
211 }

213 async Task WaitForRoomsCreatedAsync()

215 using (var waiter = new Waiter(ClientApi!.ClientApiPtr))
216 {
217 await waiter.WaitUntilRoomsCreated();
218 }
219 Logger!.Info("Rooms verified and ready.");
220 }

286 public async Task<bool> JoinRoomAsync(ulong roomId)
287 {
288 if (roomId == 0)
289 {
290 throw new ArgumentException("Room id must not be zero.", nameof(roomId));
291 }
292
293 ThrowIfNotConnected();
294
295 if (ClientRoomId == roomId && IsReady)
296 {
297 Logger!.Info($"Already in room {roomId}.");
298 return true;
299 }
300
301 Logger!.Info($"Joining room {roomId}...");
302 using (var joinPacket = Packet.RoomManagementJoin(ClientApi!.ClientApiPtr, roomId))
303 {
304 SendPacket(joinPacket, PacketTransport.Tcp);
305 }
306
307 bool joined = await WaitUntil(() => ClientApi!.ClientRoomId() == roomId, RoomJoinTimeoutMs);
308 if (!joined)
309 {
310 Logger!.Warning($"Timed out joining room {roomId}.");
311 return false;
312 }
313
314 EnterRoom();
315 return true;
316 }

324 public async Task<bool> JoinAnyRoomAsync()
325 {
326 ThrowIfNotConnected();
327
328 var rooms = GetAvailableRooms();
329 if (rooms.Count == 0)
330 {
331 throw new InvalidOperationException("No active rooms available to join.");
332 }
333
334 ulong roomId = SelectRoomToJoin(rooms);
335 if (roomId == 0)
336 {
337 throw new InvalidOperationException("No active rooms found.");
338 }
339
340 return await JoinRoomAsync(roomId);
341 }
342
347 ulong SelectRoomToJoin(IReadOnlyList<RoomInfo> rooms)
348 {
349 if (PreferredRoomId.HasValue)
350 {
351 foreach (var room in rooms)
352 {
353 if (room.Id == PreferredRoomId.Value)
354 {
355 Logger!.Info($"Using preferred room {room.Id}.");
356 return room.Id;
357 }
358 }
359 Logger!.Warning($"Preferred room {PreferredRoomId.Value} is not available, picking another one.");
360 }
361
362 foreach (var room in rooms)
363 {
364 if (room.HasFreeSpace)
365 {
366 return room.Id;
367 }
368 }
369
370 // Everything is full, join the newest one and let the server decide.
371 return rooms[rooms.Count - 1].Id;
372 }

380 public async Task<bool> LeaveRoomAsync()
381 {
382 ThrowIfNotConnected();
383
384 if (ClientRoomId == 0)
385 {
386 return false;

388
389 ulong leavingRoomId = ClientRoomId;
390 Logger!.Info($"Leaving room {leavingRoomId}...");
391 using (var leavePacket = Packet.RoomManagementLeave(ClientApi!.ClientApiPtr))
392 {
393 SendPacket(leavePacket, PacketTransport.Tcp);
394 }
395
396 bool left = await WaitUntil(() => ClientApi!.ClientRoomId() == 0, RoomJoinTimeoutMs);
397 if (!left)
398 {
399 Logger.Warning($"Timed out leaving room {leavingRoomId}.");
400 return false;
401 }
402
403 ExitRoom();
404 Logger.Info($"Left room {leavingRoomId}.");
405 TryInvoke(OnLeftRoom);
406 return true;

420 public async Task<bool> CreateRoomAsync(string roomName, RoomContext context = RoomContext.ROOM_CONTEXT_3D)
421 {
422 if (string.IsNullOrWhiteSpace(roomName))
423 {
424 throw new ArgumentException("Room name must not be empty.", nameof(roomName));
425 }
426
427 ThrowIfNotConnected();
428
429 var clientApi = ClientApi!;
430 var knownRoomIds = new HashSet<ulong>();
431 foreach (var room in clientApi.GetActiveRoomInfos())
432 {
433 knownRoomIds.Add(room.Id);
434 }
435
436 Logger!.Info($"Creating room '{roomName}'...");
437 using (var createPacket = Packet.RoomManagementCreate(clientApi.ClientApiPtr, context, roomName))
438 {
439 SendPacket(createPacket, PacketTransport.Tcp);
440 }
441
442 ulong newRoomId = 0;
443 bool appeared = await WaitUntil(() => FindCreatedRoom(knownRoomIds, out newRoomId), RoomJoinTimeoutMs);
444 if (!appeared || newRoomId == 0)
445 {
446 Logger.Warning($"Timed out waiting for the created room '{roomName}'.");
447 return false;
448 }
449
450 Logger.Info($"Created room {newRoomId}, joining it...");
451 return await JoinRoomAsync(newRoomId).ConfigureAwait(false);
452 }

459 bool FindCreatedRoom(HashSet<ulong> knownRoomIds, out ulong roomId)
460 {
461 roomId = 0;
462 ulong myClientId = ClientApi!.GetClientId();
463 ulong fallbackId = 0;
464
465 foreach (var room in ClientApi!.GetActiveRoomInfos())
466 {
467 if (knownRoomIds.Contains(room.Id))
468 {
469 continue;
470 }
471
472 if (myClientId != 0 && room.CreatorId == myClientId)
473 {
474 roomId = room.Id;
475 return true;
476 }
477
478 if (fallbackId == 0)
479 {
480 fallbackId = room.Id;
481 }
482 }
483
484 if (fallbackId == 0)
485 {
486 return false;
487 }
488
489 Logger!.Info($"Room {fallbackId} appeared without a creator id, assuming it is the room we created.");
490 roomId = fallbackId;
491 return true;
492 }

498 async Task<bool> WaitUntil(Func<bool> condition, int timeoutMs)
499 {
500 int interval = Math.Max(1, RoomPollIntervalMs);
501 int waited = 0;
502
503 while (waited < timeoutMs)
504 {
505 if (condition())
506 {
507 return true;
508 }
509
510 await Task.Delay(interval).ConfigureAwait(false);
511 waited += interval;
512 }
513
514 return condition();
515 }

521 void EnterRoom()
522 {
523 ClientRoomId = ClientApi!.ClientRoomId();
524 ClientId = ClientApi.GetClientId();
525
526 // The previous room wrapper is a native handle of our own, drop it
527 // before replacing it. The room it wrapped is owned by the
528 // collection, not by us, so only the wrapper is released.
529 SafeDispose(ClientRoom);
530 ClientRoom = ClientApi.GetRoomFromId(ClientRoomId);
531 ClientRoomInfo = ClientRoom != null ? RoomInfo.FromRoom(ClientRoom) : null;
532
533 Logger!.Info($"Ready. ClientId={ClientId}, room {ClientRoomId} has {ClientRoom?.GetClientAmount() ?? 0} clients.");
534 IsReady = true;
535 TryInvoke(OnReady);
536 }

557 public void Update()
558 {
559 if (_disposed || !IsReady || ClientRoom == null) return;
560
561 var roomClients = ClientRoom.GetClients();
562 var present = new HashSet<ulong>();
563
564 foreach (var roomClient in roomClients)
565 {
566 var id = roomClient.GetId();
567 if (id == 0 || id == ClientId) continue;
568
569 present.Add(id);
570 if (_remoteClients.Add(id))
571 {
572 Logger!.Info($"Remote client added: {id}");
573 TryInvoke(OnRemoteClientAdded, id);
574 }
575 }
576
577 foreach (var id in _remoteClients.Where(id => !present.Contains(id)).ToList())
578 {
579 _remoteClients.Remove(id);
580 Logger!.Info($"Remote client removed: {id}");
581 TryInvoke(OnRemoteClientRemoved, id);
582 }
583
584 foreach (var roomClient in roomClients)
585 {
586 var id = roomClient.GetId();
587 if (id == 0 || id == ClientId || !_remoteClients.Contains(id)) continue;
588
589 if (TryGetPosition(roomClient, out var position))
590 {
591 TryInvoke(OnRemoteClientPosition, id, position);
592 }
593 }
594 }

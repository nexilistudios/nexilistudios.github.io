85 template <typename T, typename... Args>
86 T createProtocol(Args&&... args)
87 {
88 static_assert(detail::has_get_type<T>::value,
89 "Type must derive from nexilis::Protocol or have a "
90 "getType() method returning Protocol::Type");
91
92 // This is little hacky. I'd much rather prefer if this was std::move call.
93 // However this would require that the parameters cannot be references.
94 // So this is technically always move call and it works and the api is nice.
95 //
96 // It's also important to note that this gives compile-time error
97 // if the construction fails.
98 auto protocol = T(std::forward<Args>(args)...);
99 m_items.emplace_back(ProtocolData(protocol.getType()));
100
101 return protocol;
102 }
103

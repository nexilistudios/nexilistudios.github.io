35struct StreamServer::Connection
36{
37 explicit Connection(int socket) : fd(socket)
38 {
39 }
40 ~Connection()
41 {
42 close();
43 }
44
45 void shutdown()
46 {
47 std::lock_guard lock(mutex);
48 open = false;
49 if (fd != -1)
50 ::shutdown(fd, SHUT_RDWR);
51 }
52 void close()
53 {
54 std::lock_guard lock(mutex);
55 open = false;
56 if (fd != -1)
57 {
58 ::shutdown(fd, SHUT_RDWR);
59 ::close(fd);
60 fd = -1;
61 }
62 }
63 void send(const nx_data& bytes)
64 {
65 std::lock_guard lock(mutex);
66 if (!open)
67 throw std::runtime_error("Unix stream connection is closed");
68 std::size_t offset = 0;
69 while (offset < bytes.size())
70 {
71 const auto sent = ::send(fd, bytes.data() + offset, bytes.size() - offset, MSG_NOSIGNAL | MSG_DONTWAIT);
72 if (sent < 0 && errno == EINTR)
73 continue;
74 if (sent <= 0)
75 {
76 open = false;
77 ::shutdown(fd, SHUT_RDWR);
78 throw std::runtime_error("Unix stream send failed");
79 }
80 offset += static_cast<std::size_t>(sent);
81 }
82 }
83 // Only the receiving worker closes fd. stop() only shuts it down to wake
84 // recv; stale send callbacks retain this state and cannot use a recycled fd.
85 int fd;
86 bool open = true;
87 std::mutex mutex;
88};

155void StreamServer::start()
156{
157 if (!m_stopped || m_stopped->load() || m_receiveThread.joinable() || m_serverSocket == -1)
158 return;
159 m_receiveThread = std::thread([this]
160 {
161 while (!m_stopped->load())

163 try { handleMessages(); }

168void StreamServer::stop()
169{
170 if (!m_stopped)
171 return;
172 m_stopped->store(true);
173 {
174 std::lock_guard lock(*m_mutex);
175 if (m_serverSocket != -1)
176 ::shutdown(m_serverSocket, SHUT_RDWR);
177 if (m_client)
178 m_client->shutdown();
179 }
180 if (m_receiveThread.joinable())
181 {
182 if (m_receiveThread.get_id() == std::this_thread::get_id())
183 return;
184 m_receiveThread.join();
185 }
186 if (m_serverSocket != -1)
187 {
188 ::close(m_serverSocket);
189 m_serverSocket = -1;
190 }
191}

208nx_data StreamServer::receiveMessage(int socket)
209{
210 auto readExact = [socket](uint8_t* data, size_t length)
211 {
212 size_t offset = 0;
213 while (offset < length)
214 {
215 ssize_t count = recv(socket, data + offset, length - offset, 0);
216 if (count <= 0)
217 return false;
218 offset += static_cast<size_t>(count);
219 }
220 return true;
221 };
222
223 uint8_t header[4]{};
224 if (!readExact(header, sizeof(header)))
225 return {};
226 uint32_t length = (static_cast<uint32_t>(header[0]) << 24) |
227 (static_cast<uint32_t>(header[1]) << 16) |
228 (static_cast<uint32_t>(header[2]) << 8) | header[3];
229 if (length == 0 || length > NEXILIS_BUFFER)
230 return {};
231 nx_data message(length);
232 if (!readExact(message.data(), message.size()))
233 return {};
234 return message;
235}

237void StreamServer::handleMessages()
238{
239 const int clientSocket = accept(m_serverSocket, nullptr, nullptr);
240 if (clientSocket == -1)
241 return;
242 if (!unix_socket::sameUser(clientSocket))
243 {
244 ::close(clientSocket);
245 return;
246 }
247 std::shared_ptr<Connection> connection;
248 try
249 {
250 connection = std::make_shared<Connection>(clientSocket);
251 }
252 catch (...)
253 {
254 ::close(clientSocket);
255 throw;
256 }
257 {
258 std::lock_guard lock(*m_mutex);
259 if (m_stopped->load())
260 return;
261 m_client = connection;
262 }

264 struct CloseConnection
265 {
266 std::shared_ptr<Connection> connection;
267 ~CloseConnection()
268 {
269 connection->close();
270 }
271 } closeOnExit{connection};
272 while (!m_stopped->load())
273 {
274 nx_data message = receiveMessage(clientSocket);
275
276 if (message.empty())
277 {
278 break;
279 }
280
281 auto handledMessage = getMessageHandler().readMessage("127.0.0.1", message, &getSettings());
282
283 if (!handledMessage || !handledMessage->getUser())
284 {
285 Log::error("Message from unauthorized client!");
286 break;
287 }
288
289 auto* user = handledMessage->getUser();
290 if (!user->isUnixStreamSet())
291 {
292 user->setUnixStreamSend([connection](const nx_data& bytes)
293 { connection->send(bytes); });
294 }
295
296 auto type = handledMessage->getType();
297
298 if (type == BaseMessage::Type::auth_message)
299 {
300 // Reply to the handshake so the client receives its client id.
301 auto msgPtr = static_cast<AuthMessage*>(handledMessage.get());
302 for (auto&& commandData : msgPtr->getData())
303 {
304 CommandResult result =
305 getCommand().read(commandData, *user, *this, msgPtr->getMessageId());
306 checkResult(result);
307 }
308 }
309 else if (type == BaseMessage::Type::message)
310 {
311 auto msgPtr = static_cast<Message*>(handledMessage.get());
312 CommandResult result =
313 getCommand().read(msgPtr->getData(), *user, *this, msgPtr->getMessageId());
314
315 checkResult(result);
316
317 if (result == CommandResult::success)
318 {
319 Log::debug("StreamServer: Passed");
320 }
321 }
322 }
323 connection->close();
324 {
325 std::lock_guard lock(*m_mutex);
326 m_client.reset();
327 }
328}

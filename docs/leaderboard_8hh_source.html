39class RoomLeaderboardCommand : public BaseAPICommand
40{
41public:
42 explicit RoomLeaderboardCommand(const boost::json::array& entries)
43 {
44 for (const auto& item : entries)
45 {
46 if (!item.is_object())
47 continue;
48
49 const auto& obj = item.as_object();
50 ClientAPI::LeaderboardEntry entry;
51
52 if (obj.contains("id"))
53 entry.id = Util::toUint64(obj.at("id"));
54 if (obj.contains("username"))
55 entry.username = obj.at("username").as_string().c_str();
56 if (obj.contains("team"))
57 entry.team = obj.at("team").as_string().c_str();
58 if (obj.contains("kills"))
59 entry.kills = Util::toUint64(obj.at("kills"));
60 if (obj.contains("deaths"))
61 entry.deaths = Util::toUint64(obj.at("deaths"));
62
63 m_entries.emplace_back(std::move(entry));
64 }
65 }
66
67 ReadResult execute(ClientAPI&, ClientAPI::ClientAPIData& data) override
68 {
69 ClientAPI::LeaderboardEvent event;
70 event.entries = std::move(m_entries);
71 data.pushLeaderboardEvent(std::move(event));
72 return ReadResult::success;
73 }
74
75private:
76 std::vector<ClientAPI::LeaderboardEntry> m_entries;
77};

26CommandResult ServerImpl::room_management_join(const DefaultArgs& args)
27{
28 Log::debug("Commands::Room::Management::join", args);
29 auto user_id = args.getUser().getId();
30
31 auto payload = Util::removeAmountOfBytesFromVector(args.getData(), 3);
32 uint64_t roomId = Util::convertToType<uint64_t>(payload);
33 auto* room = RoomStorage::getRoomById(roomId);
34
35 if (!room)
36 {
37 Log::error("Cannot find room with specified id!");
38 return CommandResult::invalid_input;
39 }
40
41 if (RoomStorage::getRoomById(roomId)->contains(user_id))
42 {
43 Log::error("Cannot join room where the client already is!");
44 return CommandResult::error;
45 }
46
47 room->joinRoom(user_id);
48
49 args.getUser().setRoomId(room->getId());
50 assert(ClientStorage::getClientById(user_id)->getRoomId() == room->getId());
51 assert(RoomStorage::getRoomById(roomId)->contains(user_id));
52 assert(args.getUser().getRoomId() == roomId);
53
54 auto roomCommand = Command::createRoomCommand(roomId, args.getUser(), args.getData(),
55 {{"username", boost::json::value(args.getUser().getUsername())}}, args.getMessageId());
56 Command::sendRoomCommand(roomCommand, args.getUser(), args.getProtocol());
57
58 // Send existing players to the joining client
59 auto& user = args.getUser();
60 auto& protocol = args.getProtocol();
61 for (auto& existingClientId : room->getClients())
62 {
63 if (existingClientId == user_id)
64 continue;
65 auto* existingUser = ClientStorage::getClientById(existingClientId);
66 if (!existingUser)
67 continue;
68
69 // Send "join" to create a ClientSession on the joining client
70 std::map<std::string, boost::json::value> joinParams{{"username", boost::json::value(existingUser->getUsername())}};
71 auto joinMsg = Command::createRoomCommand(roomId, *existingUser, args.getData(), joinParams, args.getMessageId());
72 Command::sendMessageToClient(joinMsg, user, protocol);
73
74 // Send current position
75 auto pos = existingUser->getObject3D().getPosition();
76 std::map<std::string, boost::json::value> posParams{
77 {"action", boost::json::value("position")},
78 {"room_id", boost::json::value(roomId)},
79 {"client_id", boost::json::value(existingClientId)},
80 {"x", boost::json::value(pos.x)},
81 {"y", boost::json::value(pos.y)},
82 {"z", boost::json::value(pos.z)},
83 {"yaw", boost::json::value(existingUser->getObject3D().getYaw())}};
84 auto posMsg = Command::clientMessageData(CommandType::room, "player_3D", 0, posParams);
85 Command::sendMessageToClient(posMsg, user, protocol);
86
87 // Send current dimensions
88 auto dim = existingUser->getObject3D().getDimensions();
89 std::map<std::string, boost::json::value> dimParams{
90 {"action", boost::json::value("dimensions")},
91 {"room_id", boost::json::value(roomId)},
92 {"client_id", boost::json::value(existingClientId)},
93 {"x", boost::json::value(dim.x)},
94 {"y", boost::json::value(dim.y)},
95 {"z", boost::json::value(dim.z)}};
96 auto dimMsg = Command::clientMessageData(CommandType::room, "player_3D", 0, dimParams);
97 Command::sendMessageToClient(dimMsg, user, protocol);
98 }
99
100 // Seed the new client's team roster before they choose a side.
101 auto roster = Command::createRoomLeaderboardCommand(
102 roomId, user, Command::playerStatsEntries(*room, room->getClients()), 0);
103 Command::sendMessageToClient(roster, user, protocol);
104
105 // Send existing game items to the joining client
106 for (auto& item : room->getGameItems())
107 {
108 auto pos = item.getPosition();
109 auto dim = item.getDimensions();
110 std::map<std::string, boost::json::value> itemParams{
111 {"action", boost::json::value("create")},
112 {"room_id", boost::json::value(roomId)},
113 {"client_id", boost::json::value(user_id)},
114 {"id", boost::json::value(item.getId())},
115 {"x", boost::json::value(pos.x)},
116 {"y", boost::json::value(pos.y)},
117 {"z", boost::json::value(pos.z)},
118 {"w", boost::json::value(dim.x)},
119 {"h", boost::json::value(dim.y)},
120 {"d", boost::json::value(dim.z)},
121 {"item_type", boost::json::value(item.getType())},
122 {"status", boost::json::value(item.getStatus())},
123 {"filepath", boost::json::value(item.getFilepath())}};
124 auto itemMsg = Command::clientMessageData(CommandType::room, "game_item", 0, itemParams);
125 Command::sendMessageToClient(itemMsg, user, protocol);
126 }
127
128 // Send the room's broadcast history to the joining client
129 for (const auto& broadcast : room->getBroadcasts())
130 {
131 std::map<std::string, boost::json::value> broadcastParams{
132 {"action", boost::json::value("broadcast")},
133 {"room_id", boost::json::value(roomId)},
134 {"client_id", boost::json::value(broadcast.clientId)},
135 {"message", boost::json::value(broadcast.message)}};
136 auto broadcastMsg = Command::clientMessageData(CommandType::room, "communication", 0, broadcastParams);
137 Command::sendMessageToClient(broadcastMsg, user, protocol);
138 }
139
140 return CommandResult::success;
141}

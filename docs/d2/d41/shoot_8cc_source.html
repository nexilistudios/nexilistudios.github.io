33CommandResult ServerImpl::room_player3d_shoot(const DefaultArgs& args)
34{
35 auto& user = args.getUser();
36 if (user.getRoomId() == 0)
37 {
38 Log::error("Shoot: User not in room!");
39 return CommandResult::error;
40 }
41
42 auto payload = Util::removeAmountOfBytesFromVector(args.getData(), 3);
43 auto room = RoomStorage::getRoomById(user.getRoomId());
44 if (!room)
45 {
46 Log::error("Shoot: Room not found");
47 return CommandResult::failure;
48 }
49
50 uint64_t targetId = 0;
51 float damage = 0.0f;
52 // A client cannot decide whom it hit or how much damage it dealt.
53 if (!room->hasShotHandler() || payload.size() != 14)
54 return CommandResult::invalid_input;
55 const auto direction = Util::vector3fFromFront(payload);
56 if (!std::isfinite(direction.x) || !std::isfinite(direction.y) ||
57 !std::isfinite(direction.z))
58 return CommandResult::invalid_input;
59
60 const auto& limits = GameplayLimits::get();
61 // Reserve the cooldown before invoking the application callback. A miss
62 // still consumes a shot; reload requests do not.
63 const auto interval = limits.shotInterval();
64 if (payload[13] == 0 && interval.count() > 0.0f &&
65 !user.acceptShot(std::chrono::steady_clock::now(), interval))
66 {
67 Log::warning("Shoot: player ", user.getId(),
68 " exceeded the shooting rate limit of ",
69 limits.getMaxShootingRate(), " shots per second");
70 return CommandResult::invalid_input;
71 }
72 const auto hit = room->resolveShot(user.getId(), direction, payload[12], payload[13]);
73 if (!hit)
74 return CommandResult::success;
75 targetId = hit->targetId;
76 damage = hit->damage;
77
78 if (targetId == user.getId() || !std::isfinite(damage) || damage <= 0.0f)
79 return CommandResult::invalid_input;
80 damage = limits.clampDamage(damage);
81
82 bool targetFound = std::any_of(room->getClients().begin(), room->getClients().end(),
83 [targetId](auto clientId)
84 { return clientId == targetId; });
85 if (!targetFound)
86 {
87 Log::warning("Shoot: Target ", targetId, " not in room");
88 return CommandResult::failure;
89 }
90
91 const auto shooterTeam = room->getPlayerTeam(user.getId());
92 const auto targetTeam = room->getPlayerTeam(targetId);
93 if (shooterTeam.empty() || targetTeam.empty() || shooterTeam == targetTeam)
94 return CommandResult::invalid_input;
95 if (!room->canDamage(user.getId(), targetId))
96 return CommandResult::invalid_input;
97 if (room->getPlayerHealth(user.getId()) <= 0.0f)
98 return CommandResult::invalid_input;
99
100 // Only triggers a death when this hit actually brought the target from
101 // alive to dead. Shots that land on an already-dead (not yet respawned)
102 // player are no-ops, so a kill/death is never recorded twice.
103 bool causedDeath = room->damagePlayer(targetId, damage);
104 float newHealth = room->getPlayerHealth(targetId);
105
106 Log::info("Player ", user.getId(), " shot player ", targetId,
107 " for ", damage, " damage. HP: ", newHealth);
108
109 std::map<std::string, boost::json::value> params{
110 {"target_id", boost::json::value(targetId)},
111 {"damage", boost::json::value(static_cast<double>(damage))},
112 {"new_health", boost::json::value(static_cast<double>(newHealth))}};
113
114 if (Command::sendRoomCommand(
115 Command::createRoomCommand(
116 user.getRoomId(), user, args.getData(), params, args.getMessageId()),
117 user, args.getProtocol()) == false)
118 return CommandResult::failed_room_send;
119
120 if (causedDeath)
121 {
122 room->onPlayerDied(user.getId(), targetId);
123 }
124
125 return CommandResult::success;
126}

87CommandResult Command::read(const nx_data& command, User& user, NXProtocol& protocol, uint64_t messageId)
88{
89 assert(user.getId() != 0);
90
91 if (command.size() < 2)
92 {
93 return CommandResult::invalid_input;
94 }
95
96 // Missing optional command levels cannot match a valid action.
97 std::array<uint8_t, 4> commandBytes;
98 commandBytes.fill(255);
99 std::copy_n(command.begin(), std::min(command.size(), commandBytes.size()), commandBytes.begin());
100
101 auto arg = static_cast<CommandType>(commandBytes[0]);
102 auto arg2 = commandBytes[1];
103 auto arg3 = commandBytes[2];
104
105 // Authentication payloads contain credentials, so do not dump their bytes.
106 if (arg == CommandType::authentication)
107 {
108 Log::debug("Command: authentication payload redacted");
109 }
110 else
111 {

113 Util::debugUint8Vector(command);
114 }
115
116 // DefaultArgs from Commands.
117 DefaultArgs args(user, protocol, command, messageId);
118
119 // Keep the movement system and the gameplay limits bound to the
120 // live server settings: the static pointers must be refreshed on
121 // every dispatch so a fresh Command (e.g. a recreated server or a
122 // test fixture) does not leave detached movement threads reading a
123 // destroyed configuration's tickrate or command handlers reading
124 // destroyed limits.
125 Movement::_initialize(m_settings);
126 GameplayLimits::_initialize(m_settings.getGameplayLimits());

128 Log::debug(header(), commandTypeAsString(arg));
129 switch (arg)

478 case CommandType::authentication:
479 {
480 switch (arg2)
481 {
482 // Setup authentication.
483 case 0:
484 {
485 Log::info("New client wants to authenticate, not implemented");
486 return CommandResult::not_found;
487 }
488
489 // Root access is available only on an authenticated TLS-PSK
490 // Boost TCP session. Failed attempts are shared by source
491 // address across reconnects to this server instance.
492 case 1:
493 {
494 if (m_settings.getMode() != AuthenticationMode::password_protected ||
495 !m_settings.isTlsEnabled() ||
496 protocol.getType() != NXProtocol::Type::BOOST_TCP_SERVER ||
497 !user.hasCommonAccess() || !m_settings.hasRootPassword())
498 {
499 return CommandResult::unauthorized;
500 }
501
502 if (user.hasRootAccess())
503 {
504 return CommandResult::success;
505 }
506
507 const auto now = std::chrono::steady_clock::now();
508 std::lock_guard lock(m_rootAuthState->mutex);
509 auto& attempts = m_rootAuthState->byAddress;
510 auto it = attempts.find(user.getIPAddress());
511 if (it != attempts.end() && now - it->second.firstFailure >= kRootAuthWindow)
512 {
513 it = attempts.erase(it);
514 }
515 if (it != attempts.end() && it->second.failures >= kMaxRootAuthFailures)
516 {
517 return CommandResult::unauthorized;
518 }
519 if (it == attempts.end() && attempts.size() >= kMaxTrackedRootAuthAddresses)
520 {
521 std::erase_if(attempts, [now](const auto& entry)
522 { return now - entry.second.firstFailure >= kRootAuthWindow; });
523 if (attempts.size() >= kMaxTrackedRootAuthAddresses)
524 {
525 return CommandResult::unauthorized;
526 }
527 }
528
529 auto payload = Util::removeAmountOfBytesFromVector(command, 2);
530 const std::string password = Util::convertToString(payload);
531 if (m_settings.isRootPassword(password))
532 {
533 if (it != attempts.end())
534 {
535 attempts.erase(it);
536 }
537 user.setRootAccess(true);
538 Log::info("Client ", user.getIPAddress(), " has root access!");
539 return CommandResult::success;
540 }
541
542 if (it != attempts.end())
543 {
544 ++it->second.failures;
545 }
546 else
547 {
548 attempts.emplace(user.getIPAddress(), RootAuthState::Attempt{now, 1});
549 }
550 return CommandResult::unauthorized;
551 }
552
553 // Passphrase authentication belongs to the initial handshake,
554 // not a command on an established session.
555 case 2:
556 {
557 return CommandResult::unauthorized;
558 }
559 }
560 return CommandResult::not_found;
561 }
562
563 case CommandType::server_management:
564 {
565 return CommandResult::not_found;
566 }
567
568 case CommandType::player_management:
569 {
570 return CommandResult::not_found;
571 }

581 Log::critical("Internal server error: x");
582 return CommandResult::error;
583 }
584 }
585 return CommandResult::not_found;
586 }
587
588 case CommandType::undefined:
589 {
590 return CommandResult::error;
591 }
592 }
593 return CommandResult::not_found;

596bool Command::sendMessageToClient(nx_data data, User& user, NXProtocol& protocol)
597{
598 auto trySend = [&data, &user](NXProtocol::Type type) -> bool
599 {
600 switch (type)
601 {
602 case NXProtocol::Type::BOOST_TCP_SERVER:
603 return user.boostTCPSend(data);
604 case NXProtocol::Type::BOOST_UDP_SERVER:
605 return user.boostUDPSend(data);
606 case NXProtocol::Type::AF_UNIX_SOCK_STREAM_SERVER:
607 return user.unixStreamSend(data);
608 case NXProtocol::Type::AF_UNIX_SOCK_DGRAM_SERVER:
609 return user.unixDgramSend(data);
610 case NXProtocol::Type::AF_INET_TCP_SERVER:
611 return user.inetTCPSend(data);
612 case NXProtocol::Type::AF_INET_UDP_SERVER:
613 return user.inetUDPSend(data);
614 default:
615 return false;
616 }
617 };
618
619 if (trySend(protocol.getType()))
620 {
621 return true;
622 }
623
624 static const NXProtocol::Type allServerProtocols[] = {
625 NXProtocol::Type::BOOST_TCP_SERVER,
626 NXProtocol::Type::BOOST_UDP_SERVER,
627 NXProtocol::Type::AF_UNIX_SOCK_STREAM_SERVER,
628 NXProtocol::Type::AF_UNIX_SOCK_DGRAM_SERVER,
629 NXProtocol::Type::AF_INET_TCP_SERVER,
630 NXProtocol::Type::AF_INET_UDP_SERVER,
631 };
632
633 for (auto fallbackType : allServerProtocols)
634 {
635 if (fallbackType == protocol.getType())
636 continue;
637 if (trySend(fallbackType))
638 return true;
639 }
640
641 Log::error("Failed to send message to client via any protocol");
642 return false;
643}

678nx_data Command::createRoomCommand(uint64_t roomId, User& user, const nx_data& messageData, const std::map<std::string, boost::json::value>& params, uint64_t messageId)
679{
680 if (messageData.size() < 3)
681 {
682 Log::error("createRoomCommand: Insufficient messageData");
683 return nx_data();
684 }
685
686 auto room_type = static_cast<RoomCommandType::Root>(messageData[1]);
687 auto action = messageData[2];
688
689 std::string room_command_action = roomActionToString(room_type, action);
690 if (room_command_action.empty())
691 {
692 Log::error("createRoomCommand: Failed to convert room action to string");
693 return nx_data();
694 }
695
696 std::string room_command_type = RoomCommandType::RoomTypeToString(room_type);
697 if (room_command_type.empty())
698 {
699 Log::error("createRoomCommand: Failed to convert roomType");
700 return nx_data();
701 }
702
703 auto all_params = ClientMsgType{
704 {"action", boost::json::value(room_command_action)},
705 {"room_id", boost::json::value(roomId)},
706 {"client_id", boost::json::value(user.getId())},
707 };
708 all_params.insert(params.begin(), params.end());
709
710 return clientMessageData(CommandType::room, room_command_type, messageId, all_params);
711}

767void Command::runWithTickrate(double tickrate, double durationSeconds, const std::function<void(double)>& tickFunction)
768{
769 using namespace std::chrono;
770
771 auto interval = duration_cast<milliseconds>(milliseconds(static_cast<int>(1000 / tickrate)));
772 auto startTime = steady_clock::now();
773 auto endTime = startTime + duration_cast<milliseconds>(milliseconds(static_cast<int>(durationSeconds * 1000)));
774
775 double totalTicks = tickrate * durationSeconds;
776
777 for (int tick = 1; steady_clock::now() <= endTime && tick <= totalTicks; ++tick)
778 {
779 auto loopStart = steady_clock::now();
780 double progress = static_cast<double>(tick) / totalTicks;
781 tickFunction(progress);
782 std::this_thread::sleep_until(loopStart + interval);
783 }
784}

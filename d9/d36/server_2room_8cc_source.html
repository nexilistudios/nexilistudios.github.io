82bool Room::contains(uint64_t userId)
83{
84 // Use std::any_of algorithm to check if the user is in the room.
85 return std::any_of(m_clientIds.begin(), m_clientIds.end(),
86 [&userId](uint64_t id)
87 { return userId == id; });
88}

90void Room::joinRoom(uint64_t userId)
91{
92 if (contains(userId))
93 {
94 Log::warning("User already in this room!");
95 }
96 else
97 {

103void Room::leaveRoom(uint64_t userId)
104{
105 m_clientIds.erase(std::remove_if(m_clientIds.begin(), m_clientIds.end(),
106 [&userId](uint64_t id)
107 { return userId == id; }),
108 m_clientIds.end());
109 std::lock_guard<std::mutex> lock(m_stateMutex);
110 m_playerHealth.erase(userId);
111 m_playerKills.erase(userId);
112 m_playerDeaths.erase(userId);
113 m_playerTeams.erase(userId);
114}

130bool Room::damagePlayer(uint64_t clientId, float damage)
131{
132 std::lock_guard<std::mutex> lock(m_stateMutex);
133
134 auto it = m_playerHealth.find(clientId);
135 float health;
136 if (it != m_playerHealth.end())
137 {
138 // Ignore damage to players who are already dead. This is what prevents
139 // a single kill from being recorded twice: near-simultaneous shots from
140 // multiple clients all see health <= 0 and only the first one reports
141 // a death.
142 if (it->second <= 0.0f)
143 return false;
144 health = std::max(0.0f, it->second - damage);
145 it->second = health;
146 }
147 else
148 {
149 health = std::max(0.0f, m_defaultHealth - damage);
150 m_playerHealth[clientId] = health;
151 }
152
153 return health <= 0.0f;
154}

311bool Room::broadcastToAll(const nx_data& data)
312{
313 bool all_success = true;
314 for (auto clientId : m_clientIds)
315 {
316 auto* client = ClientStorage::getClientById(clientId);
317 if (!client)
318 continue;
319
320 if (client->isBoostTCPSet())
321 {
322 if (!client->boostTCPSend(data))
323 all_success = false;
324 }
325 else if (client->isBoostUDPSet())
326 {
327 if (!client->boostUDPSend(data))
328 all_success = false;
329 }
330 else if (client->isUnixStreamSet())
331 {
332 if (!client->unixStreamSend(data))
333 all_success = false;
334 }
335 else if (client->isInetTCPSet())
336 {
337 if (!client->inetTCPSend(data))
338 all_success = false;
339 }
340 else if (client->isInetUDPSet())
341 {
342 if (!client->inetUDPSend(data))
343 all_success = false;
344 }
345 else
346 {
347 all_success = false;
348 }
349 }
350 return all_success;
351}

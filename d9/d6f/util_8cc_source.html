46bool Util::constantTimeEquals(const std::string& expected, const std::string& given)
47{
48 uint8_t result = static_cast<uint8_t>(expected.size() != given.size());
49
50 for (size_t i = 0; i < expected.size(); ++i)
51 {
52 uint8_t expectedByte = static_cast<uint8_t>(expected[i]);
53 uint8_t givenByte = i < given.size() ? static_cast<uint8_t>(given[i]) : 0;
54 result |= static_cast<uint8_t>(expectedByte ^ givenByte);
55 }
56
57 return result == 0;
58}

85uint16_t Util::convertoToUint16(const nx_data& bytes)
86{
87 if (bytes.size() < 2)
88 {
89 throw std::invalid_argument("Input nx_data must contain at least 2 bytes");
90 }
91
92 if (server::Config::getBigEndian())
93 {
94 return static_cast<uint16_t>((bytes[0] << 8) | bytes[1]);
95 }
96 else
97 {
98 return static_cast<uint16_t>(bytes[0] | static_cast<uint16_t>(bytes[1]) << 8);
99 }

102// Big-endian float conversion helper.
103float bytesToFloatBigEndian(const uint8_t* bytes)
104{
105 uint32_t temp = 0;
106 temp |= static_cast<uint32_t>(bytes[0]) << 24;
107 temp |= static_cast<uint32_t>(bytes[1]) << 16;
108 temp |= static_cast<uint32_t>(bytes[2]) << 8;
109 temp |= static_cast<uint32_t>(bytes[3]);
110
111 float result;
112 memcpy(&result, &temp, sizeof(float));
113 return result;
114}

116Vector2f Util::convertToVector2(const nx_data& bytes)
117{
118 // Ensure the vector has enough bytes for two floats
119 if (bytes.size() < sizeof(float) * 2)
120 {
121 throw std::invalid_argument("The input vector does not contain enough bytes for two floats.");
122 }
123 float x, y;
124
125 if (server::Config::getBigEndian())
126 {
127 x = bytesToFloatBigEndian(bytes.data());
128 y = bytesToFloatBigEndian(bytes.data() + sizeof(float));
129 }
130 else
131 {
132 // Copy the first 4 bytes.
133 memcpy(&x, bytes.data(), sizeof(float));
134 // Copy the next 4 bytes.
135 memcpy(&y, bytes.data() + sizeof(float), sizeof(float));
136 }
137 return Vector2f(x, y);
138}

140Vector3f Util::convertToVector3(const nx_data& bytes)
141{
142 if (bytes.size() < sizeof(float) * 3)
143 {
144 throw std::invalid_argument("The input vector does not contain enough bytes for three floats");
145 }
146 float x, y, z;
147
148 if (server::Config::getBigEndian())
149 {
150 x = bytesToFloatBigEndian(bytes.data());
151 y = bytesToFloatBigEndian(bytes.data() + sizeof(float));
152 z = bytesToFloatBigEndian(bytes.data() + 2 * sizeof(float));
153 }
154 else
155 {
156 memcpy(&x, bytes.data(), sizeof(float));
157 memcpy(&y, bytes.data() + sizeof(float), sizeof(float));
158 memcpy(&z, bytes.data() + sizeof(float) * 2, sizeof(float));
159 }
160 return Vector3f(x, y, z);
161}

319uint64_t Util::getRandomUint64()
320{
321 std::mt19937_64 generator(rand_dev());
322 std::uniform_int_distribution<uint64_t> dist(0, NEXILIS_MAX);
323
324 uint64_t randomValue;
325 do
326 {
327 randomValue = dist(generator);
328 } while (std::bitset<64>(randomValue).count() < 32); // Ensure at least 32 bits are set
329
330 assert((std::is_same<decltype(randomValue), uint64_t>::value));
331
332 if (randomValue == 0)
333 return getRandomUint64();
334
335 return randomValue;
336}

412uint64_t Util::uint64FromFront(const nx_data& vec)
413{
414 if (vec.size() < 8)
415 {
416 // Not enough data to read 8 bytes
417 Log::error("Not enough data to read 8 bytes");
418 return 0;
419 }
420
421 uint64_t value = 0;
422
423 if (server::Config::getBigEndian())
424 {
425 // Big-endian: Most significant byte is at the lowest address
426 for (int i = 0; i < 8; ++i)
427 {
428 value |= static_cast<uint64_t>(vec[i]) << ((7 - i) * 8);
429 }
430 }
431 else
432 {
433 // Little-endian: Least significant byte is at the lowest address
434 for (int i = 0; i < 8; ++i)
435 {
436 value |= static_cast<uint64_t>(vec[i]) << (i * 8);
437 }
438 }
439 return value;
440}

568std::optional<uint16_t> Util::readPortFromFile(NXProtocol::Type protocol_type)
569{
570 std::string file_path = getPortFilePath(protocol_type);
571
572 try
573 {
574 if (!std::filesystem::exists(file_path))
575 {
576 return std::nullopt;
577 }
578
579 std::ifstream port_file(file_path);
580 if (!port_file.is_open())
581 {
582 return std::nullopt;
583 }
584
585 int port_value;
586 if (!(port_file >> port_value))
587 {
588 return std::nullopt;
589 }
590
591 if (port_value < 1024 || port_value > 65535)
592 {
593 return std::nullopt;
594 }
595
596 return static_cast<uint16_t>(port_value);
597 }
598 catch (...)
599 {
600 return std::nullopt;
601 }

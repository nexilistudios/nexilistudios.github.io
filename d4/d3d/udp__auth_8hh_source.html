50inline int hexValue(char c)
51{
52 if (c >= '0' && c <= '9')
53 return c - '0';
54 if (c >= 'a' && c <= 'f')
55 return c - 'a' + 10;
56 return -1;
57}

59inline std::string toHex(const uint8_t* bytes, size_t count)
60{
61 constexpr char digits[] = "0123456789abcdef";
62 std::string result;
63 result.reserve(count * 2);
64 for (size_t i = 0; i < count; ++i)
65 {
66 result += digits[bytes[i] >> 4];
67 result += digits[bytes[i] & 15];
68 }
69 return result;
70}

72inline bool fromHex(std::string_view hex, std::array<uint8_t, token_size>& bytes)
73{
74 if (hex.size() != token_size * 2)
75 return false;
76 for (size_t i = 0; i < token_size; ++i)
77 {
78 int high = hexValue(hex[i * 2]);
79 int low = hexValue(hex[i * 2 + 1]);
80 if (high < 0 || low < 0)
81 return false;
82 bytes[i] = static_cast<uint8_t>((high << 4) | low);
83 }
84 return true;
85}

112inline bool readSession(const nx_data& packet, std::string_view token, nx_data& payload)
113{
114 if (!hasPrefix(packet, session) || packet.size() < 4 + 8 + 8 + 2 + message_auth::tag_size)
115 return false;
116 nx_data signedPayload(packet.begin() + 4, packet.end());
117 nx_data verified;
118 if (!message_auth::verify(signedPayload, token, verified))
119 return false;
120 payload = std::move(signedPayload);
121 return true;
122}

124// A stateless, source-address-and-port-bound challenge. Only a peer that can
125// receive the challenge can complete initial UDP authentication. The reply is
126// exactly the same size as the required hello (20 bytes).
127inline std::optional<std::array<uint8_t, cookie_size>> cookieFor(
128 const std::array<uint8_t, 32>& key, std::string_view endpoint, uint64_t bucket)
129{
130 std::string input(endpoint);
131 input += ':';
132 input += std::to_string(bucket);
133 unsigned int digest_size = 0;
134 unsigned char digest[EVP_MAX_MD_SIZE]{};
135 if (!HMAC(EVP_sha256(), key.data(), static_cast<int>(key.size()),
136 reinterpret_cast<const unsigned char*>(input.data()), input.size(), digest, &digest_size) ||
137 digest_size < cookie_size)
138 return std::nullopt;
139 std::array<uint8_t, cookie_size> result{};
140 std::copy_n(digest, cookie_size, result.begin());
141 return result;
142}

146 return static_cast<uint64_t>(std::chrono::duration_cast<std::chrono::seconds>(

152inline nx_data makeChallenge(const std::array<uint8_t, 32>& key, std::string_view endpoint)
153{
154 auto cookie = cookieFor(key, endpoint, currentBucket());
155 if (!cookie)
156 return {};
157 nx_data packet(challenge.begin(), challenge.end());
158 packet.insert(packet.end(), cookie->begin(), cookie->end());
159 return packet;
160}

162inline bool checkAuthentication(const nx_data& packet, const std::array<uint8_t, 32>& key,
163 std::string_view endpoint, nx_data& payload)
164{
165 if (!hasPrefix(packet, authentication) || packet.size() <= 4 + cookie_size)
166 return false;
167 auto bucket = currentBucket();
168 auto current = cookieFor(key, endpoint, bucket);
169 auto previous = bucket > 0 ? cookieFor(key, endpoint, bucket - 1) : std::nullopt;
170 if (!current)
171 return false;
172 const auto* received = packet.data() + 4;
173 if (CRYPTO_memcmp(current->data(), received, cookie_size) != 0 &&
174 (!previous || CRYPTO_memcmp(previous->data(), received, cookie_size) != 0))
175 return false;
176 payload.assign(packet.begin() + 4 + cookie_size, packet.end());
177 return true;
178}

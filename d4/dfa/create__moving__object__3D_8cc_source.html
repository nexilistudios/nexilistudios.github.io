27CommandResult ServerImpl::room_object3d_createMoving(const DefaultArgs& args)
28{
29 auto& user = args.getUser();
30 auto data = args.getData();
31
32 // Get messagedata.
33 auto payload = Util::removeAmountOfBytesFromVector(data, 3);
34 auto startingPosition = Util::vector3fFromFront(payload);
35 auto dimensions = Util::vector3fFromFront(Util::removeAmountOfBytesFromVector(payload, 12));
36 auto movement = Util::vector3fFromFront(Util::removeAmountOfBytesFromVector(payload, 24));
37 auto deltaTime = Util::floatFromFront(Util::removeAmountOfBytesFromVector(payload, 36));
38 auto movementType = static_cast<MovementType>(payload[40]);
39 auto fileBytes = Util::removeAmountOfBytesFromVector(payload, 41);
40 auto filepath = Util::convertToString(fileBytes);
41
42 // Create new object.
43 auto object = nexilis::Object3D(Util::getRandomUint64(), startingPosition, dimensions);
44 object.setFilepath(filepath);
45 uint64_t objectId = object.getId();
46 Log::info("Created object with id: ", objectId);
47
48 // Add newly created object to storage.
49 auto room = RoomStorage::getRoomById(user.getRoomId());
50 room->addObject(std::move(object));
51
52 std::map<std::string, boost::json::value> params{
53 {"createMovingType", boost::json::value("create")},
54 {"x", boost::json::value(startingPosition.x)},
55 {"y", boost::json::value(startingPosition.y)},
56 {"z", boost::json::value(startingPosition.z)},
57 {"w", boost::json::value(dimensions.x)},
58 {"h", boost::json::value(dimensions.y)},
59 {"d", boost::json::value(dimensions.z)},
60 {"filepath", boost::json::value(filepath)},
61 {"id", boost::json::value(objectId)}};
62 auto roomCommand = Command::createRoomCommand(user.getRoomId(), user, data, params, args.getMessageId());
63 Command::sendRoomCommand(roomCommand, user, args.getProtocol());
64
65 std::function<double(double, double)> movementFunction;
66 switch (movementType)
67 {
68 case MovementType::eased:
69 movementFunction = [](double progress, double totalDistance) -> double
70 {
71 return Movement::easing(progress, totalDistance);
72 };
73 break;
74 case MovementType::linear:
75 movementFunction = [](double progress, double totalDistance) -> double
76 {
77 return Movement::linear(progress, totalDistance);
78 };
79 break;
80 default:
81 Log::error("Undefined movement type!");
82 }
83
84 auto movement_data = MovementData(objectId, deltaTime, data, args.getMessageId());
85 auto base_move = std::make_unique<Movement3D>(movement_data, movement, movementFunction);
86
87 Movement::object3D(std::move(base_move), user, args.getProtocol()).detach();
88 return CommandResult::success;
89}

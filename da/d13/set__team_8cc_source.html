26CommandResult ServerImpl::room_player3d_setTeam(const DefaultArgs& args)
27{
28 auto& user = args.getUser();
29 if (user.getRoomId() == 0)
30 {
31 Log::error("Set team: User not in room!");
32 return CommandResult::error;
33 }
34
35 auto payload = Util::removeAmountOfBytesFromVector(args.getData(), 3);
36 std::string team = Util::convertToString(payload);
37
38 if (team != "Terrorist" && team != "Counter Terrorist")
39 {
40 Log::warning("Set team: Unknown team '", team, "' from user ", user.getId());
41 return CommandResult::invalid_input;
42 }
43
44 auto room = RoomStorage::getRoomById(user.getRoomId());
45 if (!room)
46 {
47 Log::error("Set team: Room not found");
48 return CommandResult::failure;
49 }
50
51 if (!room->canSelectTeam(user.getId()))
52 return CommandResult::invalid_input;
53
54 room->setPlayerTeam(user.getId(), team);
55 Log::info("Player ", user.getId(), " [", user.getUsername(), "] joined team ", team);
56
57 // Announce the joining player's stats to the whole room so every client
58 // can keep its local leaderboard table up to date...
59 auto statsPayload = Command::createRoomLeaderboardCommand(
60 room->getId(), user,
61 Command::playerStatsEntries(*room, {user.getId()}), 0);
62 room->broadcastToAll(statsPayload);
63
64 // ...and seed the joining player with everyone currently in the room.
65 auto fullPayload = Command::createRoomLeaderboardCommand(
66 room->getId(), user,
67 Command::playerStatsEntries(*room, room->getClients()),
68 args.getMessageId());
69 Command::sendMessageToClient(fullPayload, user, args.getProtocol());
70
71 return CommandResult::success;
72}

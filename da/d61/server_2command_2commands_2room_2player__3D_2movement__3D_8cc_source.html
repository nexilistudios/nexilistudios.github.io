29CommandResult ServerImpl::room_player3d_movement(const DefaultArgs& args)
30{
31 auto& user = args.getUser();
32 auto& protocol = args.getProtocol();
33 auto messageId = args.getMessageId();
34 auto data = args.getData();
35
36 auto payload = Util::removeAmountOfBytesFromVector(data, 3);
37 if (payload.size() != 16 && payload.size() != 20)
38 return CommandResult::invalid_input;
39 auto vec_x = Util::floatFromFront(payload);
40 auto vec_y = Util::floatFromFront(Util::removeAmountOfBytesFromVector(payload, 4));
41 auto vec_z = Util::floatFromFront(Util::removeAmountOfBytesFromVector(payload, 8));
42 auto delta = Util::floatFromFront(Util::removeAmountOfBytesFromVector(payload, 12));
43 const float yaw = payload.size() == 20
44 ? Util::floatFromFront(Util::removeAmountOfBytesFromVector(payload, 16))
45 : 0.0f;
46 if (!std::isfinite(vec_x) || !std::isfinite(vec_y) || !std::isfinite(vec_z) ||
47 !std::isfinite(delta) || !std::isfinite(yaw))
48 return CommandResult::invalid_input;
49
50 auto* inputRoom = RoomStorage::getRoomById(user.getRoomId());
51 if (inputRoom && inputRoom->hasMovementInputHandler())
52 return inputRoom->onMovementInput(user.getId(), Vector3f(vec_x, vec_y, vec_z), yaw)
53 ? CommandResult::success
54 : CommandResult::invalid_input;
55
56 // Keep the movement speed inside the configured limit by
57 // scaling the velocity down to the maximum.
58 const auto& limits = GameplayLimits::get();
59 float speed = std::sqrt(vec_x * vec_x + vec_y * vec_y + vec_z * vec_z);
60 if (speed > limits.getMaxMovementSpeed())
61 {
62 float scale = limits.getMaxMovementSpeed() / speed;
63 vec_x *= scale;
64 vec_y *= scale;
65 vec_z *= scale;
66 Log::warning("Movement speed ", speed, " clamped to ", limits.getMaxMovementSpeed());
67 }
68 auto movement_vector = Vector3f(vec_x, vec_y, vec_z);
69 auto mtx = std::make_shared<std::mutex>();
70
71 // clang-format off
72 std::thread([mtx, movement_vector, limits, &user, data, &protocol, messageId, delta, yaw]()
73 {
74 try
75 {
76 auto* room = RoomStorage::getRoomById(user.getRoomId());
77 if (!room)
78 {
79 Log::warning("room_player3d_movement: room not found for user ", user.getRoomId());
80 return;
81 }
82
83 // The client sends a movement vector (velocity) together with the
84 // frame duration. Displacement for this packet is velocity * delta.
85 // Applying it exactly once (instead of via a tick-rate easing loop)
86 // guarantees the movement is applied even when delta is shorter than
87 // a single server tick.
88 if (delta <= 0.0f)
89 {
90 return;
91 }
92
93 auto current_pos = user.getObject3D().getPosition();
94 auto new_pos = Vector3f(current_pos.x + movement_vector.x * delta,
95 current_pos.y + movement_vector.y * delta,
96 current_pos.z + movement_vector.z * delta);
97
98 // Keep the resulting position inside the configured
99 // world limits: the horizontal coordinates and the
100 // height are clamped to the nearest allowed value.
101 Vector3f clamped_pos(limits.clampHorizontal(new_pos.x),
102 limits.clampHeight(new_pos.y),
103 limits.clampHorizontal(new_pos.z));
104 if (clamped_pos.x != new_pos.x || clamped_pos.y != new_pos.y ||
105 clamped_pos.z != new_pos.z)
106 {
107 Log::warning("Movement to (", new_pos.x, ", ", new_pos.y, ", ", new_pos.z,
108 ") clamped to (", clamped_pos.x, ", ", clamped_pos.y, ", ", clamped_pos.z, ")");
109 }
110
111 // Overlap validation: when the room forbids clients from sharing
112 // positions, drop this move if the new position would intersect
113 // another client. The position is not applied nor broadcast.
114 if (!room->isOverlappingAllowed())
115 {
116 auto dimensions = user.getObject3D().getDimensions();
117 for (auto clientId : room->getClients())
118 {
119 if (clientId == user.getId())
120 {
121 continue;
122 }
123 auto* other = ClientStorage::getClientById(clientId);
124 if (!other || other->getRoomId() != room->getId())
125 {
126 continue;
127 }
128 auto other_pos = other->getObject3D().getPosition();
129 auto other_dimensions = other->getObject3D().getDimensions();
130
131 bool intersects =
132 new_pos.x - dimensions.x / 2 < other_pos.x + other_dimensions.x / 2 &&
133 new_pos.x + dimensions.x / 2 > other_pos.x - other_dimensions.x / 2 &&
134 new_pos.y - dimensions.y / 2 < other_pos.y + other_dimensions.y / 2 &&
135 new_pos.y + dimensions.y / 2 > other_pos.y - other_dimensions.y / 2 &&
136 new_pos.z - dimensions.z / 2 < other_pos.z + other_dimensions.z / 2 &&
137 new_pos.z + dimensions.z / 2 > other_pos.z - other_dimensions.z / 2;
138 if (intersects)
139 {
140 return;
141 }
142 }
143 }
144
145 {
146 std::lock_guard<std::mutex> lock(*mtx);
147 user.getObject3D().setPosition(clamped_pos);
148 user.getObject3D().setYaw(yaw);
149 std::map<std::string, boost::json::value> params
150 {
151 {"x", boost::json::value(clamped_pos.x)},
152 {"y", boost::json::value(clamped_pos.y)},
153 {"z", boost::json::value(clamped_pos.z)},
154 {"yaw", boost::json::value(yaw)},
155 };
156 auto room_command = Command::createRoomCommand(user.getRoomId(), user, data, params, messageId);
157 Command::sendRoomCommand(room_command, user, protocol);
158 }
159 }
160 catch (const std::exception& e)
161 {
162 Log::error("Error with 3D movement thread: ", e.what());
163 }
164 catch (...)
165 {
166 Log::error("Other error with 3D movement thread");
167 }
168 })
169 .detach();
170 // clang-format on
171 return CommandResult::success;
172}

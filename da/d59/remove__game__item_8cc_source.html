24CommandResult ServerImpl::room_gameitem_destroy(const DefaultArgs& args)
25{
26 auto& user = args.getUser();
27 auto data = args.getData();
28 auto payload = Util::removeAmountOfBytesFromVector(data, 3);
29 if (payload.size() != 8)
30 return CommandResult::invalid_input;
31 auto itemId = Util::uint64FromFront(payload);
32
33 auto room = RoomStorage::getRoomById(user.getRoomId());
34 if (!room || !room->isClientGameItemCreationAllowed())
35 return CommandResult::failure;
36 room->deleteGameItem(itemId);
37
38 std::map<std::string, boost::json::value> params{
39 {"id", boost::json::value(itemId)}};
40
41 auto roomCommand = Command::createRoomCommand(user.getRoomId(), user, data, params, args.getMessageId());
42 if (Command::sendRoomCommand(roomCommand, user, args.getProtocol()))
43 return CommandResult::success;
44 else
45 return CommandResult::failed_room_send;
46}

35std::function<void()> create_wait_callback(nexilis_ClientAPI* client_api,
36 std::promise<void>& promise,
37 nexilis_WaitType wait_type)
38{
39 if (!client_api || !client_api->api)
40 {
41 return [&promise]()
42 {
43 promise.set_exception(std::make_exception_ptr(
44 std::runtime_error("Invalid ClientAPI")));
45 };
46 }
47
48 switch (wait_type)
49 {
50 case NEXILIS_WAIT_UNTIL_ROOMS_CREATED:
51 return [client_api, &promise]()
52 {
53 client_api->api->waitUntilRoomsCreated(promise);
54 };
55 default:
56 return [&promise]()
57 {
58 promise.set_exception(std::make_exception_ptr(
59 std::runtime_error("Unknown wait type")));
60 };
61 }
62}
63} // namespace

65nexilis_WaiterHandle* nexilis_waiter_create(nexilis_ClientAPI* client_api, nexilis_PromiseHandle* promise, nexilis_WaitType wait_type)
66{
67 if (!client_api || !promise)
68 return nullptr;
69
70 try
71 {
72 auto waiter = new nexilis_WaiterHandle();
73 waiter->client_api = client_api;
74 waiter->wait_type = wait_type;
75 waiter->callback = create_wait_callback(client_api, promise->promise, wait_type);
76 return waiter;
77 }
78 catch (...)
79 {
80 return nullptr;
81 }
82}

96void nexilis_waiter_wait(nexilis_WaiterHandle* waiter)
97{
98 if (!waiter)
99 return;
100
101 std::lock_guard<std::mutex> lock(waiter->mutex);
102 if (waiter->waited)
103 return;
104
105 if (waiter->callback)
106 {
107 try
108 {
109 waiter->callback();
110 waiter->waited = true;
111 }
112 catch (...)
113 {
114 // Handle exceptions
115 }
116 }
117}

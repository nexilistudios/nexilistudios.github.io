65std::unique_ptr<BaseAPICommand> CommandParser::parse(const boost::json::object& json)
66{
67 if (!json.contains("command") || !json.contains("type"))
68 {
69 return makeError(ReadResult::parsing_failed, "no \"command\" or \"type\" found!");
70 }
71
72 auto cmd_type = commandTypeFromString(json.at("command").as_string().c_str());
73 auto type = json.at("type").as_string();
74
75 switch (cmd_type)
76 {
77 case CommandType::setting:
78 return parseSettingCommand(json, type);
79 case CommandType::getting:
80 return parseGettingCommand(json, type);
81 case CommandType::room:
82 return parseRoomCommand(json, type);
83 case CommandType::error:
84 return parseErrorCommand(type);
85 case CommandType::authentication:
86 default:
87 return makeError(ReadResult::parsing_failed, "command parsing failed to execute");
88 }
89 return makeError(ReadResult::parsing_failed, "command type not found");
90}

92std::unique_ptr<BaseAPICommand> CommandParser::parseSettingCommand(const boost::json::object& json, std::string_view type)
93{
94 if (type == "username")
95 {
96 std::string username = json.at("username").as_string().c_str();
97
98 uint64_t client_id = 0;
99 if (json.contains("client_id"))
100 client_id = Util::toUint64(json.at("client_id"));
101
102 return std::make_unique<SetUsernameCommand>(client_id, username);
103 }
104 else if (type == "port")
105 {
106 std::string protocol = "boost_tcp";
107 if (json.contains("protocol"))
108 protocol = json.at("protocol").as_string().c_str();
109
110 uint16_t port = 0;
111 if (json.contains("port"))
112 port = static_cast<uint16_t>(Util::toUint64(json.at("port")));
113 else if (json.contains("boost_tcp_port"))
114 port = static_cast<uint16_t>(Util::toUint64(json.at("boost_tcp_port")));
115
116 return std::make_unique<SetPortCommand>(protocol, port);
117 }
118 return makeError(ReadResult::parsing_failed, "cannot find set command type");
119}

121std::unique_ptr<BaseAPICommand> CommandParser::parseGettingCommand(const boost::json::object& json, std::string_view type)
122{
123 if (type == "client_id")
124 {
125 uint64_t client_id = Util::toUint64(json.at("client_id"));
126 return std::make_unique<GetClientIdCommand>(client_id, getString(json, "udp_token"));
127 }
128
129 else if (type == "room_id")
130 {
131 uint64_t room_id = Util::toUint64(json.at("room_id"));
132 return std::make_unique<GetRoomIdCommand>(room_id);
133 }
134
135 else if (type == "info_rooms")
136 {
137 if (json.contains("rooms"))
138 {
139 return std::make_unique<GetInfoRoomsCommand>(json.at("rooms"));
140 }
141
142 return makeError(ReadResult::invalid_input);
143 }
144
145 else if (type == "info_general" && json.contains("client_amount"))
146 {
147 return std::make_unique<GetInfoClientCountCommand>(
148 Util::toUint64(json.at("client_amount")));
149 }
150
151 return makeError(ReadResult::parsing_failed, "cannot find get command type");
152}

154std::unique_ptr<BaseAPICommand> CommandParser::parseRoomCommand(const boost::json::object& json, std::string_view type)
155{
156 // Parse room commands based on 'type' and 'action' fields
157 if (!json.contains("action"))
158 {
159 return makeError(ReadResult::parsing_failed, "missing \"action\"");
160 }
161
162 std::string action = json.at("action").as_string().c_str();
163
164 // Common fields for all room commands
165 uint64_t room_id = getUint64(json, "room_id");
166 uint64_t client_id = getUint64(json, "client_id");
167
168 if (type == "match" && action == "state")
169 {
170 const std::string gameMode = getString(json, "game_mode");
171 if (gameMode == "deathmatch")
172 {
173 auto event = std::make_unique<ClientAPI::DeathmatchMatchEvent>();
174 event->room_id = room_id;
175 event->player_id = getUint64(json, "player_id");
176 event->alive = json.contains("alive") && json.at("alive").as_bool();
177 event->respawn_seconds = getFloat(json, "respawn_seconds");
178 return std::make_unique<RoomMatchCommand>(std::move(event));
179 }
180 if (gameMode.empty() || gameMode == "bomb")
181 {
182 auto event = std::make_unique<ClientAPI::BombMatchEvent>();
183 event->room_id = room_id;
184 event->round = getUint64(json, "round");
185 event->terrorist_score = getUint64(json, "terrorist_score");
186 event->counter_terrorist_score = getUint64(json, "counter_terrorist_score");
187 event->queue_position = getUint64(json, "queue_position");
188 event->phase = getString(json, "phase");
189 event->team = getString(json, "team");
190 event->notice = getString(json, "notice");
191 event->seconds = getFloat(json, "seconds");
192 event->bomb_x = getFloat(json, "bomb_x");
193 event->bomb_y = getFloat(json, "bomb_y");
194 event->bomb_z = getFloat(json, "bomb_z");
195 event->bomb_planted = json.contains("bomb_planted") && json.at("bomb_planted").as_bool();
196 event->active = json.contains("active") && json.at("active").as_bool();
197 event->alive = json.contains("alive") && json.at("alive").as_bool();
198 if (json.contains("alive_players") && json.at("alive_players").is_array())
199 for (const auto& id : json.at("alive_players").as_array())
200 event->alive_players.push_back(Util::toUint64(id));
201 return std::make_unique<RoomMatchCommand>(std::move(event));
202 }
203 return makeError(ReadResult::parsing_failed, "unknown match game mode");
204 }
205
206 if (type == "management")
207 {
208 return std::make_unique<RoomManagementCommand>(
209 action, room_id, client_id,
210 getString(json, "room_name"),
211 getUint64(json, "room_context"),
212 getString(json, "username"),
213 json.contains("overlap_allowed") ? json.at("overlap_allowed").as_bool() : true);
214 }
215 else if (type == "player_2D")
216 {
217 if (action == "shoot")
218 {
219 return std::make_unique<RoomPlayer2DShootCommand>(
220 client_id,
221 getFloat(json, "x"), getFloat(json, "y"));
222 }
223 return std::make_unique<RoomPlayer2DCommand>(action, client_id,
224 getFloat(json, "x"), getFloat(json, "y"));
225 }
226 else if (type == "player_3D")
227 {
228 if (action == "leaderboard")
229 {
230 boost::json::array entries;
231 if (json.contains("entries") && json.at("entries").is_array())
232 entries = json.at("entries").as_array();
233 return std::make_unique<RoomLeaderboardCommand>(entries);
234 }
235
236 if (action == "audio_event")
237 {
238 return std::make_unique<RoomAudioEventCommand>(
239 client_id, static_cast<uint8_t>(getUint64(json, "sound")),
240 getFloat(json, "x"), getFloat(json, "y"), getFloat(json, "z"));
241 }
242
243 if (action == "animation")
244 {
245 return std::make_unique<RoomAnimationEventCommand>(
246 room_id, client_id, getString(json, "state"),
247 getString(json, "weapon"), getFloat(json, "duration"));
248 }
249
250 return std::make_unique<RoomPlayer3DCommand>(
251 action, client_id,
252 getFloat(json, "x"), getFloat(json, "y"), getFloat(json, "z"),
253 getUint64(json, "target_id"),
254 getFloat(json, "damage"), getFloat(json, "new_health"),
255 getFloat(json, "yaw"));
256 }
257 else if (type == "object_2D")
258 {
259 return std::make_unique<RoomObject2DCommand>(
260 action, room_id,
261 getUint64(json, "id"),
262 getFloat(json, "x"), getFloat(json, "y"),
263 getFloat(json, "width"), getFloat(json, "height"),
264 getString(json, "filepath"),
265 getString(json, "createMovingType"));
266 }
267 else if (type == "object_3D")
268 {
269 return std::make_unique<RoomObject3DCommand>(
270 action, room_id,
271 getUint64(json, "id"),
272 getFloat(json, "x"), getFloat(json, "y"), getFloat(json, "z"),
273 getFloat(json, "w"), getFloat(json, "h"), getFloat(json, "d"),
274 getString(json, "filepath"),
275 getString(json, "createMovingType"));
276 }
277 else if (type == "game_item")
278 {
279 return std::make_unique<RoomGameItemCommand>(
280 action, room_id,
281 getUint64(json, "id"),
282 getFloat(json, "x"), getFloat(json, "y"), getFloat(json, "z"),
283 getFloat(json, "w"), getFloat(json, "h"), getFloat(json, "d"),
284 getString(json, "item_type"), getString(json, "status"),
285 getString(json, "filepath"));
286 }
287 else if (type == "communication")
288 {
289 return std::make_unique<RoomCommunicationCommand>(action, room_id, client_id,
290 getString(json, "message"));
291 }
292
293 return makeError<>(ReadResult::parsing_failed, "cannot find room command type");
294}

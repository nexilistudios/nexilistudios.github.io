26CommandResult ServerImpl::room_management_create(const DefaultArgs& args)
27{
28 auto payload = Util::removeAmountOfBytesFromVector(args.getData(), 3);
29 uint8_t context = payload[0];
30 std::string roomName = Util::convertToString(Util::removeAmountOfBytesFromVector(payload, 1));
31
32 if (roomName.empty())
33 {
34 Log::error("Room name cannot be empty");
35 return CommandResult::invalid_input;
36 }
37 else if (roomName == "")
38 {
39 Log::error("Room name cannot be an empty string");
40 return CommandResult::invalid_input;
41 }
42 else if (roomName == " ")
43 {
44 Log::error("Room name cannot be equal to \" \" ");
45 return CommandResult::invalid_input;
46 }
47 else
48 {
49 auto room_data = RoomData(args.getUser().getId(), roomName, Util::getRandomUint64(), static_cast<RoomData::Context>(context));
50 auto newRoom = nexilis::server::Room(room_data);
51
52 auto newRoomId = newRoom.getId();
53 RoomStorage::add(std::move(newRoom));
54
55 std::map<std::string, boost::json::value> params{
56 {"room_name", boost::json::value(roomName)},
57 {"client_id", boost::json::value(args.getUser().getId())},
58 {"action", boost::json::value("create")},
59 {"room_id", boost::json::value(newRoomId)},
60 {"room_context", boost::json::value(context)}};
61
62 auto data = Command::clientMessageData(CommandType::room, "management", args.getMessageId(), params);
63
64 if (Command::sendMessageToClient(data, args.getUser(), args.getProtocol()))
65 {
66 return CommandResult::success;
67 }
68 else
69 {
70 return CommandResult::failed_room_send;
71 }
72 }
73}
